
  // Dream Maker / Become — zero-credential CPU/WASM fallback.
  // This chunk is injected after become-live-02.txt and only replaces the
  // local generation dispatcher. WebGPU stays preferred; Transformers.js
  // running through WASM/CPU is used when WebGPU is unavailable or fails.
  const BECOME_CPU_TRANSFORMERS_URL = 'https://cdn.jsdelivr.net/npm/@huggingface/transformers@4.2.0';
  const BECOME_CPU_MODEL = 'onnx-community/SmolLM2-360M-Instruct-ONNX';
  let becomeCpuPipelinePromise = null;

  function bCpuProgress(info){
    if(!info) return;
    if(info.status === 'progress_total' && Number.isFinite(info.progress)){
      bLocalProgress(`CPU LOCAL MODEL ${Math.round(info.progress)}% · downloading`);
      return;
    }
    if(info.status === 'progress' && Number.isFinite(info.progress)){
      bLocalProgress(`CPU LOCAL MODEL · ${Math.round(info.progress)}%`);
      return;
    }
    if(info.status === 'ready') bLocalProgress('CPU local model ready. Generating scenario…');
  }

  async function bCpuPipeline(){
    if(!becomeCpuPipelinePromise){
      becomeCpuPipelinePromise=(async()=>{
        bLocalProgress('WebGPU unavailable. Loading CPU/WASM local AI…');
        const transformers=await import(BECOME_CPU_TRANSFORMERS_URL);
        const generator=await transformers.pipeline(
          'text-generation',
          BECOME_CPU_MODEL,
          {
            device:'wasm',
            dtype:'q4',
            progress_callback:bCpuProgress
          }
        );
        bLocalProgress('CPU/WASM local AI ready. Generating scenario…');
        return generator;
      })().catch(error=>{
        becomeCpuPipelinePromise=null;
        throw error;
      });
    }
    return becomeCpuPipelinePromise;
  }

  function bCpuExtractText(output){
    const generated=output?.[0]?.generated_text;
    if(typeof generated === 'string') return generated;
    if(Array.isArray(generated)){
      for(let i=generated.length-1;i>=0;i--){
        const item=generated[i];
        if(item?.role === 'assistant' && typeof item.content === 'string') return item.content;
      }
      const last=generated[generated.length-1];
      if(typeof last?.content === 'string') return last.content;
    }
    return '';
  }

  async function bCpuGenerate(index){
    const generator=await bCpuPipeline();
    const history=bLiveHistory(index);
    const rejected=[];
    for(let attempt=1;attempt<=3;attempt++){
      bLocalProgress(`CPU local generation · attempt ${attempt}/3`);
      const output=await generator(
        [
          {role:'system',content:'You are a concise scenario designer. Follow requested labelled-line formats exactly.'},
          {role:'user',content:bLocalPrompt(index,attempt,rejected)}
        ],
        {
          max_new_tokens:900,
          do_sample:true,
          temperature:0.9,
          top_p:0.95,
          repetition_penalty:1.08,
          return_full_text:true
        }
      );
      const text=bCpuExtractText(output);
      try{
        const scenario=bLocalParseLines(text);
        if(!bLocalTooSimilar(scenario,history)) return scenario;
        rejected.push({title:scenario.title,dimensions:scenario.dimensions});
      }catch(error){
        rejected.push({format_error:error instanceof Error?error.message:'invalid CPU local output'});
      }
    }
    throw new Error('CPU local AI could not produce a sufficiently distinct structured scenario after three attempts.');
  }

  const bLocalWebGPUGenerate = bLocalGenerate;
  bLocalGenerate = async function(index){
    if(navigator.gpu){
      try{
        return await bLocalWebGPUGenerate(index);
      }catch(error){
        bLocalProgress(`WebGPU local AI failed. Switching to CPU/WASM…`);
      }
    }
    return bCpuGenerate(index);
  };
