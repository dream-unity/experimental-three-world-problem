  const bClamp = (v,a,b) => Math.max(a,Math.min(b,v));
  const bMean = values => values.length ? values.reduce((a,b)=>a+b,0)/values.length : 0;
  const bEscape = value => String(value ?? '').replace(/[&<>"']/g, char => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[char]));
  const bShuffle = values => {
    const copy=[...values];
    for(let i=copy.length-1;i>0;i--){const j=(Math.random()*(i+1))|0;[copy[i],copy[j]]=[copy[j],copy[i]];}
    return copy;
  };

    const drill=bDrill();

    if(drill.field) return scenario[drill.field];
    const generic={

  }

  function bRefreshScore(){
    const values=[...bAllScores(),...Object.values(becomeState.scores),...Object.values(becomeState.transferScores)].filter(Number.isFinite);
    becomeE?.setScore(Math.round(bMean(values)*100));
    becomeE?.setMetric('DEPTH',values.length?bMean(values).toFixed(1):'—');
  }
  function bScore(value){
    value=bClamp(Math.round(value),1,10);
    const d=bDrill();
    if(becomeState.track==='scenario') becomeState.scores[d.id]=value; else becomeState.transferScores[d.id]=value;
    becomeE?.sfx('good',.75+value*.03);bRefreshScore();becomeState.drillIndex++;
    const length=becomeState.track==='scenario'?BECOME_DRILLS.length:BECOME_TRANSFER_DRILLS.length;
    if(becomeState.drillIndex<length){bRenderCue();return;}
    if(becomeState.track==='scenario'){
      becomeState.results.push({id:bScenario().id,title:bScenario().title,scores:{...becomeState.scores}});
      becomeState.scores={};bRenderScenarioSummary();
    }else bRenderEntryPrep();
  }
