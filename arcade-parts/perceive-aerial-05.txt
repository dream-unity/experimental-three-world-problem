      player.invulnerable = 0.68;
      streak = 0;
      E.setMetric('SHIELD', `${Math.ceil(player.shield)} · ×${streak}`);
      E.shake(9);
      E.flash([255, 72, 110], 0.22);
      E.burst(x, y, [255, 72, 110], 16, 175, 2.5);
      E.toast(copy || 'FORMATION HIT', 0.72);
      E.sfx('bad');
      if (player.shield <= 0) {
        writeProfile({ at: Date.now(), type: 'run', wave, level, ...stats });
        E.end('PERCEPTION LOST');
      }
    }

    function nearestInsideEscort() {
      return encounter.geometry?.inside?.[0]?.escort || null;
    }

    function classifyError(object) {
      if (!object) return 'empty-space-shot';
      if (object === leader) return 'reference-object-substitution';
      return 'relation-transfer-error';
    }

    function recordRoleOutcome(correct, errorType = null) {
      writeProfile({
        at: Date.now(),
        type: 'role-shot',
        wave,
        level,
        correct,
        errorType,
        roleSwitches: stats.roleSwitches,
        reacquisitionLatency: encounter.reacquired
          ? stats.reacquisitionLatencies.at(-1) ?? null
          : null,
        targetStableFor: Number(encounter.targetStableFor.toFixed(3)),
        turnSign: encounter.committedTurnSign,
        shots: stats.shots,
        firingAccuracy: stats.shots ? stats.hits / stats.shots : 0,
      });
    }

    function completeFormation(E) {
      encounter.solved = true;
      leader.alive = false;
      completedFormations++;
      const bonus = 620 + level * 90 + streak * 80;
      E.addScore(bonus, leader.sx, leader.sy, `FORMATION +${bonus}`);
      E.burst(leader.sx, leader.sy, [255, 215, 126], 30, 250, 3.6);
      E.flash(E.rgb, 0.22);
      E.shake(11);
      E.sfx('good', 1.13);
      E.toast('SHIELD NETWORK COLLAPSED', 1.0);
      resolveTimer = 0.9;
      wave++;
      const nextLevel = Math.min(10, 1 + Math.floor(completedFormations / 2));
      if (nextLevel > level) {
        level = nextLevel;
        E.level(`PERCEIVE ${level}`);
      }
      writeProfile({
        at: Date.now(),
        type: 'formation',
        completed: true,
        level,
        roleSwitches: stats.roleSwitches,
        medianReacquisition: stats.reacquisitionLatencies.length
          ? [...stats.reacquisitionLatencies].sort((a, b) => a - b)[Math.floor(stats.reacquisitionLatencies.length / 2)]
          : null,
        errors: stats.errors,
      });
    }

    function resolveCorrect(E, escort) {
      if (!escort?.alive || encounter.solved || encounter.failed) return;
      escort.alive = false;
      encounter.hits++;
      streak++;
      stats.correct++;
      stats.hits++;
      const reward = 300 + level * 42 + streak * 66;
      E.addScore(reward, escort.sx, escort.sy, streak > 2 ? `FOCUS ×${streak}` : `RELAY +${reward}`);
      E.burst(escort.sx, escort.sy, E.secondary, 22, 225, 3);
      E.flash(E.rgb, 0.16);
      E.shake(7);
      E.sfx('good', 1 + Math.min(streak, 8) * 0.032);
      recordRoleOutcome(true);
      encounter.previousTarget = escort;
      encounter.target = null;
      encounter.candidate = null;
      encounter.candidateTimer = 0;
      encounter.targetStableFor = 0;
      encounter.roleSwitchAt = encounter.age;
      encounter.reticleHold = 0;
      encounter.reacquired = false;
      E.setMetric('SHIELD', `${Math.ceil(player.shield)} · ×${streak}`);

      if (encounter.hits >= encounter.config.hitsRequired || activeEscorts().length < 3) completeFormation(E);
      else {
        newRelationalFormation();
        updateRelationalRole(0, true);
        E.toast('NEW FORMATION · SAME HIDDEN RELATION', 0.75);
      }
    }
