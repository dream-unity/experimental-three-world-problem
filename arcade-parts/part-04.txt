        if (visible) {
          E.glow(c, cx, cy, playerRadius * 2.5, formColors[form], .92);

  function createGravityFoundry() {
    let core = { x: 0, y: 0, tx: 0, ty: 0, r: 20 };
    let particlesMatter = [];
    let gravityOn = false;
    let stability = 100;
    let mass = 0;
    let targetMass = 7;
    let required = 0;
    let level = 1;
    let sequence = 0;
    const matterColors = [[92, 222, 255], [98, 239, 165], [210, 142, 255], [255, 72, 110]];

    function makeParticle(E, atEdge = false) {
      const typeRoll = Math.random();
      const type = typeRoll < .14 ? 3 : (Math.random() * 3) | 0;
      const margin = 18;
      let x, y;
      if (atEdge) {
        const side = (Math.random() * 4) | 0;
        if (side === 0) { x = margin; y = rand(E.bottom, E.top); }
        else if (side === 1) { x = E.w - margin; y = rand(E.bottom, E.top); }
        else if (side === 2) { x = rand(E.w - margin, margin); y = E.top + margin; }
        else { x = rand(E.w - margin, margin); y = E.bottom - margin; }
      } else {
        x = rand(E.w - margin, margin);
        y = rand(E.bottom - margin, E.top + margin);
      }
      const a = Math.random() * TAU;
      const speed = 18 + Math.random() * 42;
      return { x, y, vx: Math.cos(a) * speed, vy: Math.sin(a) * speed, type, r: type === 3 ? 5.5 : 4 + Math.random() * 3, phase: Math.random() * TAU, trail: [] };
    }

    function fill(E) {
      const count = coarse || lowCPU ? 38 : 58;
      particlesMatter = [];
      for (let i = 0; i < count; i++) particlesMatter.push(makeParticle(E));
    }

    function setCore(E) {
      core.x = core.tx = E.w * .5;
      core.y = core.ty = E.h * .52;
      core.r = Math.max(18, E.u * .03);
    }

    function nextRequirement(E) {
      required = (required + 1 + ((Math.random() * 2) | 0)) % 3;
      sequence++;
      E.toast(['CYAN', 'GREEN', 'VIOLET'][required], .65);
    }

    function capture(E, p) {
      const rgb = matterColors[p.type];
      if (p.type === required) {
        mass++;
        const reward = 65 + mass * 12 + level * 10;
        E.addScore(reward, core.x, core.y, `MASS ${mass}/${targetMass}`);
        E.burst(core.x, core.y, rgb, 12, 140, 2.5);
        E.sfx('good', .78 + p.type * .12);
        nextRequirement(E);
        if (mass >= targetMass) {
          level++;
          mass = 0;
          targetMass = Math.min(18, 7 + level * 2);
          stability = Math.min(100, stability + 18);
          E.setMetric('STABILITY', Math.round(stability));
          E.flash([255, 239, 180], .2);
          E.burst(core.x, core.y, [255, 239, 180], 38, 280, 3.4);
          E.level(`CORE ${level}`);
        }
      } else {
        const damage = p.type === 3 ? 22 : 8;
        stability -= damage;
        E.setMetric('STABILITY', Math.max(0, Math.round(stability)));
        E.shake(p.type === 3 ? 12 : 5);
        E.flash([255, 72, 110], p.type === 3 ? .27 : .12);
        E.burst(core.x, core.y, [255, 72, 110], p.type === 3 ? 20 : 8, 190, 2.8);
        E.sfx('bad', p.type === 3 ? .7 : 1);
        if (stability <= 0) E.end('MATTER UNBOUND');
      }
    }

    return {
      reset(E) {
        level = 1; stability = 100; mass = 0; targetMass = 7; required = (Math.random() * 3) | 0; sequence = 0; gravityOn = false;
        setCore(E); fill(E);
        E.setMetric('STABILITY', 100);
      },
      resize(E) { setCore(E); fill(E); },
      update(E, dt) {
        core.x = lerp(core.x, core.tx, 1 - Math.exp(-dt * 11));
        core.y = lerp(core.y, core.ty, 1 - Math.exp(-dt * 11));
        const force = gravityOn ? 115000 : 14000;
        for (let i = 0; i < particlesMatter.length; i++) {
          const p = particlesMatter[i];
          const dx = core.x - p.x, dy = core.y - p.y;
          const d2 = dx * dx + dy * dy;
          const d = Math.sqrt(d2) || 1;
          const accel = force / (d2 + 900);
          p.vx += dx / d * accel * dt;
          p.vy += dy / d * accel * dt;
          const speed = Math.hypot(p.vx, p.vy);
          const maxSpeed = gravityOn ? 280 : 120;
          if (speed > maxSpeed) { p.vx = p.vx / speed * maxSpeed; p.vy = p.vy / speed * maxSpeed; }
          p.x += p.vx * dt;
          p.y += p.vy * dt;
          p.vx *= Math.exp(-dt * .08);
          p.vy *= Math.exp(-dt * .08);
          if (p.x < 8 || p.x > E.w - 8) { p.vx *= -.85; p.x = clamp(p.x, 8, E.w - 8); }
          if (p.y < E.top + 8 || p.y > E.bottom - 8) { p.vy *= -.85; p.y = clamp(p.y, E.top + 8, E.bottom - 8); }
          p.trail.push({ x: p.x, y: p.y });
          if (p.trail.length > 7) p.trail.shift();
          if (d < core.r + p.r + 4) {
            capture(E, p);
            particlesMatter[i] = makeParticle(E, true);
          }
        }
      },
      pointerDown(E, x, y) {
        gravityOn = true;
        core.tx = clamp(x, 25, E.w - 25);
        core.ty = clamp(y, E.top + 25, E.bottom - 25);
        E.sfx('launch', .55);
      },
      pointerMove(E, x, y, down) {
        if (!down) return;
        gravityOn = true;
        core.tx = clamp(x, 25, E.w - 25);
        core.ty = clamp(y, E.top + 25, E.bottom - 25);
      },
      pointerUp() { gravityOn = false; },
      pointerCancel() { gravityOn = false; },
      draw(E, c) {
        c.save();
        c.globalCompositeOperation = 'lighter';
        const requiredRgb = matterColors[required];
        for (let r = core.r * 2; r < E.u * .34; r += core.r * 1.35) {
          const alpha = (gravityOn ? .13 : .045) * (1 - r / (E.u * .38));
          c.strokeStyle = rgba(requiredRgb, Math.max(.01, alpha));
          c.lineWidth = gravityOn ? 1.3 : .8;
          c.beginPath(); c.arc(core.x, core.y, r + Math.sin(E.t * 2 + r) * 3, 0, TAU); c.stroke();
        }

        for (const p of particlesMatter) {
          const rgb = matterColors[p.type];
          for (let i = 1; i < p.trail.length; i++) {
            const a = i / p.trail.length;
            E.line(c, p.trail[i - 1].x, p.trail[i - 1].y, p.trail[i].x, p.trail[i].y, rgb, a * .12, 1);
          }
          if (gravityOn && dist(p.x, p.y, core.x, core.y) < E.u * .23) E.line(c, p.x, p.y, core.x, core.y, rgb, .055, .7);
          E.glow(c, p.x, p.y, p.r * (p.type === required ? 1.8 : 1.2), rgb, p.type === required ? .7 : .38);
          c.fillStyle = p.type === 3 ? 'rgba(255,255,255,.28)' : rgba(rgb, .88);
          c.beginPath(); c.arc(p.x, p.y, p.r, 0, TAU); c.fill();
          if (p.type === required) {
            c.strokeStyle = rgba(rgb, .55); c.lineWidth = 1;
            c.beginPath(); c.arc(p.x, p.y, p.r * 2.2, E.t + p.phase, E.t + p.phase + Math.PI * 1.25); c.stroke();
          }
        }

        E.glow(c, core.x, core.y, core.r * (gravityOn ? 3 : 2.2), requiredRgb, gravityOn ? .92 : .7);
        const gradient = c.createRadialGradient(core.x - core.r * .25, core.y - core.r * .25, 1, core.x, core.y, core.r);
        gradient.addColorStop(0, 'rgba(255,255,255,.96)');
        gradient.addColorStop(.3, rgba(requiredRgb, .86));
        gradient.addColorStop(1, rgba(requiredRgb, .15));
        c.fillStyle = gradient;
        c.beginPath(); c.arc(core.x, core.y, core.r, 0, TAU); c.fill();
        c.strokeStyle = rgba(requiredRgb, .72); c.lineWidth = 2;
        c.beginPath(); c.arc(core.x, core.y, core.r * 1.45, -Math.PI / 2, -Math.PI / 2 + TAU * mass / targetMass); c.stroke();
        E.drawShape(c, required, core.x, core.y, core.r * .38, [255,255,255], .8, false, E.t * .3);
        c.restore();
      }
    };
  }

  function createLatticeLock() {
    const N = 1, E = 2, S = 4, WBIT = 8;
    const DIRS = [
      { bit: N, opposite: S, dr: -1, dc: 0 },
      { bit: E, opposite: WBIT, dr: 0, dc: 1 },
      { bit: S, opposite: N, dr: 1, dc: 0 },
      { bit: WBIT, opposite: E, dr: 0, dc: -1 }
    ];
    let rows = 5, cols = 5, grid = [];
    let source = 0, target = 0;
    let powered = new Set();
    let level = 1, lives = 3, timer = 0, phase = 'play', resultTimer = 0;
    let board = { x: 0, y: 0, size: 0, cell: 0 };

    const rotateMask = (mask) => ((mask << 1) & 15) | ((mask >> 3) & 1);
    const index = (r, c) => r * cols + c;

    function connectMasks(masks, a, b) {
      const ar = Math.floor(a / cols), ac = a % cols;
      const br = Math.floor(b / cols), bc = b % cols;
      if (br === ar - 1) { masks[a] |= N; masks[b] |= S; }
      else if (br === ar + 1) { masks[a] |= S; masks[b] |= N; }
      else if (bc === ac - 1) { masks[a] |= WBIT; masks[b] |= E; }
      else if (bc === ac + 1) { masks[a] |= E; masks[b] |= WBIT; }
    }

    function layout(E) {
      const availableH = E.bottom - E.top - 38;
      const maxSize = Math.min(E.w * .86, availableH * .94, E.u * .78);
      board.size = maxSize;
      board.cell = maxSize / Math.max(rows, cols);
      board.x = (E.w - cols * board.cell) * .5;
      board.y = E.top + (availableH - rows * board.cell) * .5 + 20;
    }

    function build(E) {
      rows = cols = Math.min(7, 5 + Math.floor((level - 1) / 3));
      layout(E);
      const masks = new Array(rows * cols).fill(0);
      let r = rows - 1;
      let c = Math.floor(cols / 2);
      source = index(r, c);
      const path = [source];
      const used = new Set(path);
      while (r > 0) {
        if (Math.random() < .72) {
          const candidates = [];
          if (c > 0 && !used.has(index(r, c - 1))) candidates.push(c - 1);
          if (c < cols - 1 && !used.has(index(r, c + 1))) candidates.push(c + 1);
          if (candidates.length) {
            c = choose(candidates);
            path.push(index(r, c));
            used.add(index(r, c));
          }
        }
        r--;
        path.push(index(r, c));
        used.add(index(r, c));
        if (Math.random() < .35 && r > 0) {
          const candidates = [];
          if (c > 0 && !used.has(index(r, c - 1))) candidates.push(c - 1);
          if (c < cols - 1 && !used.has(index(r, c + 1))) candidates.push(c + 1);
          if (candidates.length) {
            c = choose(candidates);
            path.push(index(r, c));
            used.add(index(r, c));
          }
        }
      }
      target = pa
