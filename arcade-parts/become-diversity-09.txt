  function d9SigDistance(a,b){
    let sum=0;
    for(const key of D9_AXES)if(String(a?.[key]||'')!==String(b?.[key]||''))sum+=D9_WEIGHTS[key];
    return sum/D9_WEIGHT_TOTAL;
  }
  function d9Candidate(world){
    const role=d9Pick(world.roles),goal=d9Pick(world.goals),pressure=d9Pick(world.pressures);
    const signature={...world.sig,family:world.id};
    const scenario={
      id:`d9-${world.id}-${Date.now().toString(36)}-${Math.random().toString(36).slice(2,8)}`,
      title:d9Pick(world.titles),tag:`MAX-DISTANCE · ${world.sig.worldClass.toUpperCase().replace(/-/g,' ')}`,visual:world.visual,
      premise:`You are the ${role} in ${d9Pick(world.settings)}. ${d9Pick(world.twists)}. Your task is to ${goal}.`,
      sensory:`See ${d9Pick(world.sights)}. Hear ${d9Pick(world.sounds)}.`,objects:`Feel ${d9Pick(world.objects)}.`,body:`Locate yourself ${d9Pick(world.bodies)}.`,
      atmosphere:`Let the scene carry ${d9Pick(world.atmospheres)}.`,motion:`Experience yourself ${d9Pick(world.motions)}.`,stakes:`${d9Pick(world.stakes)}.`,
      identity:`Inhabit the ${role} who ${d9Pick(world.identities)}.`,choice:`Choose now: ${d9Pick(world.choices)}. Commit internally to one option and experience its first consequence.`,
      exit:'Dissolve this constructed world completely. Feel your actual hands and feet, name three real objects, and state your present location.',
      dimensions:{environment:`${world.id}: ${d9Pick(world.settings)}`,role,goal,pressure,body_dynamics:world.sig.bodyClass,decision_structure:world.sig.decisionClass,emotional_tone:world.sig.toneClass,social_structure:world.sig.socialClass},
      __signature:signature,__source:'max-distance-synthesis'
    };
    return scenario;
  }
  function d9Summary(s){return{title:String(s.title||'').slice(0,100),premise:String(s.premise||'').slice(0,360),signature:{...(s.__signature||{})}};}
  function d9MemoryRead(){
    try{const raw=JSON.parse(localStorage.getItem(D9_MEMORY_KEY)||'[]');return Array.isArray(raw)?raw.filter(x=>x&&x.signature).slice(-D9_MEMORY_LIMIT):[];}catch{return [];}
  }
  function d9MemoryWrite(items){try{localStorage.setItem(D9_MEMORY_KEY,JSON.stringify(items.slice(-D9_MEMORY_LIMIT)));}catch{}}
  function d9AxisDiffCount(a,b){let n=0;for(const key of D9_AXES)if(String(a?.[key]||'')!==String(b?.[key]||''))n++;return n;}
  function d9Distance(candidate,ref){
    const structural=d9SigDistance(candidate.__signature,ref.signature);
    const lexical=1-d9Jaccard(d9Words(candidate.title+' '+candidate.premise),d9Words((ref.title||'')+' '+(ref.premise||'')));
    return structural*0.84+lexical*0.16;
  }
  function d9CandidateScore(candidate,refs,familyUse){
    if(!refs.length)return 1.25-(familyUse.get(candidate.__signature.family)||0)*0.08+Math.random()*0.01;
    const distances=refs.map(ref=>d9Distance(candidate,ref));
    const min=Math.min(...distances),avg=distances.reduce((a,b)=>a+b,0)/distances.length;
    const last=refs.slice(-D9_RECENT_HARD);
    const hardMinAxes=last.length?Math.min(...last.map(ref=>d9AxisDiffCount(candidate.__signature,ref.signature))):D9_AXES.length;
    const lexicalMax=last.length?Math.max(...last.map(ref=>d9Jaccard(d9Words(candidate.title+' '+candidate.premise),d9Words((ref.title||'')+' '+(ref.premise||''))))):0;
    const recentSameFamily=last.some(ref=>ref.signature?.family===candidate.__signature.family);
    const reused=familyUse.get(candidate.__signature.family)||0;
    let score=min*0.62+avg*0.25+(hardMinAxes/D9_AXES.length)*0.18-lexicalMax*0.18-reused*0.025;
    if(recentSameFamily)score-=0.42;
    if(hardMinAxes<8)score-=0.28;
    return score+Math.random()*0.002;
  }
  function d9BuildDiverseQueue(count){
    const started=typeof performance!=='undefined'?performance.now():Date.now();
    const memory=d9MemoryRead();
    const refs=[...memory];
    const selected=[];
    const usedSessionFamilies=new Set();
    const familyUse=new Map();
    for(const ref of memory)if(ref.signature?.family)familyUse.set(ref.signature.family,(familyUse.get(ref.signature.family)||0)+1);
    for(let slot=0;slot<count;slot++){
      const pool=[];
      for(const world of D9_WORLDS){
        if(usedSessionFamilies.has(world.id))continue;
        for(let i=0;i<D9_VARIANTS_PER_WORLD;i++)pool.push(d9Candidate(world));
      }
      if(!pool.length)throw new Error('Diversity engine exhausted world families.');
      let best=null,bestScore=-Infinity;
      for(const candidate of pool){const score=d9CandidateScore(candidate,refs,familyUse);if(score>bestScore){best=candidate;bestScore=score;}}
      best.__noveltyScore=Math.max(0,Math.min(1,bestScore));
      selected.push(best);usedSessionFamilies.add(best.__signature.family);refs.push(d9Summary(best));
      familyUse.set(best.__signature.family,(familyUse.get(best.__signature.family)||0)+1);
    }
    d9MemoryWrite([...memory,...selected.map(d9Summary)]);
    const ended=typeof performance!=='undefined'?performance.now():Date.now();
    d9BuildDiverseQueue.lastMs=Math.max(0,ended-started);
    return selected;
  }
