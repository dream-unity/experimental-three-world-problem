  function s14ImagineScene(scene){
    let text=String(scene||'').trim();
    if(S14_SCENE_OPENING.test(text)) return s13Limit(text,S13_SCENE_MAX);
    if(/^You are\b/.test(text)) text=`Imagine that you're${text.slice(7)}`;
    else if(/^You're\b/.test(text)) text=`Imagine that you're${text.slice(6)}`;
    else if(/^You\b/.test(text)) text=`Imagine that you${text.slice(3)}`;
    else text=`Imagine that you're experiencing this: ${text}`;
    return s13Limit(text,S13_SCENE_MAX);
  }

  function s14ApplySceneOpening(scenario){
    const rapid=scenario?.__rapid13||{};
    const fixedWords=s13Count(`${rapid.self||''} ${rapid.other||''} ${rapid.outside||''} ${rapid.act||''}`);
    const sceneBudget=Math.max(14,Math.min(S13_SCENE_MAX,S13_PACKET_MAX-fixedWords));
    const scene=s13Limit(s14ImagineScene(rapid.scene||scenario?.premise||''),sceneBudget);
    const totalWords=s13Count(`${scene} ${rapid.self||''} ${rapid.other||''} ${rapid.outside||''} ${rapid.act||''}`);
    scenario.premise=scene;
    scenario.__rapid13={...rapid,scene,totalWords};
    scenario.__sceneOpening14='imagine-that-you';
    return scenario;
  }

  const s14PriorApply=s10ApplySocial;
  s10ApplySocial=function(base,profile){
    const scenario=s14ApplySceneOpening(s14PriorApply(base,profile));
    const win=s13Limit(S14_WINS[profile.id]||'Preserve both perspectives while reaching one shared next step.',S14_WIN_MAX);
    scenario.__activation14={version:S14_VERSION,profileId:profile.id,win};
    scenario.__socialAgency={...scenario.__socialAgency,socialWin:win};
    return scenario;
  };

  function s14ObjectiveFor(drillId,scenario){
    const raw=S14_OBJECTIVES[drillId]||{mode:'phenomenological-change',objective:'Change something inside the lived simulation, not just your description.',solved:'The imagined state becomes detectably different.'};
    return{
      mode:raw.mode,
      objective:s13Limit(raw.objective,S14_OBJECTIVE_MAX),
      solved:s13Limit(raw.solved,S14_SOLVED_MAX),
      win:s13Limit(scenario?.__activation14?.win||'Reach one social outcome without erasing either perspective.',S14_WIN_MAX)
    };
  }

  };
