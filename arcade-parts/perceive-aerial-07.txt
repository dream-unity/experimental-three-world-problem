
    return {
      reset(E) {
        resetUiDefinition();
        wave = 1;
        level = 1;
        streak = 0;
        completedFormations = 0;
        resolveTimer = 0;
        stats = {
          correct: 0,
          wrong: 0,
          shots: 0,
          hits: 0,
          roleMigrations: 0,
          acquisitionLatencies: [],
          errors: {},
        };
        player = {
          x: E.w * 0.5,
          y: E.bottom - Math.max(58, E.u * 0.092),
          targetX: E.w * 0.5,
          targetY: E.bottom - Math.max(58, E.u * 0.092),
          size: Math.max(18, E.u * 0.035),
          bank: 0,
          shield: 5,
          heat: 0,
          fireCooldown: 0,
          invulnerable: 0,
        };
        reticle = { x: E.w * 0.5, y: E.top + E.h * 0.39 };
        pointerGesture = null;
        E.setMetric('SHIELD', '5 · ×0');
        createEncounter(E);
      },

      resize(E) {
        if (!player) return;
        player.size = Math.max(18, E.u * 0.035);
        player.x = clamp(player.x, 28, E.w - 28);
        player.y = clamp(player.y, E.h * 0.59, E.bottom - 18);
        player.targetX = clamp(player.targetX, 28, E.w - 28);
        player.targetY = clamp(player.targetY, E.h * 0.59, E.bottom - 18);
        reticle.x = clamp(reticle.x, 18, E.w - 18);
        reticle.y = clamp(reticle.y, E.top + 18, E.bottom - 42);
        if (encounter) updateFormation(E, 0);
      },

      ambient(E, dt) {
        if (!player) return;
        if (!encounter) createEncounter(E);
        updateFormation(E, dt * 0.2);
      },

      update(E, dt) {
        if (!player || !encounter) return;
        updatePlayer(E, dt);
        updateFormation(E, dt);
        updateCombat(E, dt);
        if (!encounter.solved && !encounter.failed && encounter.depth >= 1.04) failEncounter(E);
        if (encounter.solved || encounter.failed) {
          resolveTimer -= dt;
          if (resolveTimer <= 0 && player.shield > 0) createEncounter(E);
        }
      },

      pointerDown(E, x, y) {
        pointerGesture = { x, y, lastX: x, lastY: y, moved: false, startedAt: performance.now() };
        reticle.x = x;
        reticle.y = y;
        if (player) {
          player.targetX = x;
          player.targetY = clamp(y + E.u * 0.17, E.h * 0.59, E.bottom - 20);
        }
      },

      pointerMove(E, x, y, down) {
        reticle.x = x;
        reticle.y = y;
        if (down && player) {
          player.targetX = x;
          player.targetY = clamp(y + E.u * 0.17, E.h * 0.59, E.bottom - 20);
        }
        if (pointerGesture && down) {
          if (dist(pointerGesture.x, pointerGesture.y, x, y) > 12) pointerGesture.moved = true;
          pointerGesture.lastX = x;
          pointerGesture.lastY = y;
        }
      },

      pointerUp(E) {
        if (pointerGesture) {
          const duration = performance.now() - pointerGesture.startedAt;
          const tap = !pointerGesture.moved && duration < 430;
          reticle.x = pointerGesture.lastX;
          reticle.y = pointerGesture.lastY;
          if (tap) fire(E);
        }
        pointerGesture = null;
      },

      pointerCancel() { pointerGesture = null; },

      keyDown(E, key) {
        if (key === ' ') fire(E);
      },
