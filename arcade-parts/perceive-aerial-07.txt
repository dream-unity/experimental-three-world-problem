        stats.demos++;
        resolveTimer = 0.48;
        wave++;
        return;
      }
      stats.byLaw[scene.law] ||= { correct: 0, wrong: 0, withheld: 0, missed: 0 };
      if (scene.positive && scene.activeEver) {
        stats.missed++;
        stats.byLaw[scene.law].missed++;
        stats.errors['late-causal-acquisition'] = (stats.errors['late-causal-acquisition'] || 0) + 1;
        recordOutcome(false, 'late-causal-acquisition', null);
        hurt(E, 0.9, scene.jets[0]?.sx ?? E.w * 0.5, scene.jets[0]?.sy ?? E.h * 0.35, 'OPENING MISSED');
      } else {
        stats.withheld++;
        stats.byLaw[scene.law].withheld++;
        const reward = 70 + level * 8;
        E.addScore(reward, E.w * 0.5, E.top + E.h * 0.3, '+CONTROL');
      }
      episodeHistory.push(scene.positive);
      if (episodeHistory.length > 8) episodeHistory.shift();
      resolveTimer = 0.52;
      wave++;
    }

    function updateEpisodeLifecycle(E, dt) {
      if (!scene) return;
      if (!scene.resolved && scene.demoKind && scene.demoShotDone && scene.age >= scene.demoResolveAt) {
        resolveEpisodeWithoutShot(E);
      }
      if (!scene.resolved && scene.age >= scene.config.duration) resolveEpisodeWithoutShot(E);
      if (scene.resolved) {
        resolveTimer -= dt;
        if (resolveTimer <= 0 && player.shield > 0) createEpisode(E);
      }
