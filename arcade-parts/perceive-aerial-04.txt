
    function drawFighter(c, x, y, size, heading, bank, rgb, alpha = 1, pulse = 0.5) {
      c.save();
      c.translate(x, y);
      c.rotate(Math.PI + heading + bank * 0.14);
      c.globalCompositeOperation = 'lighter';
      c.globalAlpha = alpha;

      const glow = c.createRadialGradient(0, 0, 0, 0, 0, size * 2.35);
      glow.addColorStop(0, rgba(rgb, 0.1 + pulse * 0.045));
      glow.addColorStop(1, 'rgba(0,0,0,0)');
      c.fillStyle = glow;
      c.fillRect(-size * 2.4, -size * 2.4, size * 4.8, size * 4.8);

      c.fillStyle = rgba(rgb, 0.12 + pulse * 0.05);
      c.strokeStyle = rgba(rgb, 0.78);
      c.lineWidth = Math.max(1, size * 0.068);
      c.beginPath();
      c.moveTo(0, -size * 1.4);
      c.lineTo(size * 0.22, -size * 0.48);
      c.lineTo(size * 1.08, size * 0.12);
      c.lineTo(size * 0.45, size * 0.25);
      c.lineTo(size * 0.34, size * 1.02);
      c.lineTo(size * 0.1, size * 0.7);
      c.lineTo(0, size * 1.24);
      c.lineTo(-size * 0.1, size * 0.7);
      c.lineTo(-size * 0.34, size * 1.02);
      c.lineTo(-size * 0.45, size * 0.25);
      c.lineTo(-size * 1.08, size * 0.12);
      c.lineTo(-size * 0.22, -size * 0.48);
      c.closePath();
      c.fill();
      c.stroke();

      c.fillStyle = 'rgba(205,244,255,.68)';
      c.beginPath();
      c.ellipse(0, -size * 0.4, size * 0.15, size * 0.36, 0, 0, TAU);
      c.fill();

      c.fillStyle = rgba(rgb, 0.5 + pulse * 0.34);
      for (const side of [-1, 1]) {
        c.beginPath();
        c.ellipse(side * size * 0.18, size * 0.92, size * 0.085, size * (0.18 + pulse * 0.18), 0, 0, TAU);
        c.fill();
      }
      c.restore();
    }

    function drawFlightField(E, c) {
      const horizon = E.top + E.h * 0.2;
      c.save();
      c.globalCompositeOperation = 'lighter';
      const flow = (E.t * 0.34) % 1;
      for (let index = 0; index < (coarse || lowCPU ? 10 : 16); index += 1) {
        const p = (index / 16 + flow) % 1;
        const angle = index * 2.39;
        const radius = E.u * (0.08 + (index % 6) * 0.055);
        const x = E.w * 0.5 + Math.cos(angle) * radius * p;
        const y = horizon + p * p * (E.bottom - horizon);
        const length = 3 + p * 15;
        E.line(c, x, y, x + Math.cos(angle) * length, y + Math.sin(angle) * length, E.secondary, 0.035 + p * 0.12, 0.65 + p);
      }
      c.restore();
    }

    function drawTrails(E, c) {
      c.save();
      c.globalCompositeOperation = 'lighter';
      for (const jet of allEnemyJets()) {
        for (let index = 1; index < jet.trail.length; index += 1) {
          const alpha = index / jet.trail.length;
          E.line(c, jet.trail[index - 1][0], jet.trail[index - 1][1], jet.trail[index][0], jet.trail[index][1], E.rgb, 0.012 + alpha * 0.055, 0.45 + alpha * 0.62);
        }
      }
      c.restore();
    }

    function drawLinkedField(E, c) {
      if (!encounter || anchors.length < 2) return;
      const first = anchors[0];
      const second = anchors[1];
      const compression = encounter.compression;
      const strength = relationCueStrength();
      const midpointX = (first.sx + second.sx) * 0.5;
      const midpointY = (first.sy + second.sy) * 0.5;

      c.save();
      c.globalCompositeOperation = 'lighter';
      E.line(c, first.sx, first.sy, second.sx, second.sy, E.rgb, 0.18 + strength * 0.15, 1 + strength * 0.5);

      const particleCount = coarse || lowCPU ? 4 : 7;
      for (let index = 0; index < particleCount; index += 1) {
        const phase = (encounter.linkFlow + index / particleCount) % 1;
        const inward = compression >= 0;
        const side = index % 2 === 0 ? 0 : 1;
        let t;
        if (inward) t = side === 0 ? phase * 0.5 : 1 - phase * 0.5;
        else t = side === 0 ? 0.5 - phase * 0.5 : 0.5 + phase * 0.5;
        const x = lerp(first.sx, second.sx, t);
        const y = lerp(first.sy, second.sy, t);
        E.glow(c, x, y, 2.2 + strength * 1.6, inward ? E.secondary : E.rgb, 0.16 + strength * 0.24);
      }

      if (strength > 0.12) {
        const axisAngle = Math.atan2(second.sy - first.sy, second.sx - first.sx);
        const arcRadius = dist(first.sx, first.sy, second.sx, second.sy) * 0.22;
        c.strokeStyle = rgba(E.rgb, 0.05 + strength * 0.16);
        c.lineWidth = 1;
        c.beginPath();
        c.arc(midpointX, midpointY, arcRadius, axisAngle - 0.48, axisAngle + 0.48);
        c.stroke();
      }
      c.restore();
    }

    function drawShieldState(E, c, jet) {
      const cue = relationCueStrength();
      const isTarget = jet === encounter.target && encounter.vulnerability > 0.08;
      const open = isTarget ? encounter.vulnerability : 0;
      const baseAlpha = 0.055 + cue * 0.17 + jet.shieldFlash * 0.25;
      const radius = jet.size * (1.42 + open * 0.15);
      c.save();
      c.globalCompositeOperation = 'lighter';
      c.strokeStyle = rgba(isTarget ? E.secondary : E.rgb, baseAlpha + open * cue * 0.34);
      c.lineWidth = 1.1 + open * 0.7;
      const gap = 0.18 + open * 1.05;
      c.beginPath();
      c.arc(jet.sx, jet.sy, radius, gap, Math.PI - gap * 0.18);
      c.arc(jet.sx, jet.sy, radius, Math.PI + gap * 0.18, TAU - gap);
      c.stroke();

      if (isTarget && cue > 0.2) {
        const midpointX = (anchors[0].sx + anchors[1].sx) * 0.5;
        const midpointY = (anchors[0].sy + anchors[1].sy) * 0.5;
        E.line(c, midpointX, midpointY, jet.sx, jet.sy, E.secondary, cue * open * 0.16, 0.8);
        E.glow(c, jet.sx, jet.sy, jet.size * 1.65, E.secondary, cue * open * 0.14);
      }
      c.restore();
    }

    function drawDemoTracers(E, c) {
      for (const tracer of demoTracers) {
        if (!tracer.target?.alive) continue;
        const progress = 1 - clamp(tracer.life / tracer.max, 0, 1);
        const eased = clamp(progress / 0.74, 0, 1);
        const x = lerp(tracer.x1, tracer.target.sx, eased);
        const y = lerp(tracer.y1, tracer.target.sy, eased);
        E.line(c, tracer.x1, tracer.y1, x, y, E.secondary, 0.14 + (1 - progress) * 0.38, 1.2);
        E.glow(c, x, y, 3.5, E.secondary, 0.42);
      }
    }
