    function drawFlightField(E, c) {
      const horizon = E.top + E.h * 0.2;
      c.save();
      c.globalCompositeOperation = 'lighter';
      const flow = (E.t * 0.38) % 1;
      for (let i = 0; i < (coarse || lowCPU ? 10 : 16); i++) {
        const p = (i / 16 + flow) % 1;
        const angle = i * 2.39;
        const radius = E.u * (0.08 + (i % 6) * 0.055);
        const x = E.w * 0.5 + Math.cos(angle) * radius * p;
        const y = horizon + p * p * (E.bottom - horizon);
        const length = 3 + p * 15;
        E.line(c, x, y, x + Math.cos(angle) * length, y + Math.sin(angle) * length, E.secondary, 0.04 + p * 0.12, 0.65 + p);
      }
      c.restore();
    }

    function drawLeaderTrail(E, c) {
      if (!leader?.trail?.length) return;
      for (let i = 1; i < leader.trail.length; i++) {
        const a = i / leader.trail.length;
        E.line(c, leader.trail[i - 1][0], leader.trail[i - 1][1], leader.trail[i][0], leader.trail[i][1], [255, 211, 116], 0.025 + a * 0.2, 0.7 + a * 1.3);
      }
    }

    function drawOutsideRelation(E, c) {
      if (!encounter || !leader) return;
      if (encounter.phase !== 'examples') return;
      const assist = encounter.config.assist;
      const outside = encounter.outsideVisual;
      const inside = encounter.insideVisual;
      const scale = E.u * (0.34 + encounter.depth * 0.42);
      const arcRadius = scale * 0.29;
      const outsideAngle = Math.atan2(outside[1], outside[0]);
      const insidePoint = [leader.sx + inside[0] * arcRadius * 0.7, leader.sy + inside[1] * arcRadius * 0.7];

      c.save();
      c.globalCompositeOperation = 'lighter';
      E.glow(c, insidePoint[0], insidePoint[1], 3.5 + encounter.depth * 3, [255, 145, 95], 0.12 + assist * 0.12);
      c.strokeStyle = rgba(E.rgb, 0.09 + assist * 0.24);
      c.lineWidth = 1 + assist * 0.6;
      c.beginPath();
      c.arc(leader.sx, leader.sy, arcRadius, outsideAngle - 0.62, outsideAngle + 0.62);
      c.stroke();
      const tip = [leader.sx + outside[0] * arcRadius, leader.sy + outside[1] * arcRadius];
      E.line(c, leader.sx, leader.sy, tip[0], tip[1], E.rgb, 0.045 + assist * 0.12, 0.8);
      if (assist > 0.65) {
        c.fillStyle = rgba(E.rgb, 0.55);
        c.font = '800 9px Inter,system-ui,sans-serif';
        c.textAlign = 'center';
        c.fillText('OUTSIDE', tip[0] + outside[0] * 15, tip[1] + outside[1] * 15);
      }
      c.restore();
    }

    function drawFormationRelations(E, c) {
      if (!encounter || !leader) return;
      c.save();
      c.globalCompositeOperation = 'lighter';
      const network = activeEscorts();
      const networkById = new Map(network.map(escort => [escort.id, escort]));
      const facts = [];
      for (const escort of network) for (const id of escort.links || []) {
        const peer = networkById.get(id);
        if (peer && String(escort.id) < String(peer.id)) facts.push(`${escort.code} EXCHANGES WITH ${peer.code}`);
      }
      const ledgerWidth = Math.min(E.w - 24, 900);
      const columns = Math.min(3, Math.max(1, Math.floor(ledgerWidth / 250)));
      const columnWidth = (ledgerWidth - 24) / columns;
      const ledgerHeight = 25 + Math.ceil(facts.length / columns) * 18;
      c.fillStyle = 'rgba(6,18,38,.9)';
      c.fillRect(12, E.top + 8, ledgerWidth, ledgerHeight);
      c.strokeStyle = 'rgba(80,190,255,.28)';
      c.strokeRect(12.5, E.top + 8.5, ledgerWidth - 1, ledgerHeight - 1);
      c.fillStyle = 'rgba(225,245,255,.96)';
      c.font = '800 11px Inter,system-ui,sans-serif';
      c.textAlign = 'left';
      c.fillText('RELATION LEDGER · POSITION AND MOTION ARE IRRELEVANT', 22, E.top + 24);
      c.font = '700 10px Inter,system-ui,sans-serif';
      facts.forEach((fact, index) => c.fillText(fact, 22 + (index % columns) * columnWidth, E.top + 43 + Math.floor(index / columns) * 18));
      if (encounter.phase === 'examples' && encounter.target?.alive) {
        const target = encounter.target;
        c.strokeStyle = 'rgba(255,224,125,.96)';
        c.lineWidth = 2.4;
        c.beginPath();
        c.arc(target.sx, target.sy, target.size * 1.85, 0, TAU);
        c.stroke();
        c.fillStyle = 'rgba(255,224,125,.92)';
        c.font = '800 9px Inter,system-ui,sans-serif';
        c.textAlign = 'center';
        c.fillText(`ANSWER ${target.code} · EXAMPLE ${encounter.exampleIndex + 1}/${encounter.config.exampleCount}`, target.sx, target.sy - target.size * 2.25);
      }

      const segments = encounter.config.hitsRequired;
      const shieldRadius = leader.size * 1.86;
      for (let i = 0; i < segments; i++) {
        const active = i >= encounter.hits;
        c.strokeStyle = active ? 'rgba(255,215,126,.48)' : 'rgba(100,112,130,.12)';
        c.lineWidth = 1.4;
        c.beginPath();
        c.arc(leader.sx, leader.sy, shieldRadius, -Math.PI / 2 + i / segments * TAU + 0.08, -Math.PI / 2 + (i + 1) / segments * TAU - 0.08);
        c.stroke();
      }
      c.restore();
    }
