      streak = 0;
        E.setMetric('LIVES', lives);
        E.shake(9);
        E.flash([255, 70, 110], .23);
        E.burst(gateX, gateYs[correct], [255, 70, 110], 14, 170, 2.6);
        E.sfx('bad');
        if (lives <= 0) { E.end('FUTURE MISREAD'); return; }
      }
      phase = 'result';
      timer = .8;
    }

    function chooseGate(E, index) {
      if (phase === 'result' || selected >= 0) return;
      selected = clamp(index, 0, 2);
      E.sfx('tap', .9 + selected * .12);
      E.burst(gateX, gateYs[selected], E.rgb, 7, 80, 2);
    }

    return {
      reset(E) { level = 1; lives = 3; streak = 0; rounds = 0; setup(E); },
      resize(E) { setup(E); },
      update(E, dt) {
        if (phase === 'result') {
          timer -= dt;
          if (timer <= 0 && lives > 0) setup(E);
          return;
        }
        const minY = E.top + 55;
        const maxY = E.bottom - 45;
        ball.x += ball.vx * dt;
        ball.y += ball.vy * dt;
        if (ball.y < minY) { ball.y = minY + (minY - ball.y); ball.vy = Math.abs(ball.vy); E.sfx('tick', .75); }
        if (ball.y > maxY) { ball.y = maxY - (ball.y - maxY); ball.vy = -Math.abs(ball.vy); E.sfx('tick', .88); }
        trail.push({ x: ball.x, y: ball.y, life: .65 });
        if (trail.length > 36) trail.shift();
        trail.forEach((p) => { p.life -= dt; });
        if (phase === 'observe' && ball.x >= veilA) { phase = 'choose'; E.toast('CHOOSE EXIT', .65); }
        if (ball.x >= gateX) resolve(E);
      },
      pointerDown(E, x, y) {
        let index = 0, bestD = Infinity;
        gateYs.forEach((gy, i) => {
          const d = dist(x, y, gateX, gy);
          if (d < bestD) { bestD = d; index = i; }
        });
        if (x > E.w * .66 || phase === 'choose') chooseGate(E, index);
      },
      keyDown(E, key) {
        if (key === '1' || key === 'arrowup') chooseGate(E, 0);
        else if (key === '2' || key === 'arrowright') chooseGate(E, 1);
        else if (key === '3' || key === 'arrowdown') chooseGate(E, 2);
      },
      draw(E, c) {
        c.save();
        c.globalCompositeOperation = 'lighter';
        const minY = E.top + 55;
        const maxY = E.bottom - 45;

        for (let i = 0; i < trail.length; i++) {
          const p = trail[i];
          if (p.x > veilA && p.x < veilB) continue;
          const a = clamp(p.life / .65, 0, 1);
          E.glow(c, p.x, p.y, ball.r * (.35 + a * .5), E.rgb, a * .2);
        }

        const veilGradient = c.createLinearGradient(veilA, 0, veilB, 0);
        veilGradient.addColorStop(0, rgba(E.rgb, .025));
        veilGradient.addColorStop(.5, 'rgba(0,0,0,.72)');
        veilGradient.addColorStop(1, rgba(E.rgb, .025));
        c.fillStyle = veilGradient;
        c.fillRect(veilA, minY - 30, veilB - veilA, maxY - minY + 60);
        c.strokeStyle = rgba(E.rgb, .18);
        c.lineWidth = 1;
        for (let x = veilA; x <= veilB; x += Math.max(10, E.u * .018)) {
          c.beginPath(); c.moveTo(x, minY - 25); c.lineTo(x + Math.sin(E.t * 2 + x * .03) * 5, maxY + 25); c.stroke();
        }

        gateYs.forEach((gy, i) => {
          const picked = selected === i;
          const actual = phase === 'result' && correct === i;
          const rgb = actual && selected !== correct ? [255, 82, 120] : E.rgb;
          const r = Math.max(23, E.u * .047);
          E.glow(c, gateX, gy, r * (picked ? 1.25 : .85), rgb, picked || actual ? .8 : .3);
          c.strokeStyle = rgba(rgb, picked || actual ? .95 : .42);
          c.lineWidth = picked ? 3 : 1.2;
          c.beginPath(); c.arc(gateX, gy, r, 0, TAU); c.stroke();
          c.fillStyle = rgba(rgb, .12);
          c.beginPath(); c.arc(gateX, gy, r * .58, 0, TAU); c.fill();
          c.fillStyle = 'rgba(255,255,255,.58)';
          c.font = `800 ${Math.max(9, E.u * .015)}px Inter,system-ui,sans-serif`;
          c.textAlign = 'center'; c.textBaseline = 'middle'; c.fillText(String(i + 1), gateX, gy);
        });

        if (!(ball.x > veilA && ball.x < veilB)) {
          E.glow(c, ball.x, ball.y, ball.r * 2.3, E.rgb, .9);
          c.fillStyle = 'rgba(255,255,255,.94)';
          c.beginPath(); c.arc(ball.x, ball.y, ball.r * .55, 0, TAU); c.fill();
        }

        if (ball.x < veilA) {
          const previewT = Math.min((veilA - ball.x) / ball.vx, .75);
          const endX = ball.x + ball.vx * previewT;
          const endY = reflectY(ball.y + ball.vy * previewT, minY, maxY);
          c.setLineDash([5, 7]);
          E.line(c, ball.x, ball.y, endX, endY, E.rgb, .42, 1);
          c.setLineDash([]);
        }
        c.restore();
      }
    };
  }

  function createVectorVow() {
    let origin = { x: 0, y: 0 };
    let target = { x: 0, y: 0, r: 30, phase: 0, speed: .7 };
    let projectile = null;
    let aiming = false;
    let aimX = 0, aimY = 0;
    let charge = 0;
    let level = 1;
    let vows = 4;
    let hits = 0;
    let combo = 0;
    let transition = 0;

    function place(E) {
      origin.x = E.w * .5;
      origin.y = E.h * .79;
      target.phase = Math.random() * TAU;
      target.r = Math.max(22, E.u * (.042 - Math.min(level, 12) * .0012));
      target.speed = .52 + level * .045;
      target.x = E.w * .5;
      target.y = E.top + (E.bottom - E.top) * .23;
      aimX = E.w * .5;
      aimY = E.h * .25;
    }

    function updateTarget(E) {
      const rx = Math.min(E.w * .31, E.u * .36);
      const ry = Math.min((E.bottom - E.top) * .14, E.u * .13);
      const a = E.t * target.speed + target.phase;
      target.x = E.w * .5 + Math.cos(a) * rx;
      target.y = E.top + (E.bottom - E.top) * .24 + Math.sin(a * 1.7) * ry;
    }

    function launch(E) {
      if (!aiming || projectile) return;
      aiming = false;
      const dx = aimX - origin.x;
      const dy = aimY - origin.y;
      const length = Math.max(1, Math.hypot(dx, dy));
      const power = 360 + charge * 310;
      projectile = {
        x: origin.x, y: origin.y,
        vx: dx / length * power,
        vy: dy / length * power,
        r: Math.max(6, E.u * .011),
        life: 4.2,
        bounces: 0,
        trail: []
      };
      charge = 0;
      E.sfx('launch', .82 + power / 1000);
      E.burst(origin.x, origin.y, E.rgb, 11, 130, 2.4);
    }

    function miss(E) {
      if (!projectile) return;
      E.burst(projectile.x, projectile.y, [255, 72, 110], 13, 150, 2.5);
      projectile = null;
      vows--;
      combo = 0;
      E.setMetric('VOWS', vows);
      E.shake(8);
      E.flash([255, 72, 110], .19);
      E.sfx('bad');
      if (vows <= 0) E.end('INTENTION BROKE');
    }

    function hit(E) {
      combo++;
      hits++;
      const reward = 180 + combo * 45 + level * 25 + Math.floor(charge * 40);
      E.addScore(reward, target.x, target.y, combo > 1 ? `VOW ×${combo}` : `+${reward}`);
      E.burst(target.x, target.y, E.rgb, 26, 240, 3.2);
      E.flash(E.rgb, .17);
      E.sfx('good', 1 + combo * .04);
      projectile = null;
      target.phase += rand(2.4, .8);
      transition = .42;
      if (hits >= 3) {
        hits = 0;
        level++;
        vows = Math.min(5, vows + 1);
        E.setMetric('VOWS', vows);
        E.level(`VECTOR ${level}`);
      }
    }

    return {
      reset(E) {
        level = 1; vows = 4; hits = 0; combo = 0; projectile = null; aiming = false; charge = 0; transition = 0;
        place(E);
        E.setMetric('VOWS', vows);
      },
      resize(E) { place(E); projectile = null; aiming = false; },
      update(E, dt) {
        updateTarget(E);
        if (transition > 0) transition -= dt;
        if (aiming && !projectile) charge = clamp(charge + dt * .85, 0, 1);
        if (!projectile) return;
        const p = projectile;
        p.life -= dt;
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.trail.push({ x: p.x, y: p.y, life: .5 });
        if (p.trail.length > 28) p.trail.shift();
        p.trail.forEach((q) => { q.life -= dt; });

        const left = 15, right = E.w - 15, top = E.top + 15;
        let bounced = false;
        if (p.x < left) { p.x = left + (left - p.x); p.vx = Math.abs(p.vx); bounced = true; }
        if (p.x > right) { p.x = right - (p.x - right); p.vx = -Math.abs(p.vx); bounced = true; }
        if (p.y < top) { p.y = top + (top - p.y); p.vy = Math.abs(p.vy); bounced = true; }
        if (bounced) {
          p.bounces++;
          E.sfx('tick', 1 + p.bounces * .15);
          E.burst(p.x, p.y, E.rgb, 5, 70, 1.8);
        }
        if (dist(p.x, p.y, target.x, target.y) < p.r + target.r) { hit(E); return; }
        if (p.y > E.bottom + 40 || p.life <= 0 || p.bounces > 1) miss(E);
      },
      pointerDown(E, x, y) {
        if (projectile || transition > 0) return;
        aiming = true;
        aimX = x; aimY = Math.min(y, origin.y - 20);
        charge = .05;
        E.sfx('tap', .75);
      },
      pointerMove(E, x, y, down) {
        if (!aiming || !down) return;
        aimX = x;
        aimY = Math.min(y, origin.y - 20);
      },
      pointerUp(E, x, y) {
        if (!aiming) return;
        aimX = x; aimY = Math.min(y, origin.y - 20);
        launch(E);
      },
      keyDown(E, key) {
        if (key === ' ' && !projectile && !aiming) {
          aiming = true;
          aimX = target.x; aimY = target.y; charge = .25;
        }
      },
      keyUp(E, key) { if (key === ' ' && aiming) launch(E); },
      draw(E, c) {
        updateTarget(E);
        c.save();
        c.globalCompositeOperation = 'lighter';

        const orbitRx = Math.min(E.w * .31, E.u * .36);
        const orbitRy = Math.min((E.bottom - E.top) * .14, E.u * .13);
        c.strokeStyle = rgba(E.rgb, .12);
        c.lineWidth = 1;
        c.beginPath(); c.ellipse(E.w * .5, E.top + (E.bottom - E.top) * .24, orbitRx, orbitRy, 0, 0, TAU); c.stroke();

        const targetPulse = .5 + .5 * Math.sin(E.t * 3.2) ** 2;
        E.glow(c, target.x, target.y, target.r * (1.5 + targetPulse * .35), E.rgb, .72 + targetPulse * .2);
        c.strokeStyle = 'rgba(255,255,255,.78)';
        c.lineWidth = 1.4;
        c.beginPath(); c.arc(target.x, target.y, target.r, E.t * .8, E.t * .8 + Math.PI * 1.55); c.stroke();
        c.beginPath(); c.arc(target.x, target.y, target.r * .58, -E.t * 1.1, -E.t * 1.1 + Math.PI * 1.3); c.stroke();

        E.glow(c, origin.x, origin.y, Math.max(22, E.u * .035), E.rgb, .62);
        c.fillStyle = 'rgba(255,255,255,.88)';
        c.beginPath(); c.arc(origin.x, origin.y, Math.max(4, E.u * .007), 0, TAU); c.fill();
        c.strokeStyle = rgba(E.rgb, .4);
        c.lineWidth = 1;
        c.beginPath(); c.arc(origin.x, origin.y, Math.max(22, E.u * .035), 0, TAU); c.stroke();

        if (aiming && !projectile) {
          const dx = aimX - origin.x, dy = aimY - origin.y;
          const len = Math.max(1, Math.hypot(dx, dy));
          const nx = dx / len, ny = dy / len;
          const reach = Math.min(E.u * .46, 120 + charge * E.u * .35);
          c.setLineDash([7, 8]);
          E.line(c, origin.x, origin.y, origin.x + nx * reach, origin.y + ny * reach, E.rgb, .7, 1.4);
          c.setLineDash([]);
          for (let i = 1; i <= 8; i++) {
            const u = i / 8;
            E.glow(c, origin.x + nx * reach * u, origin.y + ny * reach * u, 3 + 3 * u, E.rgb, .28 + charge * .25);
          }
          c.strokeStyle = rgba(E.rgb, .36);
          c.lineWidth = 5;
          c.beginPath(); c.arc(origin.x, origin.y, Math.max(28, E.u * .045), -Math.PI / 2, -Math.PI / 2 + TAU * charge); c.stroke();
        }

        if (projectile) {
          for (const p of projectile.trail) {
            const a = clamp(p.life / .5, 0, 1);
            E.glow(c, p.x, p.y, projectile.r * (.4 + a), E.rgb, a * .34);
          }
          E.glow(c, projectile.x, projectile.y, projectile.r * 2.4, E.rgb, .9);
          c.fillStyle = 'rgba(255,255,255,.95)';
          c.beginPath(); c.arc(projectile.x, projectile.y, projectile.r * .58, 0, TAU); c.fill();
        }
        c.restore();
      }
    };
  }

  function createImpulseRun() {
    let player = { x: 0, y: 0, tx: 0, r: 13 };
    let entities = [];
    let spawnTimer = 0;
    let level = 1;
    let lives = 3;
    let invulnerable = 0;
    let distancePoints = 0;
    let bonusPoints = 0;
    let combo = 0;
    let speed = 250;

    function resetPosition(E) {
      player.x = player.tx = E.w * .5;
      player.y = E.h * .82;
      player.r = Math.max(10, E.u * .018);
    }

    function spawn(E) {
      const margin = Math.max(28, E.w * .08);
      const difficulty = clamp(level / 12, 0, 1);
      const roll = Math.random();
      let type = roll < .23 ? 'orb' : roll < .34 + difficulty * .08 ? 'wall' : 'hazard';
      const r = type === 'orb' ? Math.max(8, E.u * .013) : Math.max(14, E.u * (.022 + Math.random() * .009));
      const x = rand(E.w - margin, margin);
      entities.push({
        type, x, y: E.top - r - 12, r,
        vy: speed * rand(1.18, .88),
        spin: rand(TAU),
        rot: rand(2.2, -.8),
        width: type === 'wall' ? rand(E.w * .22, E.w * .1) : r * 2,
        phase: rand(TAU)
      });
      spawnTimer = Math.max(.2, .78 - level * .035) * rand(1.18, .78);
    }

    function collide(E, entity) {
      if (entity.type === 'orb') {
        combo++;
        const reward = 70 + combo * 18;
        bonusPoints += reward;
        E.addScore(reward, entity.x, entity.y, combo > 2 ? `FLOW ×${combo}` : `+${reward}`);
        E.burst(entity.x, entity.y, E.rgb, 16, 190, 2.8);
        E.sfx('good', .9 + combo * .03);
        return true;
      }
      if (invulnerable > 0) return false;
      lives--;
      combo = 0;
      invulnerable = 1.35;
      E.setMetric('LIVES', lives);
      E.shake(12);
      E.flash([255, 70, 110], .25);
      E.burst(player.x, player.y, [255, 70, 110], 22, 240, 3);
      E.sfx('bad');
      if (lives <= 0) E.end('MOTION ENDED');
      return true;
    }

    return {
      reset(E) {
        level = 1; lives = 3; invulnerable = 0; distancePoints = 0; bonusPoints = 0; combo = 0; speed = 250; entities = []; spawnTimer = .35;
        resetPosition(E);
        E.setMetric('LIVES', lives);
      },
      resize(E) { resetPosition(E); entities = []; },
      update(E, dt) {
        level = 1 + Math.floor(E.t / 13);
        speed = Math.min(620, 250 + level * 26);
        if (invulnerable > 0) invulnerable -= dt;
        const keyDirection = (E.keys.has('arrowleft') || E.keys.has('a') ? -1 : 0) + (E.keys.has('arrowright') || E.keys.has('d') ? 1 : 0);
        if (keyDirection) player.tx += keyDirection * E.w * .8 * dt;
        player.tx = clamp(player.tx, player.r + 12, E.w - player.r - 12);
        player.x = lerp(player.x, player.tx, 1 - Math.exp(-dt * 13));

        spawnTimer -= dt;
        if (spawnTimer <= 0) spawn(E);
        for (let i = entities.length - 1; i >= 0; i--) {
          const entity = entities[i];
          entity.y += entity.vy * dt;
          entity.spin += entity.rot * dt;
          const dx = Math.abs(player.x - entity.x);
          const dy = Math.abs(player.y - entity.y);
          let hit = false;
          if (entity.type === 'wall') hit = dy < player.r + entity.r && dx < entity.width * .5 + player.r;
          else hit = Math.hypot(player.x - entity.x, player.y - entity.y) < player.r + entity.r;
          if (hit && collide(E, entity)) entities.splice(i, 1);
          else if (entity.y > E.h + entity.r + 30) entities.splice(i, 1);
        }
        distancePoints += dt * (18 + level * 2.5);
        E.setScore(distancePoints + bonusPoints);
      },
      pointerDown(E, x) { player.tx = x; },
      pointerMove(E, x, y, down) { if (down) player.tx = x; },
      keyDown(E, key) { if (key === 'arrowleft' || key === 'a') player.tx -= E.w * .08; if (key === 'arrowright' || key === 'd') player.tx += E.w * .08; },
      draw(E, c) {
        c.save();
        c.globalCompositeOperation = 'lighter';
        const vanishX = E.w * .5;
        const vanishY = E.top + 35;
        const bottomY = E.bottom + 20;
        for (let i = -5; i <= 5; i++) {
          const bx = E.w * .5 + i * E.w * .12;
          c.strokeStyle = rgba(E.rgb, .06 + (i === 0 ? .07 : 0));
          c.lineWidth = i === 0 ? 1.4 : .8;
          c.beginPath(); c.moveTo(vanishX, vanishY); c.lineTo(bx, bottomY); c.stroke();
        }
        const stripeOffset = (E.t * speed * .8) % 90;
        for (let y = vanishY + stripeOffset; y < bottomY; y += 90) {
          const p = clamp((y - vanishY) / (bottomY - vanishY), 0, 1);
          const half = E.w * .47 * p;
          c.strokeStyle = rgba(E.rgb, .06 + p * .12);
          c.beginPath(); c.moveTo(vanishX - half, y); c.lineTo(vanishX + half, y); c.stroke();
        }

        const sorted = [...entities].sort((a, b) => a.y - b.y);
        for (const entity of sorted) {
          const depth = clamp((entity.y - E.top) / Math.max(1, E.bottom - E.top), 0, 1);
          const scale = .58 + depth * .65;
          if (entity.type === 'orb') {
            E.glow(c, entity.x, entity.y, entity.r * 2.1 * scale, E.rgb, .76);
            c.fillStyle = 'rgba(255,255,255,.9)';
            c.beginPath(); c.arc(entity.x, entity.y, entity.r * .4 * scale, 0, TAU); c.fill();
            c.strokeStyle = rgba(E.rgb, .65); c.lineWidth = 1;
            c.beginPath(); c.arc(entity.x, entity.y, entity.r * 1.25 * scale, entity.spin, entity.spin + Math.PI * 1.5); c.stroke();
          } else if (entity.type === 'wall') {
