        escort.size = Math.max(9, E.u * (0.012 + depth * 0.014));
        escort.heading = leader.heading + Math.sin(encounter.age * 0.68 + escort.weavePhase) * 0.15;
        escort.bank = encounter.turnValue * 0.38 + Math.cos(orbit) * 0.32;
        escort.pulse = 0.42 + 0.58 * Math.sin(encounter.age * 2.1 + escort.weavePhase) ** 2;
        escort.hitFlash = Math.max(0, escort.hitFlash - dt * 4.4);
        escort.trail.push([escort.sx, escort.sy]);
        if (escort.trail.length > 10) escort.trail.shift();
      }

      updateRelationalRole(dt, false);
      updateAcquisition(dt);
      if (encounter.phase === 'examples' && encounter.age >= encounter.nextExampleAt) {
        encounter.exampleIndex++;
        if (encounter.exampleIndex >= encounter.config.exampleCount) {
          encounter.phase = 'transfer';
          newRelationalFormation();
          E.toast('TRANSFER · APPLY THE SAME RELATION', 1.1);
        } else {
          encounter.turnTarget *= -1;
          encounter.turnValue = encounter.turnTarget;
          encounter.committedTurnSign = encounter.turnTarget;
          encounter.nextTurnAt = encounter.age + encounter.config.turnHoldMax;
          newRelationalFormation();
          encounter.nextExampleAt = encounter.age + encounter.config.exampleDuration;
          E.toast(`EXAMPLE ${encounter.exampleIndex + 1} OF ${encounter.config.exampleCount}`, 0.75);
        }
        updateRelationalRole(0, true);
      }
    }

    function rankedOutsideEscorts() {
      if (!encounter.geometry) {
        encounter.geometry = roleDriftGeometry(activeEscorts(), encounter.outsideAxis, {
          threshold: encounter.config.outsideThreshold,
          yScale: 0.84,
        });
      }
      return encounter.geometry.outside.map((metric) => metric.escort);
    }

    function switchRole(next, forced = false) {
      if (!next || !next.alive) return;
      if (encounter.target === next) return;
      const oldTarget = encounter.target;
      encounter.previousTarget = oldTarget && oldTarget.alive ? oldTarget : encounter.previousTarget;
      encounter.target = next;
      encounter.candidate = null;
      encounter.candidateTimer = 0;
      encounter.targetStableFor = 0;
      encounter.roleSwitchAt = encounter.age;
      encounter.reticleHold = 0;
      encounter.reacquired = false;
      if (oldTarget && oldTarget !== next) stats.roleSwitches++;
    }

    function updateRelationalRole(dt, force = false) {
      const next = roleGraphSelect(activeEscorts(), encounter.rule);
      if (!next) {
        encounter.target = null;
        return;
      }
      if (!encounter.target || !encounter.target.alive || force) {
        switchRole(next, true);
        return;
      }
      if (next !== encounter.target) switchRole(next);
      else encounter.targetStableFor += dt;
    }

    function updateAcquisition(dt) {
      const target = encounter.target;
      if (!target || !target.alive || encounter.reacquired) return;
      const radius = Math.max(18, target.size * 1.55);
      if (dist(reticle.x, reticle.y, target.sx, target.sy) <= radius) encounter.reticleHold += dt;
      else encounter.reticleHold = Math.max(0, encounter.reticleHold - dt * 1.8);
      if (encounter.reticleHold >= 0.17) {
        encounter.reacquired = true;
        const latency = Math.max(0, encounter.age - encounter.roleSwitchAt);
        stats.reacquisitionLatencies.push(latency);
        stats.reacquisitions++;
      }
    }

    function drawFighter(c, x, y, size, heading, bank, rgb, kind = 'escort', alpha = 1, pulse = 0.5) {
      const enemy = kind !== 'player';
      c.save();
      c.translate(x, y);
      c.rotate((enemy ? Math.PI : 0) + heading + bank * 0.16);
      c.globalCompositeOperation = 'lighter';
      c.globalAlpha = alpha;

      const glow = c.createRadialGradient(0, 0, 0, 0, 0, size * 2.45);
      glow.addColorStop(0, rgba(rgb, kind === 'leader' ? 0.2 : 0.12 + pulse * 0.055));
      glow.addColorStop(1, 'rgba(0,0,0,0)');
      c.fillStyle = glow;
      c.fillRect(-size * 2.5, -size * 2.5, size * 5, size * 5);

      c.fillStyle = rgba(rgb, kind === 'leader' ? 0.24 : 0.14 + pulse * 0.055);
      c.strokeStyle = rgba(rgb, kind === 'leader' ? 0.96 : 0.76);
      c.lineWidth = Math.max(1, size * 0.07);
      c.beginPath();
      c.moveTo(0, -size * 1.42);
      c.lineTo(size * 0.22, -size * 0.5);
