
    function candidateById(id) {
      return candidates.find((jet) => jet.id === id && jet.alive) || null;
    }

    function migrateTarget(nextTarget) {
      if (encounter.target === nextTarget) return;
      if (encounter.target?.alive) encounter.previousTarget = encounter.target;
      const hadTarget = Boolean(encounter.target);
      encounter.target = nextTarget;
      encounter.targetStableFor = 0;
      encounter.relationOnsetAt = encounter.age;
      encounter.vulnerability = 0;
      encounter.demoLockout = Math.max(encounter.demoLockout, 0.06);
      if (hadTarget && nextTarget) {
        encounter.roleMigrations += 1;
        stats.roleMigrations += 1;
      }
    }

    function updateRelationState(E, dt) {
      const config = encounter.config;
      const state = causalInterpositionState(
        { id: anchors[0].id, x: anchors[0].localX, y: anchors[0].localY },
        { id: anchors[1].id, x: anchors[1].localX, y: anchors[1].localY },
        activeCandidates().map((jet) => ({ id: jet.id, x: jet.localX, y: jet.localY, alive: jet.alive })),
        {
          lineThreshold: config.lineThreshold,
          edgeMargin: 0.14,
          compression: encounter.compression,
          requireCompression: config.requireCompression,
          compressionThreshold: config.compressionThreshold,
        },
      );
      encounter.relationState = state;
      const rawCandidateId = state.activeTargetId;

      if (!rawCandidateId) {
        encounter.relationCandidateId = null;
        encounter.relationCandidateTimer = 0;
        migrateTarget(null);
      } else if (encounter.relationCandidateId === rawCandidateId) {
        encounter.relationCandidateTimer += dt;
      } else {
        encounter.relationCandidateId = rawCandidateId;
        encounter.relationCandidateTimer = 0;
      }

      if (rawCandidateId && encounter.relationCandidateTimer >= config.relationDwell) {
        migrateTarget(candidateById(rawCandidateId));
      }

      if (encounter.target?.alive && encounter.target.id === rawCandidateId) {
        encounter.targetStableFor += dt;
        encounter.vulnerability = lerp(encounter.vulnerability, 1, 1 - Math.exp(-dt * 12));
      } else {
        encounter.vulnerability = lerp(encounter.vulnerability, 0, 1 - Math.exp(-dt * 15));
      }

      encounter.demoLockout = Math.max(0, encounter.demoLockout - dt);
      if (encounter.demoPending && !encounter.demoPlayed && encounter.target && encounter.vulnerability > 0.82) {
        encounter.demoPlayed = true;
        encounter.demoPending = false;
        encounter.demoLockout = 0.55;
        encounter.target.shieldFlash = 1;
        demoTracers.push({
          x1: E.w * 0.16,
          y1: E.bottom - E.u * 0.08,
          target: encounter.target,
          life: 0.7,
          max: 0.7,
          struck: false,
        });
        E.sfx('good', 0.82);
      }
    }

    function updateDemo(E, dt) {
      for (let index = demoTracers.length - 1; index >= 0; index -= 1) {
        const tracer = demoTracers[index];
        tracer.life -= dt;
        const progress = 1 - clamp(tracer.life / tracer.max, 0, 1);
        if (!tracer.struck && progress > 0.72 && tracer.target?.alive) {
          tracer.struck = true;
          tracer.target.shieldFlash = 1;
          E.burst(tracer.target.sx, tracer.target.sy, E.secondary, 8, 90, 1.8);
          E.shake(3);
        }
        if (tracer.life <= 0) demoTracers.splice(index, 1);
      }
    }

    function updateFormation(E, dt) {
      if (!encounter) return;
      encounter.age += dt;
      if (!encounter.solved && !encounter.failed) {
        encounter.depth = Math.min(1.08, encounter.depth + dt * encounter.config.approachSpeed);
      }
      encounter.linkFlow = (encounter.linkFlow + dt * (0.42 + level * 0.028)) % 1;
      const frame = formationFrame();
      updateAnchorGeometry(dt, frame);
      updateCandidateGeometry(dt, frame);
      projectFormation(E);
      updateRelationState(E, dt);
      updateDemo(E, dt);
    }

    function relationCueStrength() {
      const base = encounter?.config?.cueAlpha || 0;
      const learnedFade = encounter?.hits ? Math.max(0.35, 1 - encounter.hits * 0.2) : 1;
      return base * learnedFade;
    }

    function currentRelationPhase() {
      if (!encounter?.relationState) return 'none';
      if (encounter.target && encounter.vulnerability > 0.62) return 'open';
      if (encounter.relationState.geometricTargetId && !encounter.relationState.compressionGate) return 'geometry-without-compression';
      if (encounter.relationState.geometricTargetId) return 'forming';
      return 'closed';
    }
