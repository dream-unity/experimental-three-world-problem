
  // Dream Machine / PERCEIVE: PARALLAX WING — CAUSAL WEAVE.
  // No proposition is stated to the player. The world repeatedly demonstrates
  // which relational configurations open or preserve shields, then requires
  // the player to exploit the same causal structure under transformation.
  function createPerceptionAerialGame() {
    const PROFILE_KEY = 'dream-unity.perceive.parallax-wing.causal-weave.v1';
    const LAW_ORDER = ['between', 'flow', 'enclosure'];

    let player;
    let reticle;
    let enemyShots = [];
    let tracers = [];
    let scene = null;
    let wave = 1;
    let level = 1;
    let streak = 0;
    let successes = 0;
    let resolveTimer = 0;
    let wrongShotCooldown = 0;
    let pointerGesture = null;
    let stats;
    const demonstrations = new Map(LAW_ORDER.map((law) => [law, 0]));
    const episodeHistory = [];

    const rotate2 = (x, y, angle) => {
      const c = Math.cos(angle);
      const s = Math.sin(angle);
      return [x * c - y * s, x * s + y * c];
    };

    const shuffled = (items) => {
      const output = [...items];
      for (let i = output.length - 1; i > 0; i--) {
        const j = (Math.random() * (i + 1)) | 0;
        [output[i], output[j]] = [output[j], output[i]];
      }
      return output;
    };

    function writeProfile(record) {
      try {
        const prior = JSON.parse(localStorage.getItem(PROFILE_KEY) || '[]');
        const rows = Array.isArray(prior) ? prior : [];
        rows.push(record);
        localStorage.setItem(PROFILE_KEY, JSON.stringify(rows.slice(-240)));
      } catch { /* local telemetry is optional */ }
    }

    function lawForLevel(value) {
      if (value <= 3) return 'between';
      if (value <= 6) return 'flow';
      if (value <= 9) return 'enclosure';
      return choose(LAW_ORDER);
    }

    function levelConfig(value, law) {
      const l = clamp(value, 1, 10);
      const cueStrength = clamp(0.92 - (l - 1) * 0.082, 0.16, 0.92);
      return {
        law,
        duration: Math.max(3.25, 4.75 - l * 0.105),
        approachSpeed: 0.027 + l * 0.00225,
        cueStrength,
        shotsRequired: l < 4 ? 2 : l < 8 ? 3 : 4,
        activeTolerance: Math.max(0.048, 0.082 - l * 0.0032),
        crossingSpeed: 0.028 + l * 0.0015,
        enemyFireInterval: Math.max(1.35, 2.75 - l * 0.11),
        projectileSpeed: 175 + l * 13,
        incomingLimit: Math.min(11, 4 + Math.floor(l / 2)),
        playerShield: 5,
        negativeRate: clamp(0.35 + l * 0.012, 0.35, 0.48),
      };
    }

    function makeJet(id) {
      return {
        id,
        role: 'unassigned',
        rx: 0,
        ry: 0,
        vx: 0,
        vy: 0,
        previousRx: 0,
        previousRy: 0,
        sx: 0,
        sy: 0,
        size: 10,
        heading: 0,
        bank: 0,
        pulse: 0,
        alive: true,
        hitFlash: 0,
        shieldFlash: 0,
        trail: [],
      };
    }

    function buildRoles(law) {
      const count = law === 'enclosure' ? 5 : 4;
      const jets = shuffled(Array.from({ length: count }, (_, index) => makeJet(`cw-${wave}-${index}-${Math.random().toString(36).slice(2, 7)}`)));
      const roles = {
        anchorAId: jets[0].id,
        anchorBId: jets[1].id,
        relayId: jets[2].id,
        decoyId: jets[3].id,
        anchorCId: law === 'enclosure' ? jets[4].id : null,
      };
      for (const jet of jets) {
        if (jet.id === roles.anchorAId) jet.role = 'anchor-a';
        else if (jet.id === roles.anchorBId) jet.role = 'anchor-b';
        else if (jet.id === roles.anchorCId) jet.role = 'anchor-c';
        else if (jet.id === roles.relayId) jet.role = 'relay';
        else jet.role = 'decoy';
      }
      return { jets, roles };
    }

    function chooseNegativeVariant(law) {
      if (law === 'enclosure') return choose(['outside-triangle', 'boundary-touch', 'open-triangle']);
      return choose(['line-extension', 'near-offset', 'parallel-near']);
    }

    function episodePolarity(config, demoKind) {
      if (demoKind === 'positive') return true;
      if (demoKind === 'negative') return false;
      const recent = episodeHistory.slice(-3);
      if (recent.length >= 2 && recent.at(-1) === recent.at(-2)) return !recent.at(-1);
      return Math.random() >= config.negativeRate;
    }

    function nextDemoKind(law) {
      const progress = demonstrations.get(law) || 0;
      if (progress === 0) return 'positive';
      if (progress === 1) return 'negative';
      return null;
    }
