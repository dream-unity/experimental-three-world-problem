
  // Dream Machine / PERCEIVE: PARALLAX WING — RELATIONAL IDENTITY DRIFT.
  // The target is never an unusual aircraft. It is the escort currently
  // occupying a live relational role: nearest to the leader on the outside
  // of the leader's turn. As the formation crosses, that role moves.
  function createPerceptionAerialGame() {
    const PROFILE_KEY = 'dream-unity.perceive.parallax-wing.role-drift.v2';

    let player;
    let reticle;
    let leader;
    let escorts = [];
    let enemyShots = [];
    let tracers = [];
    let encounter = null;
    let wave = 1;
    let level = 1;
    let streak = 0;
    let completedFormations = 0;
    let resolveTimer = 0;
    let wrongShotCooldown = 0;
    let pointerGesture = null;
    let stats;

    const rotate2 = (x, y, angle) => {
      const c = Math.cos(angle);
      const s = Math.sin(angle);
      return [x * c - y * s, x * s + y * c];
    };

    const length2 = (v) => Math.hypot(v[0], v[1]);
    const normalize2 = (v) => {
      const length = Math.max(0.000001, length2(v));
      return [v[0] / length, v[1] / length];
    };

    function writeProfile(record) {
      try {
        const existing = JSON.parse(localStorage.getItem(PROFILE_KEY) || '[]');
        const rows = Array.isArray(existing) ? existing : [];
        rows.push(record);
        localStorage.setItem(PROFILE_KEY, JSON.stringify(rows.slice(-240)));
      } catch { /* Local telemetry is optional. */ }
    }

    function levelConfig(value) {
      const l = clamp(value, 1, 10);
      const relationalTier = l <= 2 ? 1 : l <= 6 ? 2 : 3;
      return {
        escortCount: Math.min(8, 6 + Math.floor((l - 1) / 4)),
        hitsRequired: l <= 2 ? 2 : l <= 6 ? 3 : 4,
        relationalTier,
        exampleCount: l <= 2 ? 4 : 3,
        exampleDuration: l <= 2 ? 2.7 : l <= 6 ? 2.15 : 1.7,
        approachSpeed: 0.034 + l * 0.0025,
        orbitSpeed: 0.11 + l * 0.014,
        weaveSpeed: 0.72 + l * 0.075,
        weaveAmplitude: 0.42 + l * 0.025,
        radialAmplitude: 0.13 + l * 0.012,
        turnRate: 0.36 + l * 0.024,
        turnResponse: 1.65 + l * 0.09,
        turnHoldMin: Math.max(2.15, 4.7 - l * 0.21),
        turnHoldMax: Math.max(3.0, 6.2 - l * 0.22),
        turnChanges: l >= 3,
        candidateDwell: Math.max(0.11, 0.24 - l * 0.011),
        roleHold: Math.max(0.4, 0.82 - l * 0.035),
        outsideThreshold: 0.018,
        distanceAdvantage: Math.max(0.045, 0.12 - l * 0.006),
        fireInterval: Math.max(1.35, 2.8 - l * 0.105),
        projectileSpeed: 170 + l * 15,
        assist: 0,
        incomingLimit: Math.min(10, 4 + Math.floor(l / 2)),
      };
    }

    function makeEscort(index, count) {
      const spacing = TAU / count;
      return {
        id: `escort-${wave}-${index}-${Math.random().toString(36).slice(2, 7)}`,
        code: String.fromCharCode(65 + index),
        slotAngle: index * spacing + rand(spacing * 0.16, -spacing * 0.16),
        baseRadius: rand(0.34, 0.235),
        weavePhase: rand(TAU, 0),
        radialPhase: rand(TAU, 0),
        weaveRate: rand(1.18, 0.84),
        orbitDirection: choose([-1, 1]),
        rx: 0,
        ry: 0,
        sx: 0,
        sy: 0,
        size: 10,
        heading: 0,
        bank: 0,
        pulse: 0,
        alive: true,
        hitFlash: 0,
        outsideProjection: 0,
        distanceToLeader: 1,
        links: [],
        trail: [],
      };
    }

    function activeEscorts() {
      return escorts.filter((escort) => escort.alive);
    }

    function buildRelationalNetwork(rule) {
      const connect = (a, b) => {
        if (a === b || a.links.includes(b.id)) return;
        a.links.push(b.id); b.links.push(a.id);
      };
      for (let attempt = 0; attempt < 120; attempt++) {
        for (const escort of escorts) escort.links = [];
        const order = [...escorts].sort(() => Math.random() - 0.5);
        for (let i = 1; i < order.length; i++) connect(order[i], order[(Math.random() * i) | 0]);
        for (let i = 0; i < order.length; i++) for (let j = i + 1; j < order.length; j++) if (Math.random() < 0.27) connect(order[i], order[j]);
        const ranked = roleGraphRank(escorts, rule);
        if (ranked[0] && (!ranked[1] || ranked[0].score !== ranked[1].score)) return;
      }
      throw new Error(`Could not generate an identifiable network for ${rule.id}.`);
    }

    function newRelationalFormation() {
      escorts = Array.from({ length: encounter.config.escortCount }, (_, index) => makeEscort(index, encounter.config.escortCount));
      buildRelationalNetwork(encounter.rule);
      encounter.target = null;
      encounter.previousTarget = null;
      encounter.candidate = null;
      encounter.candidateTimer = 0;
      encounter.targetStableFor = 0;
      encounter.roleSwitchAt = encounter.age;
      encounter.reticleHold = 0;
      encounter.reacquired = false;
    }

    function refreshRoleGeometry() {
      encounter.geometry = roleDriftGeometry(activeEscorts(), encounter.outsideAxis, {
        threshold: encounter.config.outsideThreshold,
        yScale: 0.84,
      });
      encounter.outsideVisual = encounter.geometry.axis;
      encounter.insideVisual = [-encounter.geometry.axis[0], -encounter.geometry.axis[1]];
      for (const metric of encounter.geometry.metrics) {
        metric.escort.outsideProjection = metric.projection;
        metric.escort.distanceToLeader = metric.distance;
      }
      return encounter.geometry;
    }

    function createEncounter(E) {
      const config = levelConfig(level);
      const initialTurn = choose([-1, 1]);
      const rule = choose(ROLE_DRIFT_RULES.filter(candidate => candidate.complexity === config.relationalTier));
      leader = {
        sx: E.w * 0.5,
        sy: E.top + E.h * 0.26,
        size: Math.max(13, E.u * 0.022),
        heading: 0,
        bank: initialTurn * 0.72,
        alive: true,
        trail: [],
        pulse: 0.6,
      };
      escorts = Array.from({ length: config.escortCount }, (_, index) => makeEscort(index, config.escortCount));
      enemyShots = [];
      tracers = [];
      encounter = {
        config,
        rule,
        phase: 'examples',
        exampleIndex: 0,
        nextExampleAt: config.exampleDuration,
        age: 0,
        depth: 0.075,
        lane: rand(0.08, -0.08),
        laneVelocity: 0,
        coursePhase: rand(TAU, 0),
        turnTarget: initialTurn,
        turnValue: initialTurn,
