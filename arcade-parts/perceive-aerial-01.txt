
  // Dream Machine / PERCEIVE — PARALLAX WING: CAUSAL INTERPOSITION.
  // Nothing is designated as an answer. Two visually identical fighters create
  // a live energy relation. Any third fighter entering the relation becomes
  // physically vulnerable; leaving it closes the shield again.
  function createPerceptionAerialGame() {
    const PROFILE_KEY = 'dream-unity.perceive.parallax-wing.causal-interposition.v1';

    let player;
    let reticle;
    let anchors = [];
    let candidates = [];
    let enemyShots = [];
    let tracers = [];
    let demoTracers = [];
    let encounter = null;
    let pointerGesture = null;
    let wave = 1;
    let level = 1;
    let streak = 0;
    let completedFormations = 0;
    let resolveTimer = 0;
    let stats;

    const rotate2 = (x, y, angle) => {
      const c = Math.cos(angle);
      const s = Math.sin(angle);
      return [x * c - y * s, x * s + y * c];
    };

    const shuffle = (items) => {
      const output = [...items];
      for (let index = output.length - 1; index > 0; index -= 1) {
        const other = (Math.random() * (index + 1)) | 0;
        [output[index], output[other]] = [output[other], output[index]];
      }
      return output;
    };

    function writeProfile(record) {
      try {
        const prior = JSON.parse(localStorage.getItem(PROFILE_KEY) || '[]');
        const rows = Array.isArray(prior) ? prior : [];
        rows.push(record);
        localStorage.setItem(PROFILE_KEY, JSON.stringify(rows.slice(-240)));
      } catch { /* local evidence storage remains optional */ }
    }

    function levelConfig(value) {
      const l = clamp(value, 1, 10);
      return {
        candidateCount: Math.min(7, 4 + Math.floor((l - 1) / 2)),
        insideTracks: Math.min(5, 3 + Math.floor((l - 1) / 3)),
        hitsRequired: Math.min(4, 2 + Math.floor((l - 1) / 3)),
        approachSpeed: 0.024 + l * 0.0018,
        weaveSpeed: 0.68 + l * 0.065,
        rotationSpeed: 0.09 + l * 0.016,
        halfSpan: 0.27 + l * 0.004,
        lineThreshold: Math.max(0.026, 0.052 - l * 0.0022),
        relationDwell: Math.max(0.08, 0.18 - l * 0.009),
        requireCompression: l >= 4,
        compressionAmplitude: l >= 4 ? 0.15 + Math.min(0.08, (l - 4) * 0.012) : 0.06,
        compressionSpeed: 0.56 + l * 0.035,
        compressionThreshold: l >= 4 ? 0.004 : -1,
        cueAlpha: l === 1 ? 1 : l === 2 ? 0.72 : l === 3 ? 0.42 : Math.max(0.045, 0.24 - (l - 4) * 0.028),
        enemyFire: l >= 3,
        fireInterval: Math.max(1.35, 3.4 - l * 0.17),
        projectileSpeed: 160 + l * 14,
        incomingLimit: Math.min(11, 3 + Math.floor(l / 2)),
        demo: l === 1 && wave === 1,
      };
    }

    function makeJet(id) {
      return {
        id,
        alive: true,
        role: 'candidate',
        track: 'inside',
        phaseOffset: 0,
        phaseRate: 1,
        radialBias: 1,
        localX: 0,
        localY: 0,
        previousLocalX: 0,
        previousLocalY: 0,
        velocityX: 0,
        velocityY: 0,
        sx: 0,
        sy: 0,
        size: 10,
        heading: 0,
        bank: 0,
        pulse: 0,
        hitFlash: 0,
        shieldFlash: 0,
        trail: [],
      };
    }

    function activeCandidates() {
      return candidates.filter((jet) => jet.alive);
    }

    function allEnemyJets() {
      return [...anchors.filter((jet) => jet.alive), ...activeCandidates()];
    }
