
    function spawnEnemyShot(E) {
      const shooters = allEnemyJets();
      if (!shooters.length || enemyShots.length >= encounter.config.incomingLimit) return;
      const shooter = choose(shooters);
      const dx = player.x - shooter.sx;
      const dy = player.y - shooter.sy;
      const length = Math.max(1, Math.hypot(dx, dy));
      const speed = encounter.config.projectileSpeed;
      enemyShots.push({
        x: shooter.sx,
        y: shooter.sy,
        vx: dx / length * speed + rand(12, -12),
        vy: dy / length * speed + rand(10, -10),
        r: 3.1 + encounter.depth * 1.6,
        life: 4.8,
      });
      E.sfx('tick', 0.64 + encounter.depth * 0.25);
    }

    function updateCombat(E, dt) {
      if (encounter.config.enemyFire && !encounter.solved && !encounter.failed) {
        encounter.fireTimer -= dt;
        if (encounter.fireTimer <= 0 && encounter.age > 1.45) {
          spawnEnemyShot(E);
          encounter.fireTimer = rand(encounter.config.fireInterval * 1.2, encounter.config.fireInterval * 0.72);
        }
      }

      for (let index = enemyShots.length - 1; index >= 0; index -= 1) {
        const shot = enemyShots[index];
        shot.x += shot.vx * dt;
        shot.y += shot.vy * dt;
        shot.life -= dt;
        if (shot.life <= 0 || shot.x < -30 || shot.x > E.w + 30 || shot.y > E.bottom + 35) {
          enemyShots.splice(index, 1);
          continue;
        }
        if (player.invulnerable <= 0 && dist(shot.x, shot.y, player.x, player.y) < player.size * 0.7 + shot.r) {
          enemyShots.splice(index, 1);
          hurt(E, 0.55, player.x, player.y, 'EVASIVE FAILURE');
        }
      }

      for (let index = tracers.length - 1; index >= 0; index -= 1) {
        tracers[index].life -= dt;
        if (tracers[index].life <= 0) tracers.splice(index, 1);
      }
    }

    function updatePlayer(E, dt) {
      const left = E.keys.has('arrowleft') || E.keys.has('a');
      const right = E.keys.has('arrowright') || E.keys.has('d');
      const up = E.keys.has('arrowup') || E.keys.has('w');
      const down = E.keys.has('arrowdown') || E.keys.has('s');
      const keyboardX = (right ? 1 : 0) - (left ? 1 : 0);
      const keyboardY = (down ? 1 : 0) - (up ? 1 : 0);
      if (keyboardX || keyboardY) {
        reticle.x += keyboardX * E.u * 0.72 * dt;
        reticle.y += keyboardY * E.u * 0.72 * dt;
        player.targetX += keyboardX * E.u * 0.58 * dt;
        player.targetY += keyboardY * E.u * 0.44 * dt;
      }
      reticle.x = clamp(reticle.x, 18, E.w - 18);
      reticle.y = clamp(reticle.y, E.top + 18, E.bottom - 42);
      player.targetX = clamp(player.targetX, 28, E.w - 28);
      player.targetY = clamp(player.targetY, E.h * 0.59, E.bottom - 18);

      const previousX = player.x;
      const response = 1 - Math.exp(-dt * 8.2);
      player.x = lerp(player.x, player.targetX, response);
      player.y = lerp(player.y, player.targetY, response);
      player.bank = lerp(player.bank, clamp((player.x - previousX) / Math.max(dt, 0.001) * 0.007, -1, 1), 1 - Math.exp(-dt * 9));
      player.fireCooldown = Math.max(0, player.fireCooldown - dt);
      player.heat = Math.max(0, player.heat - dt * (player.heat > 0.76 ? 0.22 : 0.32));
      player.invulnerable = Math.max(0, player.invulnerable - dt);
    }

    function failEncounter(E) {
      if (!encounter || encounter.solved || encounter.failed) return;
      encounter.failed = true;
      stats.errors['late-relational-acquisition'] = (stats.errors['late-relational-acquisition'] || 0) + 1;
      hurt(E, 1.05, (anchors[0].sx + anchors[1].sx) * 0.5, (anchors[0].sy + anchors[1].sy) * 0.5, 'FIELD BROKE THROUGH');
      for (const jet of allEnemyJets()) jet.alive = false;
      resolveTimer = 0.72;
      wave += 1;
      writeProfile({
        at: Date.now(),
        type: 'formation',
        completed: false,
        level,
        errorType: 'late-relational-acquisition',
        successfulOpenings: encounter.hits,
        requiredOpenings: encounter.config.hitsRequired,
        roleMigrations: encounter.roleMigrations,
      });
    }
