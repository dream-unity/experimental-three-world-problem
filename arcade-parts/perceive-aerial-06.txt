      player.heat = Math.min(1, player.heat + 0.13);
      stats.shots++;
      const lock = nearestAimObject();
      const targetX = lock ? lock.value.x ?? lock.value.sx : reticle.x;
      const targetY = lock ? lock.value.y ?? lock.value.sy : reticle.y;
      tracers.push({ x1: player.x - 5, y1: player.y - player.size * 0.8, x2: targetX, y2: targetY, life: 0.12, max: 0.12 });
      tracers.push({ x1: player.x + 5, y1: player.y - player.size * 0.8, x2: targetX + 2, y2: targetY, life: 0.1, max: 0.1 });
      E.sfx('tap', 1.6);
      if (!lock) return;
      stats.hits++;
      if (lock.type === 'projectile') {
        const index = enemyShots.indexOf(lock.value);
        if (index >= 0) enemyShots.splice(index, 1);
        E.addScore(20, targetX, targetY, '+20');
        E.burst(targetX, targetY, E.secondary, 8, 120, 1.8);
        return;
      }
      const jet = lock.value;
      if (scene.evaluation.active && jet.id === scene.evaluation.targetId) completeCorrectShot(E, jet);
      else rejectShot(E, jet);
    }

    function spawnEnemyShot(E) {
      if (scene.demoKind || enemyShots.length >= scene.config.incomingLimit) return;
      const shooters = scene.jets.filter((jet) => jet.alive);
      if (!shooters.length) return;
      const shooter = choose(shooters);
      const dx = player.x - shooter.sx;
      const dy = player.y - shooter.sy;
      const length = Math.max(1, Math.hypot(dx, dy));
      const speed = scene.config.projectileSpeed;
      enemyShots.push({
        x: shooter.sx,
        y: shooter.sy,
        vx: dx / length * speed + rand(13, -13),
        vy: dy / length * speed + rand(9, -9),
        r: 3.1 + scene.depth * 1.7,
        life: 4.8,
      });
      E.sfx('tick', 0.66 + scene.depth * 0.28);
    }

    function updatePlayer(E, dt) {
      const left = E.keys.has('arrowleft') || E.keys.has('a');
      const right = E.keys.has('arrowright') || E.keys.has('d');
      const up = E.keys.has('arrowup') || E.keys.has('w');
      const down = E.keys.has('arrowdown') || E.keys.has('s');
      const keyboardX = (right ? 1 : 0) - (left ? 1 : 0);
      const keyboardY = (down ? 1 : 0) - (up ? 1 : 0);
      if (keyboardX || keyboardY) {
        reticle.x += keyboardX * E.u * 0.72 * dt;
        reticle.y += keyboardY * E.u * 0.72 * dt;
        player.targetX += keyboardX * E.u * 0.58 * dt;
        player.targetY += keyboardY * E.u * 0.44 * dt;
      }
      reticle.x = clamp(reticle.x, 18, E.w - 18);
      reticle.y = clamp(reticle.y, E.top + 18, E.bottom - 42);
      player.targetX = clamp(player.targetX, 28, E.w - 28);
      player.targetY = clamp(player.targetY, E.h * 0.59, E.bottom - 18);
      const previousX = player.x;
      const response = 1 - Math.exp(-dt * 8.2);
      player.x = lerp(player.x, player.targetX, response);
      player.y = lerp(player.y, player.targetY, response);
      player.bank = lerp(player.bank, clamp((player.x - previousX) / Math.max(dt, 0.001) * 0.007, -1, 1), 1 - Math.exp(-dt * 9));
      player.fireCooldown = Math.max(0, player.fireCooldown - dt);
      player.heat = Math.max(0, player.heat - dt * (player.heat > 0.76 ? 0.22 : 0.32));
      player.invulnerable = Math.max(0, player.invulnerable - dt);
    }

    function updateCombat(E, dt) {
      wrongShotCooldown = Math.max(0, wrongShotCooldown - dt);
      if (!scene.demoKind && !scene.resolved) {
        scene.fireTimer -= dt;
        if (scene.fireTimer <= 0 && scene.age > 1.1) {
          spawnEnemyShot(E);
          scene.fireTimer = rand(scene.config.enemyFireInterval * 1.2, scene.config.enemyFireInterval * 0.72);
        }
      }
      for (let i = enemyShots.length - 1; i >= 0; i--) {
        const shot = enemyShots[i];
        shot.x += shot.vx * dt;
        shot.y += shot.vy * dt;
        shot.life -= dt;
        if (shot.life <= 0 || shot.x < -30 || shot.x > E.w + 30 || shot.y > E.bottom + 35) {
          enemyShots.splice(i, 1);
          continue;
        }
        if (player.invulnerable <= 0 && dist(shot.x, shot.y, player.x, player.y) < player.size * 0.7 + shot.r) {
          enemyShots.splice(i, 1);
          hurt(E, 0.55, player.x, player.y, 'EVASIVE FAILURE');
        }
      }
      for (let i = tracers.length - 1; i >= 0; i--) {
        tracers[i].life -= dt;
        if (tracers[i].life <= 0) tracers.splice(i, 1);
      }
    }
