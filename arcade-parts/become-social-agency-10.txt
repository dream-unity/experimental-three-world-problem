  function s10ComboScore(combo,selected,baseRefs,socialRefs,coverage,baseUse,profileUse){
    const selectedDistances=selected.map(ref=>s10CombinedDistance(combo,ref));
    const pairMin=selectedDistances.length?Math.min(...selectedDistances):1;
    const pairAvg=selectedDistances.length?selectedDistances.reduce((a,b)=>a+b,0)/selectedDistances.length:1;
    const baseDistances=baseRefs.map(ref=>d9SigDistance(combo.baseSignature,ref.signature||ref.baseSignature||{}));
    const socialDistances=socialRefs.map(ref=>s10SocialDistance(combo.socialSignature,ref.socialSignature||{}));
    const baseMin=baseDistances.length?Math.min(...baseDistances):1;
    const socialMin=socialDistances.length?Math.min(...socialDistances):1;
    const selectedSocialAxes=selected.length?Math.min(...selected.map(ref=>s10SocialAxisDiff(combo.socialSignature,ref.socialSignature))):S10_AXES.length;
    let score=pairMin*0.46+pairAvg*0.08+baseMin*0.16+socialMin*0.22+(selectedSocialAxes/S10_AXES.length)*0.08;
    for(const axis of S10_COVERAGE_AXES)if(!coverage[axis].has(combo.socialSignature[axis]))score+=0.018;
    const recentBase=baseRefs.slice(-D9_WORLDS.length).some(ref=>(ref.signature||ref.baseSignature)?.family===combo.world.id);
    const recentProfile=socialRefs.slice(-S10_RECENT_HARD).some(ref=>ref.profileId===combo.profile.id||ref.socialSignature?.profile===combo.profile.id);
    if(recentBase)score-=0.34;
    if(recentProfile)score-=0.48;
    score-=(baseUse.get(combo.world.id)||0)*0.025;
    score-=(profileUse.get(combo.profile.id)||0)*0.03;
    if(selectedSocialAxes<10)score-=0.24;
    return score+Math.random()*0.001;
  }

  function s10SelectCombos(count){
    const baseRefs=d9MemoryRead();
    const socialRefs=s10MemoryRead();
    const selected=[];
    const usedWorlds=new Set(),usedProfiles=new Set();
    const coverage=Object.fromEntries(S10_COVERAGE_AXES.map(axis=>[axis,new Set()]));
    const baseUse=new Map(),profileUse=new Map();
    for(const ref of baseRefs){const id=ref.signature?.family;if(id)baseUse.set(id,(baseUse.get(id)||0)+1);}
    for(const ref of socialRefs){const id=ref.profileId||ref.socialSignature?.profile;if(id)profileUse.set(id,(profileUse.get(id)||0)+1);}
    for(let slot=0;slot<count;slot++){
      let best=null,bestScore=-Infinity;
      for(const world of D9_WORLDS){
        if(usedWorlds.has(world.id))continue;
        const baseSignature={...world.sig,family:world.id};
        for(const profile of S10_PROFILES){
          if(usedProfiles.has(profile.id))continue;
          const combo={world,profile,baseSignature,socialSignature:s10SocialSig(profile)};
          const score=s10ComboScore(combo,selected,baseRefs,socialRefs,coverage,baseUse,profileUse);
          if(score>bestScore){best=combo;bestScore=score;}
        }
      }
      if(!best)throw new Error('Social agency planner exhausted its orthogonal world/profile space.');
      selected.push(best);usedWorlds.add(best.world.id);usedProfiles.add(best.profile.id);
      for(const axis of S10_COVERAGE_AXES)coverage[axis].add(best.socialSignature[axis]);
      baseRefs.push({signature:best.baseSignature});
      socialRefs.push({baseSignature:best.baseSignature,socialSignature:best.socialSignature,profileId:best.profile.id});
      baseUse.set(best.world.id,(baseUse.get(best.world.id)||0)+1);
      profileUse.set(best.profile.id,(profileUse.get(best.profile.id)||0)+1);
    }
    return selected;
  }
