
  // Dream Maker / Become — internet-only live AI scenario director.
  // Every scenario is generated remotely on demand. No model weights or AI runtime are downloaded to the browser.
  const BECOME_LIVE_API = (() => {
    const explicit = typeof window.DREAM_UNITY_BECOME_API === 'string' ? window.DREAM_UNITY_BECOME_API.trim() : '';
    return explicit || 'https://dream-unity-become-live.vercel.app/api/become-scenario';
  })();
  const BECOME_GROQ_KEYS_URL = 'https://console.groq.com/keys';
  const BECOME_GROQ_SESSION_KEY = 'dream-unity:become:groq-key';
  const BECOME_LIVE_HISTORY_LIMIT = 8;
  const BECOME_LIVE_VISUALS = new Set(['hockey','stage','mountain','orbit','courtroom','ocean','sprint','conversation','wildfire','contact','future']);
  const becomeLivePending = new Map();
  let becomeLiveSessionNonce = '';
  let becomeLiveLastError = '';
  let becomeLiveLastErrorCode = '';

  const bLiveRequired = ['id','title','tag','visual','premise','sensory','objects','body','atmosphere','motion','stakes','identity','choice','exit'];
  const bLiveNewNonce = () => `${Date.now().toString(36)}-${Math.random().toString(36).slice(2,10)}`;
  const bLiveScenarioSummary = scenario => ({
    title: String(scenario?.title || '').slice(0,100),
    tag: String(scenario?.tag || '').slice(0,120),
    premise: String(scenario?.premise || '').slice(0,700),
    dimensions: scenario?.dimensions && typeof scenario.dimensions === 'object' ? scenario.dimensions : undefined
  });
  const bLivePerformance = () => BECOME_DRILLS.map(metric => ({
    id: metric.id,
    title: metric.title,
    score: Number.isFinite(becomeState.scores?.[metric.id]) ? becomeState.scores[metric.id] : null
  })).filter(item => item.score !== null);

  function bLiveGetGroqKey(){
    try{return String(sessionStorage.getItem(BECOME_GROQ_SESSION_KEY)||'').trim();}catch{return '';}
  }
  function bLiveSetGroqKey(value){
    const key=String(value||'').trim();
    if(key.length < 20)return '';
    try{sessionStorage.setItem(BECOME_GROQ_SESSION_KEY,key);}catch{}
    return key;
  }
  function bLiveClearGroqKey(){
    try{sessionStorage.removeItem(BECOME_GROQ_SESSION_KEY);}catch{}
  }
  function bLiveResolveGroqKey(){
    const input=document.getElementById('becomeGroqKey');
    const entered=String(input?.value||'').trim();
    if(entered)return bLiveSetGroqKey(entered);
    return bLiveGetGroqKey();
  }
  function bLiveCredentialInput(){
    return `<div class="become-divider"></div><p class="become-label">OPTIONAL CLOUD KEY</p><label class="become-custom">GROQ BACKUP CREDENTIAL <input id="becomeGroqKey" type="password" autocomplete="off" spellcheck="false" placeholder="OPTIONAL GROQ API KEY"></label><p class="become-microcopy">Generation always stays online. This key is only needed if the remote service is not provisioned with its own server credential. A pasted key is kept only in this browser tab's session storage and is sent only to the remote generation service. Create one at <a href="${BECOME_GROQ_KEYS_URL}" target="_blank" rel="noopener noreferrer">console.groq.com/keys</a>.</p>`;
  }

  function bLiveValidateScenario(raw){
    if(!raw || typeof raw !== 'object') throw new Error('The live generator returned an invalid scenario object.');
    const scenario = {...raw};
    for(const key of bLiveRequired){
      if(typeof scenario[key] !== 'string' || scenario[key].trim().length < 2) throw new Error(`Live scenario is missing ${key}.`);
      scenario[key] = scenario[key].trim();
    }
    scenario.visual = BECOME_LIVE_VISUALS.has(scenario.visual) ? scenario.visual : 'future';
    scenario.id = scenario.id.toLowerCase().replace(/[^a-z0-9-]+/g,'-').replace(/^-+|-+$/g,'').slice(0,72) || `live-${bLiveNewNonce()}`;
    if(!scenario.id.startsWith('live-')) scenario.id = `live-${scenario.id}-${Math.random().toString(36).slice(2,6)}`;
    if(!scenario.dimensions || typeof scenario.dimensions !== 'object') scenario.dimensions = {};
    return scenario;
  }

  function bLiveHistory(index){
    return becomeState.queue
      .slice(Math.max(0,index-BECOME_LIVE_HISTORY_LIMIT),index)
      .filter(Boolean)
      .map(bLiveScenarioSummary);
  }

  async function bCloudGenerate(index,sessionNonce,groqKey){
    const previous=bRead('latest');
    const payload={
      trialIndex:index,trialCount:becomeState.count,sessionNonce,
      recentScenarios:bLiveHistory(index),currentPerformance:bLivePerformance(),previousWeakestMetric:previous?.weakest||null
    };
    const headers={'Content-Type':'application/json'};
    if(groqKey)headers['X-Groq-Api-Key']=groqKey;
    const response=await fetch(BECOME_LIVE_API,{
      method:'POST',headers,cache:'no-store',credentials:'omit',body:JSON.stringify(payload)
    });
    let data=null;try{data=await response.json();}catch{}
    if(!response.ok){
      const error=new Error(data?.error||`Live cloud request failed (${response.status}).`);
      error.code=data?.code||'';
      throw error;
    }
    return bLiveValidateScenario(data?.scenario);
  }

  async function bLiveGenerate(index,sessionNonce){
    const groqKey=bLiveGetGroqKey();
    try{return await bCloudGenerate(index,sessionNonce,groqKey);}
    catch(error){
      if(error?.code==='GROQ_KEY_INVALID')bLiveClearGroqKey();
      throw error;
    }
  }

  function bLiveLoading(index,copy){
    becomeState.stage='live-generating';
    bPanel(`${bHeader(`LIVE CLOUD · SCENARIO ${index+1} OF ${becomeState.count}`,'GENERATING',0)}<p class="become-kicker">REAL-TIME INTERNET GENERATION</p><h2 class="become-title compact">BUILDING A<br>NEW SCENARIO</h2><p class="become-copy">${bEscape(copy||'GPT-OSS 120B is constructing a new world remotely now.')}</p><div class="become-facts"><span class="become-fact">INTERNET GENERATED</span><span class="become-fact">NOVELTY CHECKED</span><span class="become-fact">NO MODEL DOWNLOAD</span><span class="become-fact">NO STORED FALLBACK</span></div><p class="become-microcopy">The browser sends a compact generation request and receives the finished scenario. No language model, model weights or AI runtime are downloaded to this device.</p>`);
  }

  function bLiveError(error){
    becomeLiveLastError=error instanceof Error?error.message:String(error||'Live generation failed.');
    becomeLiveLastErrorCode=String(error?.code||'');
    becomeState.stage='live-error';
    const credentialCopy=(becomeLiveLastErrorCode==='GROQ_KEY_REQUIRED'||becomeLiveLastErrorCode==='GROQ_KEY_INVALID')?bLiveCredentialInput():'';
    bPanel(`${bHeader('LIVE CLOUD','GENERATION ERROR',0)}<p class="become-kicker">REMOTE ENGINE UNAVAILABLE</p><h2 class="become-title compact">NO STORED WORLD<br>WAS SUBSTITUTED</h2><p class="become-copy">${bEscape(becomeLiveLastError)}</p>${credentialCopy}<p class="become-rule">Become uses internet generation only. It will not download a local model and will not substitute a prewritten scenario.</p><div class="become-actions"><button class="become-primary" type="button" data-become-action="begin-session">RETRY LIVE GENERATION</button></div>`);
  }

  function bLivePrefetch(index){
    if(index>=becomeState.count||becomeState.queue[index]||becomeLivePending.has(index))return becomeLivePending.get(index)||Promise.resolve(becomeState.queue[index]);
    const nonce=becomeLiveSessionNonce;
    let promise;
    promise=bLiveGenerate(index,nonce).then(scenario=>{
      if(nonce===becomeLiveSessionNonce&&becomeState.queue.length===becomeState.count)becomeState.queue[index]=scenario;
      return scenario;
    }).catch(error=>{becomeLiveLastError=error instanceof Error?error.message:String(error);becomeLiveLastErrorCode=String(error?.code||'');throw error;}).finally(()=>{
      if(becomeLivePending.get(index)===promise)becomeLivePending.delete(index);
    });
    promise.catch(()=>{});becomeLivePending.set(index,promise);return promise;
  }

  bRenderSetup=function(){
    becomeState.stage='setup';
    const previous=bRead('latest');
    const choices=BECOME_PRESETS.map(value=>`<button class="become-choice ${becomeState.count===value?'active':''}" type="button" data-become-action="set-count" data-value="${value}">${value}</button>`).join('');
    const hasKey=Boolean(bLiveGetGroqKey());
    bPanel(`<p class="become-kicker">DREAM MAKER · BECOME · LIVE AI</p><h1 class="become-title">REALITY<br>TRAINING LAB</h1><p class="become-copy">Every trial is generated live over the internet at the moment it is needed. Become does not select from the ten-world bank and does not run an AI model on this device. The remote director generates a fresh world, rejects excessive similarity to recent worlds, then returns it for the existing perception, embodiment, consequence, identity and action drills.</p><p class="become-rule"><b>CONTROL RULE:</b> operate as if the scenario is real while retaining awareness that you selected it. Stop if intensity becomes unhelpful. Never use this while driving or during hazardous activity.</p><div class="become-grid three"><div class="become-card"><small>DEFAULT</small><strong>LIVE CLOUD</strong><p>Generated remotely on demand. No browser model or model-weight download.</p></div><div class="become-card"><small>MODEL</small><strong>GPT-OSS 120B</strong><p>Each world is constructed by the online generation service in real time.</p></div><div class="become-card"><small>NOVELTY</small><strong>REJECT + RETRY</strong><p>Recent worlds are compared and excessive similarity is rejected before delivery.</p></div></div><div class="become-facts"><span class="become-fact">INTERNET ONLY</span><span class="become-fact">NO MODEL DOWNLOAD</span><span class="become-fact">NO STORED FALLBACK</span>${hasKey?'<span class="become-fact">CLOUD KEY LOADED FOR THIS TAB</span>':''}</div><div class="become-divider"></div><p class="become-label">NUMBER OF LIVE SCENARIOS</p><div class="become-counts">${choices}</div><label class="become-custom">CUSTOM 1–10 <input id="becomeCustomCount" type="number" min="1" max="10" value="${becomeState.count}" inputmode="numeric"></label>${bLiveCredentialInput()}<div class="become-actions"><button class="become-primary" type="button" data-become-action="begin-session">GENERATE LIVE SESSION</button></div><p class="become-microcopy">${previous?`Previous adaptive-entry score: ${Number(previous.adaptive||0).toFixed(1)}. Its weakest numerical faculty may shape the first trial.`:'No previous numerical session stored.'} Personal life-transfer descriptions are not sent to the scenario generator.</p>`);
  };

  bBeginSession=async function(){
    bLiveResolveGroqKey();
    const nonce=bLiveNewNonce();
    becomeLiveSessionNonce=nonce;becomeLivePending.clear();becomeLiveLastError='';becomeLiveLastErrorCode='';
    becomeState.queue=Array.from({length:becomeState.count},()=>null);
    becomeState.scenarioIndex=0;becomeState.results=[];becomeState.transferScores={};
    bLiveLoading(0,'The online model is constructing the first world now. Nothing is being generated or downloaded locally.');
    try{
      const first=await bLiveGenerate(0,nonce);
      if(nonce!==becomeLiveSessionNonce)return;
      becomeState.queue[0]=first;bRenderBrief();
    }catch(error){if(nonce===becomeLiveSessionNonce)bLiveError(error);}
  };

  const bLiveOriginalRun=bRun;
  bRun=function(){
    if(becomeState.track==='scenario'&&becomeState.drillIndex===BECOME_DRILLS.length-1)bLivePrefetch(becomeState.scenarioIndex+1);
    return bLiveOriginalRun();
  };

  bNext=async function(){
    const next=becomeState.scenarioIndex+1;
    if(next>=becomeState.queue.length){bRenderTransferForm();return;}
    if(!becomeState.queue[next]){
      bLiveLoading(next,'The next world is being generated online from the worlds you have already inhabited and the scores you just produced.');
      try{const scenario=await bLivePrefetch(next);if(!scenario)throw new Error('The live generator returned no scenario.');}
      catch(error){bLiveError(error);return;}
    }
    becomeState.scenarioIndex=next;bRenderBrief();
  };

  window.addEventListener('dreamunity:game-closed',()=>{
    becomeLiveSessionNonce='';becomeLivePending.clear();
  });
