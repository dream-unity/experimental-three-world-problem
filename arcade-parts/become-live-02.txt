
  // Dream Maker / Become — zero-key, internet-only live AI scenario director.
  // Every scenario is generated remotely on demand. No model weights, AI runtime, account or API key are required in the browser.
  const BECOME_LIVE_API = 'https://blockrun.ai/api/v1/chat/completions';
  const BECOME_LIVE_MODELS = ['nvidia/gpt-oss-120b','nvidia/step-3.7-flash','nvidia/mistral-nemotron'];
  const BECOME_LIVE_HISTORY_LIMIT = 8;
  const BECOME_LIVE_VISUALS = new Set(['hockey','stage','mountain','orbit','courtroom','ocean','sprint','conversation','wildfire','contact','future']);
  const BECOME_DIMENSION_KEYS = ['environment','role','goal','pressure','body_dynamics','decision_structure','emotional_tone','social_structure'];
  const becomeLivePending = new Map();
  let becomeLiveSessionNonce = '';
  let becomeLiveLastError = '';
  let becomeLiveLastModel = BECOME_LIVE_MODELS[0];

  const bLiveRequired = ['title','tag','visual','premise','sensory','objects','body','atmosphere','motion','stakes','identity','choice','exit'];
  const bLiveNewNonce = () => `${Date.now().toString(36)}-${Math.random().toString(36).slice(2,10)}`;
  const bLiveScenarioSummary = scenario => ({
    title: String(scenario?.title || '').slice(0,100),
    tag: String(scenario?.tag || '').slice(0,120),
    premise: String(scenario?.premise || '').slice(0,700),
    dimensions: scenario?.dimensions && typeof scenario.dimensions === 'object' ? scenario.dimensions : undefined
  });
  const bLivePerformance = () => BECOME_DRILLS.map(metric => ({
    id: metric.id,
    title: metric.title,
    score: Number.isFinite(becomeState.scores?.[metric.id]) ? becomeState.scores[metric.id] : null
  })).filter(item => item.score !== null);

  function bLiveValidateScenario(raw){
    if(!raw || typeof raw !== 'object' || Array.isArray(raw)) throw new Error('The live model returned an invalid scenario object.');
    const scenario = {...raw};
    for(const key of bLiveRequired){
      if(typeof scenario[key] !== 'string' || scenario[key].trim().length < 2) throw new Error(`Live scenario is missing ${key}.`);
      scenario[key] = scenario[key].trim();
    }
    scenario.visual = BECOME_LIVE_VISUALS.has(scenario.visual.toLowerCase()) ? scenario.visual.toLowerCase() : 'future';
    if(!scenario.dimensions || typeof scenario.dimensions !== 'object' || Array.isArray(scenario.dimensions)) throw new Error('Live scenario is missing relational dimensions.');
    scenario.dimensions = {...scenario.dimensions};
    for(const key of BECOME_DIMENSION_KEYS){
      if(typeof scenario.dimensions[key] !== 'string' || scenario.dimensions[key].trim().length < 2) throw new Error(`Live scenario is missing dimension ${key}.`);
      scenario.dimensions[key] = scenario.dimensions[key].trim();
    }
    scenario.id = `live-${Date.now().toString(36)}-${Math.random().toString(36).slice(2,9)}`;
    return scenario;
  }

  function bLiveHistory(index){
    return becomeState.queue
      .slice(Math.max(0,index-BECOME_LIVE_HISTORY_LIMIT),index)
      .filter(Boolean)
      .map(bLiveScenarioSummary);
  }

  const bLiveStopWords = new Set(['the','and','that','with','from','into','your','you','for','this','then','while','have','has','are','was','were','one','two','three','their','there','when','where','what','through','under','over','before','after','about','only','will','must','can','could','would','should']);
  function bLiveTokens(text){
    const tokens=String(text||'').toLowerCase().match(/[a-z0-9]+/g)||[];
    return new Set(tokens.filter(token=>token.length>2&&!bLiveStopWords.has(token)));
  }
  function bLiveJaccard(a,b){
    if(!a.size||!b.size)return 0;
    let overlap=0;for(const token of a)if(b.has(token))overlap++;
    return overlap/(a.size+b.size-overlap);
  }
  function bLiveScenarioText(item){
    return [item?.title,item?.tag,item?.premise,...BECOME_DIMENSION_KEYS.map(key=>item?.dimensions?.[key])].filter(Boolean).join(' ');
  }
  function bLiveTooSimilar(candidate,history){
    const candidateTokens=bLiveTokens(bLiveScenarioText(candidate));
    for(const prior of history){
      if(bLiveJaccard(candidateTokens,bLiveTokens(bLiveScenarioText(prior)))>=0.34)return true;
      let overlap=0;
      for(const key of BECOME_DIMENSION_KEYS){
        const a=String(candidate?.dimensions?.[key]||'').toLowerCase().trim();
        const b=String(prior?.dimensions?.[key]||'').toLowerCase().trim();
        if(a&&b&&(a===b||a.includes(b)||b.includes(a)))overlap++;
      }
      if(overlap>=4)return true;
    }
    return false;
  }

  function bLivePrompt(index,attempt,rejected){
    const previous=bRead('latest');
    const payload={
      trialIndex:index,
      trialCount:becomeState.count,
      generationNonce:bLiveNewNonce(),
      recentScenarios:bLiveHistory(index),
      rejectedThisTrial:rejected,
      currentPerformance:bLivePerformance(),
      previousWeakestMetric:previous?.weakest||null
    };
    return `You are the live scenario director for Dream Unity's BECOME training lab. Generate exactly one genuinely new first-person scenario for controlled imagination training.\n\nNOVELTY IS PRIMARY. Treat recentScenarios and rejectedThisTrial as exclusions, not inspiration. Do not merely reskin them. Change at least six of these eight dimensions relative to every recent scenario: environment, role, goal, pressure, body dynamics, decision structure, emotional tone, social structure. Do not reuse titles, signature objects, central dilemmas or characteristic phrasing. Prefer an unexpected but coherent combination that is immediately inhabitable.\n\nTRAINING QUALITY. Make the world concrete, spatially legible and phenomenologically rich. Every prose field must contribute a different operation: perception, touch, embodiment, atmosphere, movement, stakes, identity, agency or exit. Identity means an embodied competent perspective, not grandiosity or magical certainty. Choice must contain more than one plausible option. Exit must explicitly dissolve the simulation and reorient attention to present reality. Keep fields concise enough for rapid training.\n\nSAFETY AND CONTROL. This is imaginative training, not real-world operational instruction. Do not give procedural guidance for weapons, crime, self-harm, dangerous stunts, medical emergencies or other hazardous activity. Avoid sexual content, graphic violence and trauma bait. Stakes may be intense only when non-graphic, non-instructional and psychologically controllable.\n\nADAPTIVE INPUT: ${JSON.stringify(payload)}\nATTEMPT: ${attempt}\n\nReturn ONLY one valid JSON object and no markdown. It must have exactly this shape:\n{\n  "title":"short vivid title",\n  "tag":"short training tag",\n  "visual":"one of hockey, stage, mountain, orbit, courtroom, ocean, sprint, conversation, wildfire, contact, future",\n  "premise":"1-2 concrete first-person sentences",\n  "sensory":"what I see and hear",\n  "objects":"tactile objects, weight, texture and temperature",\n  "body":"first-person posture and spatial embodiment",\n  "atmosphere":"environmental and emotional texture",\n  "motion":"kinaesthetic movement and balance",\n  "stakes":"meaningful but controllable consequences",\n  "identity":"the competent perspective I inhabit",\n  "choice":"a genuine decision with multiple plausible options",\n  "exit":"explicitly dissolve the simulation and orient to present surroundings",\n  "dimensions":{\n    "environment":"concise setting category",\n    "role":"concise role category",\n    "goal":"concise goal category",\n    "pressure":"concise pressure category",\n    "body_dynamics":"concise body-dynamics category",\n    "decision_structure":"concise decision-structure category",\n    "emotional_tone":"concise emotional-tone category",\n    "social_structure":"concise social-structure category"\n  }\n}`;
  }

  function bLiveExtractText(data){
    const content=data?.choices?.[0]?.message?.content;
    if(typeof content==='string')return content;
    if(Array.isArray(content))return content.map(part=>typeof part==='string'?part:String(part?.text||'')).join('');
    return '';
  }
  function bLiveParseJSON(text){
    let raw=String(text||'').trim();
    if(raw.startsWith('```'))raw=raw.replace(/^```(?:json)?\s*/i,'').replace(/\s*```$/,'').trim();
    const first=raw.indexOf('{'),last=raw.lastIndexOf('}');
    if(first>=0&&last>first)raw=raw.slice(first,last+1);
    return JSON.parse(raw);
  }

  async function bLiveCallModel(model,prompt){
    const controller=new AbortController();
    const timeout=setTimeout(()=>controller.abort(),30000);
    try{
      const response=await fetch(BECOME_LIVE_API,{
        method:'POST',
        headers:{'Content-Type':'application/json'},
        mode:'cors',
        cache:'no-store',
        credentials:'omit',
        signal:controller.signal,
        body:JSON.stringify({
          model,
          messages:[
            {role:'system',content:'Generate exactly the requested controlled-imagination scenario. Output valid JSON only.'},
            {role:'user',content:prompt}
          ],
          stream:false,
          temperature:0.94,
          top_p:0.95,
          max_tokens:1800,
          response_format:{type:'json_object'}
        })
      });
      let data=null;try{data=await response.json();}catch{}
      if(!response.ok){
        const message=String(data?.error?.message||data?.message||data?.error||`Remote model request failed (${response.status}).`);
        const error=new Error(message);error.code=`HTTP_${response.status}`;throw error;
      }
      const text=bLiveExtractText(data);
      if(!text)throw new Error('The remote model returned no scenario text.');
      return bLiveValidateScenario(bLiveParseJSON(text));
    }catch(error){
      if(error?.name==='AbortError')throw new Error('The live model timed out while generating this world.');
      throw error;
    }finally{clearTimeout(timeout);}
  }

  async function bLiveCallWithFailover(prompt){
    let lastError=null;
    for(const model of BECOME_LIVE_MODELS){
      try{
        const scenario=await bLiveCallModel(model,prompt);
        becomeLiveLastModel=model;
        return scenario;
      }catch(error){lastError=error;}
    }
    throw lastError||new Error('All live cloud models are temporarily unavailable.');
  }

  async function bCloudGenerate(index){
    const history=bLiveHistory(index);
    const rejected=[];
    for(let attempt=1;attempt<=3;attempt++){
      const scenario=await bLiveCallWithFailover(bLivePrompt(index,attempt,rejected));
      if(!bLiveTooSimilar(scenario,history))return scenario;
      rejected.push({title:scenario.title,premise:scenario.premise.slice(0,260),dimensions:scenario.dimensions});
    }
    throw new Error('Live generation produced worlds too similar to recent trials after three novelty checks.');
  }

  async function bLiveGenerate(index,sessionNonce){
    if(sessionNonce!==becomeLiveSessionNonce)throw new Error('This generation session is no longer active.');
    return bCloudGenerate(index);
  }

  function bLiveLoading(index,copy){
    becomeState.stage='live-generating';
    bPanel(`${bHeader(`LIVE CLOUD · SCENARIO ${index+1} OF ${becomeState.count}`,'GENERATING',0)}<p class="become-kicker">REAL-TIME INTERNET GENERATION</p><h2 class="become-title compact">BUILDING A<br>NEW SCENARIO</h2><p class="become-copy">${bEscape(copy||'GPT-OSS 120B is constructing a new world remotely now.')}</p><div class="become-facts"><span class="become-fact">NO API KEY</span><span class="become-fact">INTERNET GENERATED</span><span class="become-fact">NOVELTY CHECKED</span><span class="become-fact">NO MODEL DOWNLOAD</span><span class="become-fact">NO STORED FALLBACK</span></div><p class="become-microcopy">Generation runs on remote cloud inference. The browser sends only the scenario-generation context and receives the finished world; no language model or model weights are installed on this device.</p>`);
  }

  function bLiveError(error){
    becomeLiveLastError=error instanceof Error?error.message:String(error||'Live generation failed.');
    becomeState.stage='live-error';
    bPanel(`${bHeader('LIVE CLOUD','GENERATION ERROR',0)}<p class="become-kicker">REMOTE GENERATION TEMPORARILY UNAVAILABLE</p><h2 class="become-title compact">LIVE WORLD<br>NOT YET GENERATED</h2><p class="become-copy">${bEscape(becomeLiveLastError)}</p><p class="become-rule">No key, account, local model or prewritten world is required. Become only advances when a genuinely live internet-generated scenario has been received and novelty-checked.</p><div class="become-actions"><button class="become-primary" type="button" data-become-action="begin-session">RETRY LIVE GENERATION</button></div>`);
  }

  function bLivePrefetch(index){
    if(index>=becomeState.count||becomeState.queue[index]||becomeLivePending.has(index))return becomeLivePending.get(index)||Promise.resolve(becomeState.queue[index]);
    const nonce=becomeLiveSessionNonce;
    let promise;
    promise=bLiveGenerate(index,nonce).then(scenario=>{
      if(nonce===becomeLiveSessionNonce&&becomeState.queue.length===becomeState.count)becomeState.queue[index]=scenario;
      return scenario;
    }).catch(error=>{becomeLiveLastError=error instanceof Error?error.message:String(error);throw error;}).finally(()=>{
      if(becomeLivePending.get(index)===promise)becomeLivePending.delete(index);
    });
    promise.catch(()=>{});becomeLivePending.set(index,promise);return promise;
  }

  bRenderSetup=function(){
    becomeState.stage='setup';
    const previous=bRead('latest');
    const choices=BECOME_PRESETS.map(value=>`<button class="become-choice ${becomeState.count===value?'active':''}" type="button" data-become-action="set-count" data-value="${value}">${value}</button>`).join('');
    bPanel(`<p class="become-kicker">DREAM MAKER · BECOME · LIVE AI</p><h1 class="become-title">REALITY<br>TRAINING LAB</h1><p class="become-copy">Every trial is generated live over the internet at the moment it is needed. Its weakest numerical faculty may shape the first trial.`:'No previous numerical session stored.'} Personal life-transfer descriptions are not sent to the live generator.</p>`);
  };

  bBeginSession=async function(){
    const nonce=bLiveNewNonce();
    becomeLiveSessionNonce=nonce;becomeLivePending.clear();becomeLiveLastError='';becomeLiveLastModel=BECOME_LIVE_MODELS[0];
    becomeState.queue=Array.from({length:becomeState.count},()=>null);
    becomeState.scenarioIndex=0;becomeState.results=[];becomeState.transferScores={};
    bLiveLoading(0,'GPT-OSS 120B is generating the first world on remote infrastructure now. No account, API key or local download is involved.');
    try{
      const first=await bLiveGenerate(0,nonce);
      if(nonce!==becomeLiveSessionNonce)return;
      becomeState.queue[0]=first;bRenderBrief();
    }catch(error){if(nonce===becomeLiveSessionNonce)bLiveError(error);}
  };

  const bLiveOriginalRun=bRun;
  bRun=function(){
    if(becomeState.track==='scenario'&&becomeState.drillIndex===BECOME_DRILLS.length-1)bLivePrefetch(becomeState.scenarioIndex+1);
    return bLiveOriginalRun();
  };

  bNext=async function(){
    const next=becomeState.scenarioIndex+1;
    if(next>=becomeState.queue.length){bRenderTransferForm();return;}
    if(!becomeState.queue[next]){
      bLiveLoading(next,`The next world is being generated online now${becomeLiveLastModel?` through ${becomeLiveLastModel}`:''}, using recent worlds as explicit novelty exclusions.`);
      try{const scenario=await bLivePrefetch(next);if(!scenario)throw new Error('The live generator returned no scenario.');}
      catch(error){bLiveError(error);return;}
    }
    becomeState.scenarioIndex=next;bRenderBrief();
  };

  window.addEventListener('dreamunity:game-closed',()=>{
    becomeLiveSessionNonce='';becomeLivePending.clear();
  });
