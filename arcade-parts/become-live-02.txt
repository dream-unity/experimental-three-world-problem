  function bLiveValidateScenario(raw){
    if(!raw || typeof raw !== 'object' || Array.isArray(raw)) throw new Error('The generator returned an invalid scenario object.');
    const scenario = {...raw};
    for(const key of bLiveRequired){
      if(typeof scenario[key] !== 'string' || scenario[key].trim().length < 2) throw new Error(`Generated scenario is missing ${key}.`);
      scenario[key] = scenario[key].trim();
    }
    scenario.visual = BECOME_LIVE_VISUALS.has(scenario.visual.toLowerCase()) ? scenario.visual.toLowerCase() : 'future';
    if(!scenario.dimensions || typeof scenario.dimensions !== 'object' || Array.isArray(scenario.dimensions)) throw new Error('Generated scenario is missing relational dimensions.');
    scenario.dimensions = {...scenario.dimensions};
    for(const key of BECOME_DIMENSION_KEYS){
      if(typeof scenario.dimensions[key] !== 'string' || scenario.dimensions[key].trim().length < 2) throw new Error(`Generated scenario is missing dimension ${key}.`);
      scenario.dimensions[key] = scenario.dimensions[key].trim();
    }
    return scenario;
  }

  const bSynthPick = values => values[(Math.random()*values.length)|0];
  function bSynthTooSimilar(candidate,history){
    for(const prior of history){
      if(String(candidate.title).toLowerCase()===String(prior.title||'').toLowerCase())return true;
      let same=0;
      for(const key of BECOME_DIMENSION_KEYS){
        const a=String(candidate.dimensions?.[key]||'').toLowerCase();
        const b=String(prior.dimensions?.[key]||'').toLowerCase();
        if(a&&b&&(a===b||a.includes(b)||b.includes(a)))same++;
      }
      if(same>=3)return true;
    }
    return false;
  }
