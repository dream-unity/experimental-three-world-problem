  function s12Words(text){return String(text||'').trim().split(/\s+/).filter(Boolean);}
  function s12Count(text){return s12Words(text).length;}
  function s12Limit(text,max){
    const words=s12Words(text);
    if(words.length<=max)return String(text||'').trim();
    return words.slice(0,max).join(' ').replace(/[,:;.!?]+$/,'')+'…';
  }
  function s12Before(text,marker){const i=String(text||'').indexOf(marker);return(i>=0?String(text).slice(0,i):String(text||'')).trim();}
  function s12Sentences(text,maxSentences=2){
    const parts=String(text||'').trim().match(/[^.!?]+[.!?]+|[^.!?]+$/g)||[];
    return parts.slice(0,maxSentences).join(' ').trim();
  }
  function s12CompactFor(profile){
    return S12_PROFILE_COMPACT[profile.id]||{
      tension:s12Limit(profile.encounter,9),
      self:s12Limit(String(profile.sig.selfBlindspot||'self-bias').replace(/-/g,' '),5),
      other:'uncertain · protecting something · differently informed',
      move:s12Limit(profile.move,9)
    };
  }
  function s12BuildBrief(scenario,profile,compact){
    const originalWorld=s12Sentences(s12Before(scenario.premise,'The relevant social perspective'),2);
    const socialTail=`${compact.tension} Watch ${compact.self}. They may be ${compact.other}. ${compact.move}`;
    const socialWords=s12Count(socialTail);
    const worldBudget=Math.max(14,S12_BRIEF_WORD_MAX-socialWords);
    return s12Limit(`${s12Limit(originalWorld,worldBudget)} ${socialTail}`,S12_BRIEF_WORD_MAX);
  }
