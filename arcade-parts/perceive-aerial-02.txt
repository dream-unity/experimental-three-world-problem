        committedTurnSign: initialTurn,
        nextTurnAt: config.turnChanges ? rand(config.turnHoldMax, config.turnHoldMin) : Number.POSITIVE_INFINITY,
        target: null,
        previousTarget: null,
        candidate: null,
        candidateTimer: 0,
        targetStableFor: 0,
        roleSwitchAt: 0,
        reacquired: false,
        reticleHold: 0,
        hits: 0,
        solved: false,
        failed: false,
        fireTimer: rand(config.fireInterval * 1.18, config.fireInterval * 0.72),
        startedAt: performance.now(),
        outsideAxis: [-initialTurn, 0],
        insideAxis: [initialTurn, 0],
        outsideVisual: [-initialTurn, 0],
        insideVisual: [initialTurn, 0],
        roleMargin: 1,
        turnTransition: false,
      };
      buildRelationalNetwork(rule);
      resolveTimer = 0;
      wrongShotCooldown = 0;
      updateFormation(E, 0);
      E.toast(`INFER THE HIDDEN RELATION · WATCH ${config.exampleCount} EXAMPLES`, 1.35);
      E.setMetric('SHIELD', `${Math.ceil(player.shield)} · ×${streak}`);
    }

    function updateLeaderTurn(dt) {
      const config = encounter.config;
      if (encounter.age >= encounter.nextTurnAt) {
        encounter.turnTarget *= -1;
        encounter.nextTurnAt = encounter.age + rand(config.turnHoldMax, config.turnHoldMin);
      }
      const blend = 1 - Math.exp(-config.turnResponse * dt);
      encounter.turnValue = lerp(encounter.turnValue, encounter.turnTarget, blend);
      encounter.turnTransition = Math.abs(encounter.turnValue) < 0.36;
      if (!encounter.turnTransition) {
        const nextCommitted = encounter.turnValue > 0 ? 1 : -1;
        if (nextCommitted !== encounter.committedTurnSign) {
          encounter.committedTurnSign = nextCommitted;
          encounter.candidate = null;
          encounter.candidateTimer = 0;
        }
      }

      encounter.coursePhase += encounter.turnValue * config.turnRate * dt;
      encounter.laneVelocity += encounter.turnValue * dt * 0.082;
      encounter.laneVelocity += -encounter.lane * dt * 0.16;
      encounter.laneVelocity *= Math.exp(-dt * 0.76);
      encounter.lane = clamp(encounter.lane + encounter.laneVelocity * dt, -0.34, 0.34);

      leader.heading = clamp(encounter.laneVelocity * 2.15 + encounter.turnValue * 0.2, -0.72, 0.72);
      leader.bank = lerp(leader.bank, encounter.turnValue * 0.9, 1 - Math.exp(-dt * 4.8));
      leader.pulse = 0.52 + 0.48 * Math.sin(encounter.age * 2.3) ** 2;

      const rightAxis = normalize2([Math.cos(leader.heading), Math.sin(leader.heading)]);
      encounter.insideAxis = [rightAxis[0] * encounter.committedTurnSign, rightAxis[1] * encounter.committedTurnSign];
      encounter.outsideAxis = [-encounter.insideAxis[0], -encounter.insideAxis[1]];
      encounter.insideVisual = normalize2([encounter.insideAxis[0], encounter.insideAxis[1] * 0.84]);
      encounter.outsideVisual = [-encounter.insideVisual[0], -encounter.insideVisual[1]];
    }

    function updateFormation(E, dt) {
      if (!encounter) return;
      encounter.age += dt;
      if (!encounter.solved && !encounter.failed) {
        encounter.depth = Math.min(1.08, encounter.depth + dt * encounter.config.approachSpeed);
      }
      updateLeaderTurn(dt);

      const depth = encounter.depth;
      const scale = E.u * (0.34 + depth * 0.42);
      const centerX = E.w * 0.5 + encounter.lane * E.u * 0.72;
      const centerY = E.top + E.h * (0.15 + depth * 0.42) + Math.sin(encounter.coursePhase * 0.52) * E.u * 0.018;
      leader.sx = centerX;
      leader.sy = centerY;
      leader.size = Math.max(13, E.u * (0.018 + depth * 0.019));
      leader.trail.push([leader.sx, leader.sy]);
      if (leader.trail.length > 28) leader.trail.shift();

      const config = encounter.config;
      const active = activeEscorts();
      for (const escort of active) {
        const orbit = escort.slotAngle
          + encounter.age * config.orbitSpeed * escort.orbitDirection
          + Math.sin(encounter.age * config.weaveSpeed * escort.weaveRate + escort.weavePhase) * config.weaveAmplitude;
        const radius = escort.baseRadius * (
          1 + Math.sin(encounter.age * (config.weaveSpeed * 0.73) + escort.radialPhase) * config.radialAmplitude
        );
        const bodyX = Math.cos(orbit) * radius;
        const bodyY = Math.sin(orbit) * radius * 0.66
          + Math.sin(encounter.age * config.weaveSpeed * 0.52 + escort.weavePhase) * 0.028;
        const world = rotate2(bodyX, bodyY, leader.heading * 0.78);
        escort.rx = world[0];
        escort.ry = world[1];
        escort.sx = leader.sx + world[0] * scale;
        escort.sy = leader.sy + world[1] * scale * 0.84;
