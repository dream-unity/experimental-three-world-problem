    function createEpisode(E) {
      const law = lawForLevel(level);
      const config = levelConfig(level, law);
      const demoKind = nextDemoKind(law);
      const positive = episodePolarity(config, demoKind);
      const { jets, roles } = buildRoles(law);
      scene = {
        law,
        config,
        jets,
        roles,
        flowDirection: choose([-1, 1]),
        age: 0,
        depth: 0.07,
        orientation: rand(Math.PI, -Math.PI),
        rotationDrift: rand(0.22, -0.22),
        scaleBias: rand(1.08, 0.9),
        lane: rand(0.08, -0.08),
        positive,
        negativeVariant: positive ? null : chooseNegativeVariant(law),
        demoKind,
        demoShotDone: false,
        demoResolveAt: 0,
        resolved: false,
        handled: false,
        activeEver: false,
        activeStartedAt: null,
        activeEndedAt: null,
        previousTargetId: null,
        currentTargetId: null,
        targetChangedAt: 0,
        evaluation: { active: false, targetId: null, strength: 0, law },
        fireTimer: rand(config.enemyFireInterval * 1.2, config.enemyFireInterval * 0.72),
        hits: 0,
        startedAt: performance.now(),
      };
      enemyShots = [];
      tracers = [];
      wrongShotCooldown = 0;
      resolveTimer = 0;
      updateSceneGeometry(E, 0.0001);
      if (!demoKind) E.toast('WATCH THE FIELD', 0.64);
      E.setMetric('SHIELD', `${Math.ceil(player.shield)} · ×${streak}`);
    }

    function jetById(id) {
      return scene?.jets.find((jet) => jet.id === id) || null;
    }

    function localForBetween(t) {
      const anchorSpread = 0.255 + Math.sin(scene.age * 0.82) * 0.012;
      const anchorA = [-anchorSpread, Math.sin(scene.age * 0.55) * 0.012];
      const anchorB = [anchorSpread, -Math.sin(scene.age * 0.55) * 0.012];
      const travel = lerp(-0.34, 0.34, ease(t));
      let relayX = Math.sin(scene.age * 1.35) * 0.024;
      let relayY = travel;
      if (!scene.positive) {
        if (scene.negativeVariant === 'line-extension') relayX = 0.39 + Math.sin(scene.age * 1.1) * 0.018;
        else if (scene.negativeVariant === 'near-offset') relayY = lerp(-0.34, -0.09, ease(t)) + Math.sin(t * Math.PI) * 0.028;
        else {
          relayX = lerp(-0.22, 0.22, ease(t));
          relayY = 0.052 + Math.sin(scene.age * 1.2) * 0.012;
        }
      }
      const decoyX = scene.positive ? -0.4 + Math.sin(scene.age * 0.9) * 0.025 : -0.31;
      const decoyY = scene.positive ? lerp(0.25, -0.23, ease(t)) : lerp(0.28, -0.25, ease(t));
      return {
        [scene.roles.anchorAId]: anchorA,
        [scene.roles.anchorBId]: anchorB,
        [scene.roles.relayId]: [relayX, relayY],
        [scene.roles.decoyId]: [decoyX, decoyY],
      };
    }

    function localForEnclosure(t) {
      const breathing = 1 + Math.sin(scene.age * 0.72) * 0.035;
      const anchorA = [-0.265 * breathing, 0.18 * breathing];
      const anchorB = [0.265 * breathing, 0.18 * breathing];
      let anchorC = [0, -0.245 * breathing];
      const travel = lerp(-0.39, 0.27, ease(t));
      let relayX = Math.sin(scene.age * 1.2) * 0.025;
      let relayY = travel;
      if (!scene.positive) {
        if (scene.negativeVariant === 'outside-triangle') relayX = 0.31 + Math.sin(scene.age * 0.8) * 0.018;
        else if (scene.negativeVariant === 'boundary-touch') relayX = 0.22;
        else anchorC = [0, 0.15 + Math.sin(scene.age) * 0.012];
      }
      const decoy = [-0.35 + Math.cos(scene.age * 0.66) * 0.035, -0.04 + Math.sin(scene.age * 0.95) * 0.055];
      return {
        [scene.roles.anchorAId]: anchorA,
        [scene.roles.anchorBId]: anchorB,
        [scene.roles.anchorCId]: anchorC,
        [scene.roles.relayId]: [relayX, relayY],
        [scene.roles.decoyId]: decoy,
      };
    }

    function updateSceneGeometry(E, dt) {
      if (!scene) return;
      scene.age += dt;
      if (!scene.resolved) scene.depth = Math.min(1.08, scene.depth + dt * scene.config.approachSpeed);
      const t = clamp(scene.age / scene.config.duration, 0, 1);
      const local = scene.law === 'enclosure' ? localForEnclosure(t) : localForBetween(t);
      const angle = scene.orientation + scene.age * scene.rotationDrift;
      const scale = scene.scaleBias * (1 + Math.sin(scene.age * 0.46) * 0.03);
      for (const jet of scene.jets) {
        const point = local[jet.id] || [0, 0];
        const transformed = rotate2(point[0] * scale, point[1] * scale, angle);
        jet.previousRx = jet.rx;
        jet.previousRy = jet.ry;
