  function rtSeed(index, salt = 0) {
    const value = Math.sin(index * 127.1 + salt * 311.7) * 43758.5453123;

  function rtMixColor(a, b, amount) {
    const u = clamp(amount, 0, 1);
    return [
      Math.round(lerp(a[0], b[0], u)),
      Math.round(lerp(a[1], b[1], u)),
      Math.round(lerp(a[2], b[2], u))
    ];
  }

  function rtColorAt(value) {
    const s = ((value % 1) + 1) % 1;
    if (s < 0.25) return rtMixColor(WORLD.machine.rgb, WORLD.maker.rgb, ease(s / 0.25));
    if (s < 0.585) return rtMixColor(WORLD.maker.rgb, WORLD.reality.rgb, ease((s - 0.25) / 0.335));
    if (s < 0.88) return rtMixColor(WORLD.reality.rgb, WORLD.machine.rgb, ease((s - 0.585) / 0.295));
    return WORLD.machine.rgb;
  }

  function rtCenter() {
    return {
      x: width > 1060 ? width * 0.53 : width * 0.5,

  function rtProject(point, scale = 1, cameraZ = 12.6, center = rtCenter()) {
    const depth = cameraZ - point.z;
    const perspective = scale * cameraZ / Math.max(3.1, depth);
    const spread = width > 1060 ? 0.101 : width > 760 ? 0.092 : 0.086;
    return {
      x: center.x + point.x * unit * spread * perspective,
      y: center.y - point.y * unit * spread * perspective,
      f: perspective,
      z: point.z,
      depth
    };
  }

  function rtCatmull(value) {
    const wrapped = ((value % 1) + 1) % 1;
    const scaled = wrapped * RT_PATH.length;
    const index = Math.floor(scaled);
    const u = scaled - index;
    const p0 = RT_PATH[(index - 1 + RT_PATH.length) % RT_PATH.length];
    const p1 = RT_PATH[index % RT_PATH.length];
    const p2 = RT_PATH[(index + 1) % RT_PATH.length];
    const p3 = RT_PATH[(index + 2) % RT_PATH.length];
    const u2 = u * u;
    const u3 = u2 * u;
    const coordinate = (key) => 0.5 * (
      2 * p1[key] +
      (-p0[key] + p2[key]) * u +
      (2 * p0[key] - 5 * p1[key] + 4 * p2[key] - p3[key]) * u2 +
      (-p0[key] + 3 * p1[key] - 3 * p2[key] + p3[key]) * u3
    );
    return { x: coordinate('x'), y: coordinate('y'), z: coordinate('z') };
  }

  function rtNormalize(vector) {
    const length = Math.max(0.0001, Math.hypot(vector.x, vector.y, vector.z));
    return { x: vector.x / length, y: vector.y / length, z: vector.z / length };
  }

  function rtCross(a, b) {
    return {
      x: a.y * b.z - a.z * b.y,
      y: a.z * b.x - a.x * b.z,
      z: a.x * b.y - a.y * b.x
    };
  }

  function rtFrameAt(value) {
    const before = rtCatmull(value - 0.0015);
    const after = rtCatmull(value + 0.0015);
    const tangent = rtNormalize({ x: after.x - before.x, y: after.y - before.y, z: after.z - before.z });
    let side = rtCross(tangent, { x: 0, y: 1, z: 0 });
    if (Math.hypot(side.x, side.y, side.z) < 0.08) side = rtCross(tangent, { x: 1, y: 0, z: 0 });
    side = rtNormalize(side);
    const up = rtNormalize(rtCross(side, tangent));
    return { tangent, side, up };
  }

  function rtOffsetPoint(value, sideAmount = 0, upAmount = 0) {
    const point = rtCatmull(value);
    const frameAt = rtFrameAt(value);
    return {
      x: point.x + frameAt.side.x * sideAmount + frameAt.up.x * upAmount,
      y: point.y + frameAt.side.y * sideAmount + frameAt.up.y * upAmount,
      z: point.z + frameAt.side.z * sideAmount + frameAt.up.z * upAmount
    };
  }

  function rtRotateProject(point, yaw = overviewYaw, pitch = overviewPitch, roll = overviewRoll, scale = overviewZoom) {
    const rotated = rotate3(point, yaw, pitch, roll);
    const projected = rtProject(rotated, scale, 12.6);
    projected.world = rotated;
    return projected;
  }

  function rtLine(a, b, color, alpha, lineWidth = 1) {
    if (alpha <= 0.002) return;
    ctx.strokeStyle = rgba(color, alpha);
    ctx.lineWidth = lineWidth;
    ctx.beginPath();
    ctx.moveTo(a.x, a.y);
    ctx.lineTo(b.x, b.y);
    ctx.stroke();
  }

  function rtDrawUnity(t, alpha) {
    const bodyVoiceState = document.body?.dataset.voiceState?.toLowerCase();
    const appVoiceState = app?.dataset.voiceState?.toLowerCase();
    const observedStates = [bodyVoiceState, appVoiceState].filter(Boolean);
    const voiceState = observedStates.find((state) => RT_ACTIVE_VOICE_STATES.has(state))
      || observedStates[0]
      || 'idle';
    const voiceDt = rtVoiceLastTime
      ? Math.min(0.10, Math.max(0.003, t - rtVoiceLastTime))
      : 1 / 60;
    rtVoiceLastTime = t;
    RT_VOICE_KEYS.forEach((name) => {
      const target = RT_VOICE_ALIASES[name].includes(voiceState) ? 1 : 0;
      const response = reduced ? 1 : 1 - Math.exp(-voiceDt * (target ? 9.5 : 5.5));
      rtVoiceMix[name] = lerp(rtVoiceMix[name], target, response);
    });

    const listening = rtVoiceMix.listening;
    const thinking = rtVoiceMix.thinking;
    const speaking = rtVoiceMix.speaking;
    const activeVoice = Math.max(listening, thinking, speaking);
    const motionTime = reduced ? 0 : t;
    const idleBreath = reduced ? 1 : 1 + Math.sin(t * 0.72) * 0.018 * (1 - activeVoice * 0.55);
    const thoughtBeat = reduced ? 0.45 : 0.5 + Math.sin(t * 2.15) * 0.5;
    const speechBeat = reduced
      ? 0.62
      : 0.55 + Math.sin(t * 5.1) * 0.22 + Math.sin(t * 8.7 + 0.8) * 0.14;
    const compression = 1 - thinking * (0.105 + thoughtBeat * 0.045);
    const aperture = 1 + listening * (0.10 + Math.sin(motionTime * 2.0) * 0.025);
    const resonance = 1 + speaking * speechBeat * 0.075;
    const loops = [
      { color: RT_GOLD, phase: 0, width: 1.25 },
      { color: RT_INK, phase: 1.7, width: 0.65 },
      { color: RT_PALE, phase: 3.1, width: 0.85 }
    ];
    loops.forEach((loop, loopIndex) => {
      let previous = null;
      for (let index = 0; index <= 86; index++) {
        const angle = index / 86 * TAU;
        const model = {
          x: Math.cos(angle) * (1.08 + loopIndex * 0.16) * idleBreath * compression * resonance,
          y: Math.sin(angle) * (0.34 + loopIndex * 0.05) * idleBreath * compression * aperture,
          z: Math.sin(angle * 2 + loop.phase + t * (reduced ? 0 : 0.08))
            * (0.61 + loopIndex * 0.10)
            * idleBreath
            * compression
            * (1 + listening * 0.06 + speaking * speechBeat * 0.10)
        };
        const projected = rtRotateProject(model);
        if (previous) {
          const lineAlpha = (loopIndex === 0 ? 0.72 : 0.40) * (1 + activeVoice * 0.12);
          rtLine(previous, projected, loop.color, alpha * lineAlpha, loop.width + speaking * 0.18);
        }
        previous = projected;
      }
    });
    const core = rtRotateProject({ x: 0, y: 0, z: 0 });
    const haloBase = Math.max(9, unit * 0.014);
    ctx.save();
    ctx.globalCompositeOperation = 'source-over';
    ctx.lineCap = 'round';

    if (listening > 0.01) {
      const listenWave = reduced ? 0.35 : (Math.sin(t * 2.25) + 1) * 0.5;
      for (let ring = 0; ring < 2; ring++) {
        const radius = haloBase * (1.28 + ring * 0.64 + listenWave * 0.13);
        const gap = 0.31 + ring * 0.05;
        ctx.strokeStyle = rgba(ring ? RT_GOLD : WORLD.machine.rgb, alpha * listening * (ring ? 0.25 : 0.42));
        ctx.lineWidth = ring ? 0.8 : 1.15;
        ctx.beginPath();
        ctx.ellipse(core.x, core.y, radius * 1.18, radius * 0.58, -0.08, gap, Math.PI - gap);
        ctx.stroke();
        ctx.beginPath();
        ctx.ellipse(core.x, core.y, radius * 1.18, radius * 0.58, -0.08, Math.PI + gap, TAU - gap);
        ctx.stroke();
      }
      const receivePhase = reduced ? 0.42 : (t * 0.68) % 1;
      const receiveRadius = haloBase * (1.25 + receivePhase * 1.42);
      ctx.strokeStyle = rgba(WORLD.machine.rgb, alpha * listening * (1 - receivePhase) * 0.22);
      ctx.lineWidth = 0.7;
      ctx.beginPath();
      ctx.ellipse(core.x, core.y, receiveRadius * 1.16, receiveRadius * 0.58, -0.08, 0, TAU);
      ctx.stroke();
    }

    if (thinking > 0.01) {
      const inward = reduced ? 0.56 : (t * 0.82) % 1;
      const thoughtRadius = haloBase * (2.75 - inward * 1.38);
      ctx.strokeStyle = rgba(WORLD.reality.rgb, alpha * thinking * (0.13 + inward * 0.20));
      ctx.lineWidth = 0.8;
      ctx.beginPath();
      ctx.ellipse(core.x, core.y, thoughtRadius * 1.12, thoughtRadius * 0.56, 0.12, 0, TAU);
      ctx.stroke();
      for (let orbit = 0; orbit < 3; orbit++) {
        const orbitRadius = haloBase * (0.86 + orbit * 0.34 - thoughtBeat * 0.08);
        const start = motionTime * (0.74 + orbit * 0.14) + orbit * TAU / 3;
        ctx.strokeStyle = rgba(orbit === 1 ? RT_GOLD : RT_PALE, alpha * thinking * (0.30 - orbit * 0.045));
        ctx.lineWidth = orbit === 1 ? 1.05 : 0.7;
        ctx.beginPath();
        ctx.ellipse(core.x, core.y, orbitRadius * 1.22, orbitRadius * 0.54, orbit * 0.18, start, start + 0.70);
        ctx.stroke();
      }
    }

    if (speaking > 0.01) {
      // Projected sound shells carry Unity's speech back through the three-world field.
      for (let shell = 0; shell < 4; shell++) {
        const travel = reduced ? (shell + 1) / 5 : (t * 1.18 + shell / 4) % 1;
        const radius = haloBase * (1.18 + travel * 2.65);
        const shellAlpha = alpha * speaking * (1 - travel) * (0.30 + speechBeat * 0.12);
        ctx.strokeStyle = rgba(shell % 2 ? WORLD.machine.rgb : RT_GOLD, shellAlpha);
        ctx.lineWidth = 0.72 + (1 - travel) * 0.48;
        ctx.beginPath();
        ctx.ellipse(core.x, core.y, radius * 1.12, radius * 0.54, 0, -0.63, 0.63);
        ctx.stroke();
        ctx.beginPath();
        ctx.ellipse(core.x, core.y, radius * 1.12, radius * 0.54, 0, Math.PI - 0.63, Math.PI + 0.63);
        ctx.stroke();
      }
    }

    const coreRadius = 5.2 + listening * 0.55 + thinking * thoughtBeat * 0.35 + speaking * speechBeat * 0.72;
    ctx.fillStyle = '#fff';ctx.beginPath();ctx.arc(core.x,core.y,coreRadius,0,TAU);ctx.fill();
    ctx.strokeStyle = rgba(listening > thinking && listening > speaking ? WORLD.machine.rgb : RT_GOLD, alpha * (0.88 + activeVoice * 0.10));
    ctx.lineWidth = 1 + activeVoice * 0.25;ctx.stroke();
    ctx.fillStyle = rgba(RT_INK, alpha * 0.74);ctx.beginPath();ctx.arc(core.x,core.y,1.25 + speaking * speechBeat * 0.48,0,TAU);ctx.fill();
    ctx.restore();
    return core;
  }

  updateLabels = function rememberedTomorrowLabels() {
    if (++labelTick % 2) return;
    const mix = ease(viewMix);
    const portrait = width < 760;
    worldKeys.forEach((key) => {
      const element = labels[key];
      const point = worldScreen[key];
      if (!element) return;
      let left = point.x;
      let top = point.y - point.r * 1.25;
      if (!portrait) {
        const outward = point.r + 74;
        if (key === 'machine') left = point.x - outward;
        else if (key === 'maker') left = point.x + outward;
        else { left = point.x + outward * .92; top = point.y + point.r * .48; }
      } else {
        if (key === 'machine') left -= point.r * .55;
        if (key === 'maker') left += point.r * .65;
        if (key === 'reality') { left += point.r * .8; top += point.r * .65; }
      }
      element.style.left = `${clamp(left, portrait ? 55 : 82, width - (portrait ? 55 : 82))}px`;
      element.style.top = `${clamp(top, 72, height - 62)}px`;
      element.style.opacity = String(clamp(1 - mix * 2.1, 0, 1));
      element.style.pointerEvents = mix < 0.25 ? 'auto' : 'none';
    });
    if (unityLabel) {
      const core = rtRotateProject({x:0,y:0,z:0});
      unityLabel.style.left = `${core.x}px`;
      unityLabel.style.top = `${core.y + unit * .055}px`;
      unityLabel.style.opacity = String(clamp(1 - mix * 2.2, 0, 1));
    }
    if (activeWorld && mix > 0.22) {
      subLabels.forEach((element, index) => {
        const point = subScreen[index];
        element.style.left = `${point.x}px`;
        element.style.top = `${point.y - point.r * 1.5}px`;
        element.style.opacity = String(ease((mix - .22) / .5));
        element.style.pointerEvents = mix > .65 ? 'auto' : 'none';
        element.classList.toggle('active', index === activeSub);
      });
    } else {
      subLabels.forEach((element) => { element.style.opacity = '0'; element.style.pointerEvents = 'none'; });
    }
    app?.classList.toggle('detail', Boolean(activeWorld && mix > .05));
  };
