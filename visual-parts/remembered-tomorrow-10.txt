  function rtSectionAt(currentTime) {
    const time = ((currentTime % RT_DURATION) + RT_DURATION) % RT_DURATION;
    const section = RT_SCORE_SECTIONS.find((entry) => time >= entry[1] && time < entry[2]) || RT_SCORE_SECTIONS[0];
    return {
      phase: section[0],
      progress: clamp((time - section[1]) / Math.max(0.001, section[2] - section[1]), 0, 1),
      presence: section[3],
      time
    };
  }

  function rtEnvelope(time, start, peak, end) {
    if (time <= start || time >= end) return 0;
    if (time < peak) return ease((time - start) / Math.max(0.001, peak - start));
    return 1 - ease((time - peak) / Math.max(0.001, end - peak));
  }

  function rtReadScore(t) {
    let sample = null;
    try { sample = window.__dreamUnityScore?.sample?.() || null; } catch {}
    const songTime = Number.isFinite(sample?.currentTime) ? sample.currentTime : (t % RT_DURATION);
    const section = rtSectionAt(songTime);
    const sampledPhase = sample?.phase || sample?.section || section.phase;
    const phase = sampledPhase === 'invocation' ? 'summon'
      : sampledPhase === 'opening' ? 'open'
      : sampledPhase === 'first-crown' ? 'war'
      : sampledPhase === 'embodiment' || sampledPhase === 'rebuild' ? 'ascent'
      : sampledPhase === 'true-war' ? 'war'
      : sampledPhase === 'abyss' ? 'bone'
      : sampledPhase === 'bone-reprise' ? 'bone'
      : sampledPhase === 'crown' ? 'awaken'
      : sampledPhase === 'collapse' ? 'return'
      : sampledPhase;
    const progress = Number.isFinite(sample?.sectionProgress) ? sample.sectionProgress
      : Number.isFinite(sample?.progress) ? sample.progress : section.progress;
    const fallbackPulse = reduced ? 0 : Math.pow(Math.max(0, Math.sin(songTime * Math.PI * 2 / (60 / 86.7))), 8);
    const bone = clamp(Number.isFinite(sample?.bone) ? sample.bone : phase === 'bone' ? 1 : 0, 0, 1);
    const rupture = clamp(Number.isFinite(sample?.rupture) ? sample.rupture
      : phase === 'awaken' ? ease(clamp((songTime - 276.80) / 2.45, 0, 1)) : phase === 'return' ? 1 : 0, 0, 1);
    const bloom = clamp(Number.isFinite(sample?.bloom) ? sample.bloom : Math.max(
      rtEnvelope(songTime, 276.80, 279.25, 300.00),
      rtEnvelope(songTime, 321.90, 323.10, 342.75)
    ), 0, 1);
    const release = clamp(Number.isFinite(sample?.release) ? sample.release : phase === 'return' ? ease(progress) : 0, 0, 1);
    const pressure = Number.isFinite(sample?.pressure) ? sample.pressure : phase === 'ascent' ? 0.45 + progress * 0.38
      : phase === 'war' ? 0.78 + progress * 0.18
      : phase === 'descent' ? 0.72 * (1 - progress)
      : phase === 'bone' ? 0.08 + progress * 0.16
      : phase === 'awaken' ? 0.84 + 0.16 * Math.sin(progress * Math.PI)
      : 0.10 + section.presence * 0.24;
    const basePresence = phase === 'return' ? lerp(0.86, 0.28, release) : section.presence;
    const bonePresence = phase === 'bone' ? lerp(0.16, 0.09, ease(progress)) : 0.13;
    rtScoreState = {
      currentTime: section.time,
      phase,
      progress,
      low: clamp(Number.isFinite(sample?.low) ? sample.low : section.presence * 0.52, 0, 1),
      mid: clamp(Number.isFinite(sample?.mid) ? sample.mid : section.presence * 0.46, 0, 1),
      high: clamp(Number.isFinite(sample?.high) ? sample.high : section.presence * 0.34, 0, 1),
      energy: clamp(Number.isFinite(sample?.energy) ? sample.energy : section.presence, 0, 1),
      flux: clamp(Number.isFinite(sample?.flux) ? sample.flux : 0, 0, 1),
      beat: clamp(Number.isFinite(sample?.beat) ? sample.beat : fallbackPulse * section.presence, 0, 1),
      presence: lerp(basePresence, bonePresence, bone),
      pressure,
      bone,
      rupture,
      bloom,
      release,
      command: ease(clamp((songTime - 276.80) / 3.4, 0, 1))
    };
    return rtScoreState;
  }

  function rtSeed(index, salt = 0) {
    const value = Math.sin(index * 127.1 + salt * 311.7) * 43758.5453123;
    return value - Math.floor(value);
  }

  function rtMixColor(a, b, amount) {
    const u = clamp(amount, 0, 1);
    return [
      Math.round(lerp(a[0], b[0], u)),
      Math.round(lerp(a[1], b[1], u)),
      Math.round(lerp(a[2], b[2], u))
    ];
  }

  function rtColorAt(value) {

  function rtCenter() {
    return {
      x: width > 1060 ? width * 0.60 : width > 760 ? width * 0.54 : width * 0.5,
      y: width > 760 ? height * 0.51 : height * 0.56
    };
  }

  function rtProject(point, scale = 1, cameraZ = 12.6, center = rtCenter()) {
    const depth = cameraZ - point.z;
    const perspective = scale * cameraZ / Math.max(3.1, depth);
    const spread = width > 1060 ? 0.139 : width > 760 ? 0.116 : 0.145;
    return {
      x: center.x + point.x * unit * spread * perspective,
      y: center.y - point.y * unit * spread * perspective,
      f: perspective,
      z: point.z,
      depth
    };
  }

  function rtCatmull(value) {
    const wrapped = ((value % 1) + 1) % 1;
    const scaled = wrapped * RT_PATH.length;
    const index = Math.floor(scaled);
    const u = scaled - index;
    const p0 = RT_PATH[(index - 1 + RT_PATH.length) % RT_PATH.length];
    const p1 = RT_PATH[index % RT_PATH.length];
    const p2 = RT_PATH[(index + 1) % RT_PATH.length];
    const p3 = RT_PATH[(index + 2) % RT_PATH.length];
    const u2 = u * u;
    const u3 = u2 * u;
    const coordinate = (key) => 0.5 * (
      2 * p1[key] +
      (-p0[key] + p2[key]) * u +
      (2 * p0[key] - 5 * p1[key] + 4 * p2[key] - p3[key]) * u2 +
      (-p0[key] + 3 * p1[key] - 3 * p2[key] + p3[key]) * u3
    );
    return { x: coordinate('x'), y: coordinate('y'), z: coordinate('z') };
  }

  function rtNormalize(vector) {
    const length = Math.max(0.0001, Math.hypot(vector.x, vector.y, vector.z));
    return { x: vector.x / length, y: vector.y / length, z: vector.z / length };
  }

  function rtCross(a, b) {
    return {
      x: a.y * b.z - a.z * b.y,
      y: a.z * b.x - a.x * b.z,
      z: a.x * b.y - a.y * b.x
    };
  }

  function rtFrameAt(value) {
    const before = rtCatmull(value - 0.0015);
    const after = rtCatmull(value + 0.0015);
    const tangent = rtNormalize({ x: after.x - before.x, y: after.y - before.y, z: after.z - before.z });
    let side = rtCross(tangent, { x: 0, y: 1, z: 0 });
    if (Math.hypot(side.x, side.y, side.z) < 0.08) side = rtCross(tangent, { x: 1, y: 0, z: 0 });
    side = rtNormalize(side);
    const up = rtNormalize(rtCross(side, tangent));
    return { tangent, side, up };
  }

  function rtOffsetPoint(value, sideAmount = 0, upAmount = 0) {
    const point = rtCatmull(value);
    const frameAt = rtFrameAt(value);
    return {
      x: point.x + frameAt.side.x * sideAmount + frameAt.up.x * upAmount,
      y: point.y + frameAt.side.y * sideAmount + frameAt.up.y * upAmount,
      z: point.z + frameAt.side.z * sideAmount + frameAt.up.z * upAmount
    };
  }

  function rtRotateProject(point, yaw = rtLivingYaw, pitch = rtLivingPitch, roll = rtLivingRoll, scale = overviewZoom) {
    const rotated = rotate3(point, yaw, pitch, roll);
    const projected = rtProject(rotated, scale, 12.6);
    projected.world = rotated;
    return projected;
  }

  function rtLine(a, b, color, alpha, lineWidth = 1) {
    if (alpha <= 0.002) return;
    ctx.strokeStyle = rgba(color, alpha);
    ctx.lineWidth = lineWidth;
    ctx.beginPath();
    ctx.moveTo(a.x, a.y);
    ctx.lineTo(b.x, b.y);
    ctx.stroke();
  }

  function rtShellPoint(latitude, longitude) {
    const vertical = Math.sin(latitude);
    const radial = Math.cos(latitude);
    const asymmetric = 1 + vertical * 0.12;
    return {
      x: Math.cos(longitude) * radial * 1.86 * asymmetric + Math.sin(latitude * 2) * 0.20,
      y: vertical * 3.28,
      z: Math.sin(longitude) * radial * 1.48 * asymmetric
    };
  }

  function rtOwnedFacetTarget(row, column, rows, columns) {
    const s = 0.505 + row / rows * 0.215;
    const lateral = (column / columns - 0.5) * 0.74;
    const lift = (row / rows - 0.5) * 0.18 + (column % 2 ? 0.035 : -0.035);
    return rtOffsetPoint(s, lateral, lift);
  }

  function rtDrawSovereignShell(t, alpha) {
    const score = rtScoreState;
    const rows = lowCPU || coarse ? 5 : 7;
    const columns = lowCPU || coarse ? 12 : 18;
    const facets = [];
    const structure = score.rupture * (1 - score.release * 0.70);
    const yaw = lerp(rtGhostYaw, rtLivingYaw, score.command);
    const pitch = lerp(rtGhostPitch, rtLivingPitch, score.command);
    const roll = lerp(rtGhostRoll, rtLivingRoll, score.command);

    for (let row = 0; row < rows; row++) {
      const latitudeA = -Math.PI * 0.5 + row / rows * Math.PI;
      const latitudeB = -Math.PI * 0.5 + (row + 1) / rows * Math.PI;
      for (let column = 0; column < columns; column++) {
        const longitudeA = column / columns * TAU;
        const longitudeB = (column + 1) / columns * TAU;
        const source = [
          rtShellPoint(latitudeA, longitudeA),
          rtShellPoint(latitudeB, longitudeA),
          rtShellPoint(latitudeB, longitudeB),
          rtShellPoint(latitudeA, longitudeB)
        ];
        const destination = [
          rtOwnedFacetTarget(row, column, rows, columns),
          rtOwnedFacetTarget(row + 1, column, rows, columns),
          rtOwnedFacetTarget(row + 1, column + 1, rows, columns),
          rtOwnedFacetTarget(row, column + 1, rows, columns)
        ];
        const stagger = clamp((structure - rtSeed(row * columns + column, 71) * 0.28) / 0.72, 0, 1);
        const releaseDrift = score.release * (0.25 + rtSeed(row * columns + column, 76) * 0.82);
        const points = source.map((point, pointIndex) => {
          const target = destination[pointIndex];
          const model = {
            x: lerp(point.x, target.x, ease(stagger)),
            y: lerp(point.y, target.y, ease(stagger)),
            z: lerp(point.z, target.z, ease(stagger))
          };
          if (score.release > 0) {
            const ether = rtOffsetPoint(0.74 + rtSeed(column + pointIndex, row + 91) * 0.25,
              (rtSeed(column, row + pointIndex + 93) - 0.5) * 1.4,
              (rtSeed(row, column + pointIndex + 97) - 0.5) * 1.2);
            model.x = lerp(model.x, ether.x, releaseDrift);
            model.y = lerp(model.y, ether.y, releaseDrift);
            model.z = lerp(model.z, ether.z, releaseDrift);
          }
          return rtRotateProject(model, yaw, pitch, roll);
        });
        facets.push({
          points,
          row,
          column,
          stagger,
          z: points.reduce((sum, point) => sum + point.z, 0) / points.length
        });
      }
    }

    facets.sort((a, b) => a.z - b.z);
    facets.forEach((facet) => {
      const isOwned = facet.stagger > 0.10;
      const fillColor = isOwned ? rtMixColor(WORLD.reality.rgb, RT_WHITE, 0.28) : RT_PALE;
      const edgeColor = isOwned ? (facet.column % 5 === 0 ? RT_GOLD : WORLD.reality.rgb) : RT_GREY;
      const shellPresence = (0.28 + score.bone * 0.54) * (1 - structure * 0.72) * (1 - score.release * 0.70);
      const ownedPresence = structure * (0.30 + score.energy * 0.30) * (1 - score.release * 0.64);
      ctx.beginPath();
      ctx.moveTo(facet.points[0].x, facet.points[0].y);
      for (let index = 1; index < facet.points.length; index++) ctx.lineTo(facet.points[index].x, facet.points[index].y);
      ctx.closePath();
      ctx.fillStyle = rgba(fillColor, alpha * (isOwned ? ownedPresence * 0.055 : shellPresence * 0.040));
      ctx.fill();
      if ((facet.row + facet.column) % 2 === 0 || score.bone > 0.5 || isOwned) {
        ctx.strokeStyle = rgba(edgeColor, alpha * (isOwned ? ownedPresence * 0.54 : shellPresence * 0.42));
        ctx.lineWidth = isOwned ? 0.75 + score.beat * 0.55 : 0.55;
        ctx.stroke();
      }
      if ((facet.row + facet.column) % 3 === 0) {
        rtLine(facet.points[0], facet.points[2], edgeColor, alpha * (isOwned ? ownedPresence * 0.28 : shellPresence * 0.17), 0.45);
      }
    });
  }

    const branches = [-1, 0, 1];

      let previous = null;
      for (let index = 0; index <= 54; index++) {
        const s = 0.27 + index / 54 * 0.39;
        const corridor = Math.sin((s - 0.27) / 0.39 * Math.PI);
        const drift = branch * corridor * (0.34 + 0.08 * Math.sin(t * 0.23 + branchIndex)) * (1 - score.command * 0.82);
        const lift = corridor * (branch === 0 ? -0.06 : 0.09);
        const projected = rtRotateProject(rtOffsetPoint(s, drift, lift));
        const branchAlpha = branch === 0
          ? 0.24 + score.command * 0.38 + score.high * 0.15
          : (0.20 * (1 - score.command * 0.88));

          alpha * branchAlpha * score.presence * (1 - score.release * 0.72),
          branch === 0 ? 0.9 + score.beat * 0.55 : 0.62);

      }
    });

  }

    if (s < 0.72) return lerp(0.10, 0.31, Math.sin((s - 0.48) / 0.24 * Math.PI));
    return lerp(0.24, 0.10, ease((s - 0.72) / 0.28));
  }

  function rtDrawDreamMembrane(t, alpha) {
    const score = rtScoreState;
    const count = lowCPU || coarse ? 72 : 116;
    const facets = [];
    for (let index = 0; index < count; index++) {
      const s = index / count;
      const next = (index + 1) / count;
      const twistA = Math.sin(s * TAU * 3 + t * 0.035) * 0.065;
      const twistB = Math.sin(next * TAU * 3 + t * 0.035) * 0.065;
      const shear = 1 - score.pressure * 0.44 + score.mid * 0.10;
      const widthA = rtRibbonWidth(s) * shear;
      const widthB = rtRibbonWidth(next) * shear;
      const leftA = rtRotateProject(rtOffsetPoint(s, -widthA, twistA));
      const rightA = rtRotateProject(rtOffsetPoint(s, widthA, -twistA));
      const leftB = rtRotateProject(rtOffsetPoint(next, -widthB, twistB));
      const rightB = rtRotateProject(rtOffsetPoint(next, widthB, -twistB));
      facets.push({
        s,
        points: [leftA, leftB, rightB, rightA],
        z: (leftA.z + rightA.z + leftB.z + rightB.z) * 0.25,
        color: rtColorAt(s)
      });
    }
    facets.sort((a, b) => a.z - b.z);
    facets.forEach((facet, index) => {
      ctx.beginPath();
      ctx.moveTo(facet.points[0].x, facet.points[0].y);

      element.style.top = `${clamp(top, 72, height - 62)}px`;
      element.style.opacity = String(clamp(1 - mix * 2.1, 0, 1));
      element.style.pointerEvents = mix < 0.25 ? 'auto' : 'none';
    });
    if (unityLabel) {
      const core = rtRotateProject({x:0,y:0,z:0});
      unityLabel.style.left = `${core.x}px`;
      unityLabel.style.top = `${core.y + unit * .055}px`;
      unityLabel.style.opacity = String(clamp(1 - mix * 2.2, 0, 1));
    }
    if (activeWorld && mix > 0.22) {
      subLabels.forEach((element, index) => {
        const point = subScreen[index];
        element.style.left = `${point.x}px`;
        element.style.top = `${point.y - point.r * 1.5}px`;
        element.style.opacity = String(ease((mix - .22) / .5));
        element.style.pointerEvents = mix > .65 ? 'auto' : 'none';
        element.classList.toggle('active', index === activeSub);
      });
    } else {
      subLabels.forEach((element) => { element.style.opacity = '0'; element.style.pointerEvents = 'none'; });
    }
    app?.classList.toggle('detail', Boolean(activeWorld && mix > .05));
  };
