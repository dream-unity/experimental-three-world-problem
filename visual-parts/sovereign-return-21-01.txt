
  // Sovereign Return 21 — the earlier white Dream Unity instrument, rebuilt
  // with higher-resolution projection, firmer proportions and a visible
  // distinction between the living form and the delayed mirror it commands.
  const SR_RELEASE = '20260831-sovereign-return-21';
  const SR_CORAL = [237, 113, 103];
  const SR_GOLD = [223, 160, 43];
  const SR_BONE = [238, 234, 224];
  const srFinite = (value, fallback = 0) => Number.isFinite(value) ? value : fallback;
  const srPointFinite = (point) => Boolean(point && Number.isFinite(point.x) && Number.isFinite(point.y));
  const SR_SHELL_VERTICES = [
    { x: 0, y: -1, z: -1.618 }, { x: 0, y: -1, z: 1.618 },
    { x: 0, y: 1, z: -1.618 }, { x: 0, y: 1, z: 1.618 },
    { x: -1, y: -1.618, z: 0 }, { x: -1, y: 1.618, z: 0 },
    { x: 1, y: -1.618, z: 0 }, { x: 1, y: 1.618, z: 0 },
    { x: -1.618, y: 0, z: -1 }, { x: 1.618, y: 0, z: -1 },
    { x: -1.618, y: 0, z: 1 }, { x: 1.618, y: 0, z: 1 },
  ].map((point) => {
    const magnitude = Math.max(0.0001, Math.hypot(point.x, point.y, point.z));
    return { x: point.x / magnitude, y: point.y / magnitude, z: point.z / magnitude };
  });
  const SR_SHELL_EDGES = [];
  for (let from = 0; from < SR_SHELL_VERTICES.length; from += 1) {
    for (let to = from + 1; to < SR_SHELL_VERTICES.length; to += 1) {
      const a = SR_SHELL_VERTICES[from];
      const b = SR_SHELL_VERTICES[to];
      if (Math.hypot(a.x - b.x, a.y - b.y, a.z - b.z) < 1.08) SR_SHELL_EDGES.push([from, to]);
    }
  }

  function srHighResolutionResize() {
    width = Math.max(1, srFinite(innerWidth, width || 1));
    height = Math.max(1, srFinite(innerHeight, height || 1));
    unit = Math.max(1, Math.min(width, height));
    const nativeRatio = Math.max(1, srFinite(devicePixelRatio, 1));
    const desired = Math.min(nativeRatio, coarse || lowCPU ? 1.9 : 2.45);
    const budget = coarse || lowCPU ? 5_800_000 : 8_900_000;
    dpr = Math.max(1, Math.min(desired, Math.sqrt(budget / Math.max(1, width * height))));
    if (!Number.isFinite(dpr)) dpr = 1;
    canvas.width = Math.round(width * dpr);
    canvas.height = Math.round(height * dpr);
    canvas.style.width = `${width}px`;
    canvas.style.height = `${height}px`;
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    ctx.imageSmoothingEnabled = true;
    ctx.imageSmoothingQuality = 'high';
  }

  // Preserve the asymmetric path, but remove the stretched, top-heavy feel.
  RT_PATH.forEach((point, index) => {
    point.x *= index === 6 || index === 11 ? 0.82 : 0.91;
    point.y *= point.y < -2 ? 0.84 : 0.91;
    point.z *= 0.86;
  });
  RT_PHASE.machine = 0.008;
  RT_PHASE.maker = 0.242;
  RT_PHASE.reality = 0.586;

  rtCenter = function sovereignReturnCenter() {
    return {
      x: width > 1180 ? width * 0.615 : width > 760 ? width * 0.575 : width * 0.50,
      y: width > 760 ? height * 0.495 : height * 0.43,
    };
  };

  rtProject = function sovereignReturnProject(point, scale = 1, cameraZ = 12.9, center = rtCenter()) {
    const model = {
      x: srFinite(point?.x, 0),
      y: srFinite(point?.y, 0),
      z: srFinite(point?.z, 0),
    };
    const safeCenter = {
      x: srFinite(center?.x, width * 0.5),
      y: srFinite(center?.y, height * 0.5),
    };
    const safeCameraZ = Math.max(3.36, srFinite(cameraZ, 12.9));
    const safeScale = Math.max(0.01, srFinite(scale, 1));
    const depth = safeCameraZ - model.z;
    const perspective = safeScale * safeCameraZ / Math.max(3.35, srFinite(depth, safeCameraZ));
    const spread = width > 1180 ? 0.126 : width > 760 ? 0.116 : 0.122;
    return {
      x: srFinite(safeCenter.x + model.x * unit * spread * perspective, safeCenter.x),
      y: srFinite(safeCenter.y - model.y * unit * spread * perspective, safeCenter.y),
      f: srFinite(perspective, 1),
      z: model.z,
      depth: srFinite(depth, safeCameraZ),
    };
  };

  function srProjectAt(point, yawValue, pitchValue, rollValue, scaleValue = overviewZoom) {
    const safePoint = {
      x: srFinite(point?.x, 0),
      y: srFinite(point?.y, 0),
      z: srFinite(point?.z, 0),
    };
    const rotated = rotate3(
      safePoint,
      srFinite(yawValue, 0),
      srFinite(pitchValue, 0),
      srFinite(rollValue, 0),
    );
    const projected = rtProject(rotated, srFinite(scaleValue, 1), 12.9);
    projected.world = rotated;
    return projected;
  }
