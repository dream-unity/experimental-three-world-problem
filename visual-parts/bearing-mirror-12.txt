  function bmWindow(value, start, peak, end) {
    if (value <= start || value >= end) return 0;
    if (value < peak) return ease((value - start) / Math.max(0.0001, peak - start));
    return 1 - ease((value - peak) / Math.max(0.0001, end - peak));
  }

  // The private reference is translated into a silent, dimensionless motion
  // anatomy: gather, subtract, pulse, rebuild, crown, return. No media from the
  // reference is required by, named in, or delivered to the browser.
  function bmMotionScore(t) {
    const phase = reduced ? 0.24 : ((t / 44) % 1 + 1) % 1;
    return {
      phase,
      inhale: bmWindow(phase, 0.00, 0.12, 0.28),
      pressure: bmWindow(phase, 0.10, 0.43, 0.535),
      subtraction: bmWindow(phase, 0.47, 0.515, 0.575),
      pulse: Math.max(
        bmWindow(phase, 0.545, 0.558, 0.575),
        bmWindow(phase, 0.575, 0.589, 0.607),
        bmWindow(phase, 0.607, 0.620, 0.64)
      ),
      rebuild: bmWindow(phase, 0.57, 0.69, 0.82),
      crown: bmWindow(phase, 0.68, 0.765, 0.855),
      release: ease(clamp((phase - 0.82) / 0.18, 0, 1))
    };
  }

  function bmCenter(detail = false) {
    if (width <= 900) return { x: width * 0.5, y: height * (detail ? 0.46 : 0.535) };

  function bmScale(detail = false) {
    if (width <= 900) {
      return detail
        ? Math.min(width * 0.82 / 5.10, height * 0.64 / 5.25)

  function bmOrientation(detail = false, ghost = false) {
    if (detail) {
      return {
        yaw: ghost ? bmDetailGhostYaw : detailYaw,
        pitch: clamp(ghost ? bmDetailGhostPitch : detailPitch, -0.314, 0.314),
        roll: clamp(detailRoll, -0.122, 0.122),
        zoom: clamp(detailZoom, 0.88, 1.12)
      };
    }
    return {
      yaw: ghost ? bmGhostYaw : overviewYaw,
      pitch: clamp(ghost ? bmGhostPitch : overviewPitch, -0.314, 0.314),
      roll: clamp(ghost ? bmGhostRoll : overviewRoll, -0.122, 0.122),
      zoom: clamp(overviewZoom, 0.88, 1.12)
    };
  }

  function bmProject(point, detail = false, ghost = false, offset = null) {
    const orientation = bmOrientation(detail, ghost);
    const rotated = rotate3(point, orientation.yaw, orientation.pitch, orientation.roll);
    const camera = 13.8;
    const perspective = orientation.zoom * camera / Math.max(5, camera - rotated.z);
    const center = bmCenter(detail);
    return {
      x: center.x + rotated.x * bmScale(detail) * perspective + (offset?.x || 0),
      y: center.y - rotated.y * bmScale(detail) * perspective + (offset?.y || 0),
      z: rotated.z,
      f: perspective
    };
  }

  function bmLocal(center, local, rotation = {}) {
    const rx = rotation.x || 0;
    const ry = rotation.y || 0;
    const rz = rotation.z || 0;
    let x = local.x;
    let y = local.y;
    let z = local.z;
    const cx = Math.cos(rx);
    const sx = Math.sin(rx);
    const cy = Math.cos(ry);
    const sy = Math.sin(ry);
    const cz = Math.cos(rz);
    const sz = Math.sin(rz);
    const y1 = y * cx - z * sx;
    const z1 = y * sx + z * cx;
    const x2 = x * cy + z1 * sy;
    const z2 = -x * sy + z1 * cy;
    const x3 = x2 * cz - y1 * sz;
    const y3 = x2 * sz + y1 * cz;
    return { x: center.x + x3, y: center.y + y3, z: center.z + z2 };
  }

  function bmFace(faces, points, color, alpha, options = {}) {
    faces.push({
      points,
      color,
      alpha,
      stroke: options.stroke || BM_INK,
      strokeAlpha: options.strokeAlpha ?? 0.56,
      lineWidth: options.lineWidth ?? 1.15,
      fill: options.fill !== false
    });
  }

  function bmPrism(faces, polygon, center, depth, color, alpha, rotation = {}, options = {}) {
    const front = polygon.map(([x, y]) => bmLocal(center, { x, y, z: depth * 0.5 }, rotation));
    const back = polygon.map(([x, y]) => bmLocal(center, { x, y, z: -depth * 0.5 }, rotation));
    bmFace(faces, back, srMixColor(color, BM_INK, 0.32), alpha * 0.72, options);
    for (let index = 0; index < polygon.length; index++) {
      const next = (index + 1) % polygon.length;
      const sideColor = index % 2
        ? srMixColor(color, BM_INK, 0.22)
        : srMixColor(color, BM_WHITE, 0.08);
      bmFace(faces, [back[index], back[next], front[next], front[index]], sideColor, alpha * 0.88, options);
    }
    bmFace(faces, front, color, alpha, options);
  }

  function bmBox(faces, center, size, color, alpha, rotation = {}, options = {}) {
    bmPrism(faces, [
      [-size.x * 0.5, -size.y * 0.5],
      [size.x * 0.5, -size.y * 0.5],
      [size.x * 0.5, size.y * 0.5],
      [-size.x * 0.5, size.y * 0.5]
    ], center, size.z, color, alpha, rotation, options);
  }

  function bmRenderFaces(faces, alpha = 1, detail = false) {
    const projected = faces.map((face) => {
      const points = face.points.map((point) => bmProject(point, detail));
      return {
        ...face,
        points,
        depth: points.reduce((sum, point) => sum + point.z, 0) / points.length
      };
    }).sort((a, b) => a.depth - b.depth);

    ctx.save();
    ctx.globalCompositeOperation = 'source-over';
    ctx.lineJoin = 'miter';
    ctx.lineCap = 'square';
    projected.forEach((face) => {
      if (!face.points.length) return;
      ctx.beginPath();
      face.points.forEach((point, index) => index
        ? ctx.lineTo(point.x, point.y)
        : ctx.moveTo(point.x, point.y));
      ctx.closePath();
      if (face.fill) {
        ctx.fillStyle = rgba(face.color, alpha * face.alpha);
        ctx.fill();
      }
      ctx.strokeStyle = rgba(face.stroke, alpha * face.strokeAlpha);
      ctx.lineWidth = face.lineWidth;
      ctx.stroke();
    });
    ctx.restore();
  }

  function bmLine3(a, b, color, alpha, lineWidth = 1, detail = false, dash = null, ghost = false, offset = null) {
    if (alpha <= 0.002) return;
    const pa = bmProject(a, detail, ghost, offset);
    const pb = bmProject(b, detail, ghost, offset);
    ctx.save();
    ctx.globalCompositeOperation = 'source-over';
    ctx.strokeStyle = rgba(color, alpha);
    ctx.lineWidth = lineWidth;
    ctx.lineCap = 'square';
    if (dash) ctx.setLineDash(dash);
    ctx.beginPath();
    ctx.moveTo(pa.x, pa.y);
    ctx.lineTo(pb.x, pb.y);
    ctx.stroke();
    ctx.restore();
  }

  function bmLoop3(points, color, alpha, lineWidth = 1, detail = false, ghost = false, offset = null, dash = null) {
    if (!points.length || alpha <= 0.002) return;
    const projected = points.map((point) => bmProject(point, detail, ghost, offset));
    ctx.save();
    ctx.globalCompositeOperation = 'source-over';
    ctx.strokeStyle = rgba(color, alpha);
    ctx.lineWidth = lineWidth;
    ctx.lineJoin = 'miter';
    if (dash) ctx.setLineDash(dash);
    ctx.beginPath();
    projected.forEach((point, index) => index
      ? ctx.lineTo(point.x, point.y)
      : ctx.moveTo(point.x, point.y));
    ctx.closePath();
    ctx.stroke();
    ctx.restore();
  }

  drawOverview = function bmOverview(t, alpha) {
    if (alpha <= 0.002) return;
    const dt = Math.min(0.05, Math.max(0.003, t - bmLastFrame));
    bmLastFrame = t;
    const moving = pointers.size > 0
      || Math.abs(overviewYawVelocity) + Math.abs(overviewPitchVelocity) + Math.abs(overviewRollVelocity) > 0.055;
    bmAuthority = damp(bmAuthority, moving ? 1 : 0.24, moving ? 8.5 : 0.22, dt);
    bmGhostYaw = damp(bmGhostYaw, overviewYaw, 1.15 + bmAuthority * 1.2, dt);
    bmGhostPitch = damp(bmGhostPitch, overviewPitch, 1.08 + bmAuthority, dt);
    bmGhostRoll = damp(bmGhostRoll, overviewRoll, 0.96 + bmAuthority, dt);

    const score = bmMotionScore(t);
    // One motion clock governs the material state. The public score bridge is
    // still sampled for continuity, but never max-mixed into a contradictory
    // crown, abyss or return. Completion belongs to interaction.
    const subtraction = score.subtraction;
    const pressure = clamp(0.18 + score.pressure * 0.30 + bmAuthority * 0.52, 0, 1);
    const rebuild = score.rebuild * (1 - subtraction) * (0.58 + bmAuthority * 0.42);
    const crown = score.crown * (1 - subtraction) * ease(clamp((bmAuthority - 0.18) / 0.82, 0, 1));
    const release = score.release * (1 - subtraction);
    const gap = 0.52 * (1 - pressure) + release * 0.18;
    const materialAlpha = clamp((0.82 + rebuild * 0.12) * (1 - subtraction * 0.88), 0.06, 0.94);
    const faces = [];

    bmOverviewGhost(gap, alpha);
    bmAddOverviewMachine(faces, gap, materialAlpha);
    bmAddOverviewMaker(faces, gap, materialAlpha);
    bmAddOverviewWorld(faces, gap, materialAlpha);
    bmAddMembrane(faces, materialAlpha, crown);
    bmAddTeeth(faces, gap, Math.max(materialAlpha, subtraction * 0.82));
    bmAddSeed(faces, materialAlpha, crown, score.pulse);
    bmRenderFaces(faces, alpha, false);

    // One exact axis survives subtraction. At the crown it becomes the lyric's
    // piercing light: three material planes in registration, never a glow.
    bmLine3({ x: 0, y: -3.02, z: 0.72 }, { x: 0, y: 3.02, z: 0.72 },
      subtraction > 0.30 ? BM_GOLD : BM_INK, alpha * (0.28 + subtraction * 0.64), 1.15, false);
    if (crown > 0.04) {
      bmLine3({ x: -2.42, y: 0.06, z: 0.96 }, { x: 2.42, y: 0.06, z: 0.96 }, BM_CYAN, alpha * crown * 0.88, 1.55);
      bmLine3({ x: -2.12, y: -0.02, z: 1.02 }, { x: 2.12, y: -0.02, z: 1.02 }, BM_GREEN, alpha * crown * 0.78, 1.25);
      bmLine3({ x: -1.82, y: -0.10, z: 1.08 }, { x: 1.82, y: -0.10, z: 1.08 }, BM_VIOLET, alpha * crown * 0.76, 1.15);
    }
    bmReturnFibres(release, alpha);

    const positions = overviewPositions(t);
    worldKeys.forEach((key) => Object.assign(worldScreen[key], positions[key]));
    worldKeys.forEach((key) => {
      const pin = bmBearing(BM_WORLD_PINS[key], key, t, alpha, hoverWorld === key);
      bmLeader(pin, bmWorldRail(key), WORLD[key].rgb, alpha);
    });
    ctx.globalAlpha = 1;
  };

  drawDetail = function bmDetail(t, alpha) {
    if (!activeWorld || alpha <= 0.002) return;
    const dt = Math.min(0.05, Math.max(0.003, t - bmLastFrame));
    bmDetailGhostYaw = damp(bmDetailGhostYaw, detailYaw, 1.18, dt);
    bmDetailGhostPitch = damp(bmDetailGhostPitch, detailPitch, 1.08, dt);
    if (activeWorld === 'machine') drawMachineField(t, alpha);
    else if (activeWorld === 'maker') drawMakerField(t, alpha);
    else drawRealityField(t, alpha);

    const pins = bmDetailPins();
    const points = detailPoints();
    points.forEach((point, index) => Object.assign(subScreen[index], point));
    pins.forEach((pin, index) => {
      const projected = bmBearing(pin, activeWorld, t + index * 0.7, alpha, index === activeSub || index === hoverSub, true);
      bmLeader(projected, bmDetailRail(index), WORLD[activeWorld].rgb, alpha);
    });
    ctx.globalAlpha = 1;
  };

  updateLabels = function bmLabels() {
    if (++labelTick % 2) return;
    const mix = ease(viewMix);
    worldKeys.forEach((key) => {
      const element = labels[key];
      if (!element) return;
      const rail = bmWorldRail(key);
      element.style.left = `${rail.x}px`;
      element.style.top = `${rail.y}px`;
      element.style.opacity = String(clamp(1 - mix * 2.1, 0, 1));
      element.style.pointerEvents = mix < 0.25 ? 'auto' : 'none';
      element.tabIndex = mix < 0.25 ? 0 : -1;
      element.setAttribute('aria-hidden', String(mix >= 0.25));
    });
    if (unityLabel) {
      const center = bmCenter(false);
      unityLabel.style.left = `${center.x}px`;
      unityLabel.style.top = `${center.y + bmScale(false) * 3.38}px`;
      unityLabel.style.opacity = String(clamp(1 - mix * 2.2, 0, 1));
    }

    if (activeWorld && mix > 0.22) {
      subLabels.forEach((element, index) => {
        if (!element) return;
        const rail = bmDetailRail(index);
        element.style.left = `${rail.x}px`;
        element.style.top = `${rail.y}px`;
        const visible = ease((mix - 0.22) / 0.5);
        element.style.opacity = String(visible);
        element.style.pointerEvents = mix > 0.65 ? 'auto' : 'none';
        element.tabIndex = mix > 0.65 ? 0 : -1;
        element.setAttribute('aria-hidden', String(mix <= 0.65));
        element.setAttribute('aria-pressed', String(index === activeSub));
        element.style.setProperty('--sub-accent', WORLD[activeWorld].css);
        element.classList.toggle('active', index === activeSub);
      });
    } else {
      subLabels.forEach((element) => {
        if (!element) return;
        element.style.opacity = '0';
        element.style.pointerEvents = 'none';
        element.tabIndex = -1;
        element.setAttribute('aria-hidden', 'true');
      });
    }
    if (back) {
      const visible = Boolean(activeWorld && mix > 0.05);
      back.tabIndex = visible ? 0 : -1;
      back.setAttribute('aria-hidden', String(!visible));
    }
    app?.classList.toggle('detail', Boolean(activeWorld && mix > 0.05));
  };
