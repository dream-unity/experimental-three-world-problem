    ctx.fillStyle = '#02040a';
    ctx.fillRect(0, 0, width, height);

    const nebulae = [
      [width * 0.22, height * 0.28, unit * 0.46, WORLD.machine.rgb, 0.10],
      [width * 0.78, height * 0.28, unit * 0.44, WORLD.maker.rgb, 0.075],
      [width * 0.51, height * 0.78, unit * 0.43, WORLD.reality.rgb, 0.07]
    ];
    for (const [x, y, radius, color, alpha] of nebulae) {
      const gradient = ctx.createRadialGradient(x, y, 0, x, y, radius);
      gradient.addColorStop(0, rgba(color, alpha));
      gradient.addColorStop(0.42, rgba(color, alpha * 0.36));
      gradient.addColorStop(1, rgba(color, 0));
      ctx.fillStyle = gradient;
      ctx.fillRect(x - radius, y - radius, radius * 2, radius * 2);
    }

    // Background parallax is deliberately much smaller than the triad's
    // rotation, making it obvious that the object—not the screen—is moving.
    const parallaxYaw = activeWorld ? detailYaw * 0.035 : overviewYaw * 0.055;
    const parallaxPitch = activeWorld ? detailPitch * 0.025 : overviewPitch * 0.045;
    const idle = t * 0.0025;
    ctx.globalCompositeOperation = 'lighter';
    for (let i = 0; i < stars.length; i++) {
      const star = stars[i];
      const rotated = rotate3(star, parallaxYaw + idle, parallaxPitch, 0);
      const projected = project(rotated, 0.74, 15);
      if (projected.x < 0 || projected.x > width || projected.y < 0 || projected.y > height) continue;
      const twinkle = 0.45 + 0.55 * Math.sin(t * (0.6 + (i % 5) * 0.08) + i * 0.91) ** 2;
      const alpha = star.alpha * twinkle;
      const tint = star.tint === 0
        ? [120, 220, 255]
        : star.tint === 1
          ? [190, 150, 255]
          : star.tint === 2
            ? [130, 255, 190]
            : [230, 235, 255];
      ctx.fillStyle = rgba(tint, alpha);
      const radius = star.size * clamp(projected.f, 0.45, 1.6);
      ctx.fillRect(projected.x, projected.y, radius, radius);
    }
  }

  function drawCurve(a, b, rgb, alpha, seed, t, packets = true) {
    const depthBias = clamp(0.65 + ((a.z || 0) + (b.z || 0)) * 0.045, 0.42, 1.15);
    const mx = (a.x + b.x) * 0.5 + Math.sin(t * 0.43 + seed) * unit * 0.025;
    const my = (a.y + b.y) * 0.5 - unit * 0.045 + Math.cos(t * 0.37 + seed) * unit * 0.02;
    ctx.globalCompositeOperation = 'lighter';
    ctx.lineWidth = 1;
    ctx.strokeStyle = rgba(rgb, alpha * 0.36 * depthBias);
    ctx.beginPath();
    ctx.moveTo(a.x, a.y);
    ctx.quadraticCurveTo(mx, my, b.x, b.y);
    ctx.stroke();
    ctx.lineWidth = 0.35;
    ctx.strokeStyle = rgba(rgb, alpha * 0.5 * depthBias);
    ctx.beginPath();
    ctx.moveTo(a.x, a.y);
    ctx.quadraticCurveTo(mx, my, b.x, b.y);
    ctx.stroke();
    if (!packets) return;

    for (let j = 0; j < 3; j++) {
      const u = (t * 0.055 + seed * 0.07 + j / 3) % 1;
      const one = 1 - u;
      const x = one * one * a.x + 2 * one * u * mx + u * u * b.x;
      const y = one * one * a.y + 2 * one * u * my + u * u * b.y;
      const radius = 1.5 + 2.2 * Math.sin(u * Math.PI);
      const gradient = ctx.createRadialGradient(x, y, 0, x, y, radius * 4);
      gradient.addColorStop(0, rgba(rgb, alpha * 0.85 * depthBias));
      gradient.addColorStop(1, rgba(rgb, 0));
      ctx.fillStyle = gradient;
      ctx.fillRect(x - radius * 4, y - radius * 4, radius * 8, radius * 8);
    }
  }

  function drawOrb(x, y, radius, config, kind, t, alpha = 1, focus = 0, depth = 0) {
    if (alpha <= 0.002 || radius <= 1) return;
    ctx.save();
    const depthAlpha = clamp(0.72 + depth * 0.055, 0.52, 1.08);
    ctx.globalAlpha = alpha * depthAlpha;
    ctx.globalCompositeOperation = 'lighter';

    const glowRadius = radius * (2.05 + focus * 0.25);
    let gradient = ctx.createRadialGradient(x, y, 0, x, y, glowRadius);
    gradient.addColorStop(0, rgba(config.rgb, 0.38));
    gradient.addColorStop(0.25, rgba(config.rgb, 0.18));
    gradient.addColorStop(1, rgba(config.rgb, 0));
    ctx.fillStyle = gradient;
    ctx.fillRect(x - glowRadius, y - glowRadius, glowRadius * 2, glowRadius * 2);

    gradient = ctx.createRadialGradient(x - radius * 0.24, y - radius * 0.28, radius * 0.03, x, y, radius);
    gradient.addColorStop(0, 'rgba(255,255,255,.96)');
    gradient.addColorStop(0.18, rgba(config.rgb, 0.88));
    gradient.addColorStop(0.58, rgba(config.rgb, 0.34));
    gradient.addColorStop(1, rgba(config.rgb, 0.045));
    ctx.fillStyle = gradient;
    ctx.beginPath();
    ctx.arc(x, y, radius, 0, TAU);
    ctx.fill();

    ctx.strokeStyle = rgba(config.rgb, 0.48);
    ctx.lineWidth = Math.max(0.7, radius * 0.008);
    for (let i = 0; i < 3; i++) {
      const ringRadius = radius * (1.15 + i * 0.23);
      ctx.save();
      ctx.translate(x, y);
      ctx.rotate(t * (0.08 + i * 0.025) + (kind === 'maker' ? i * 0.65 : i * 0.4));
      ctx.scale(1, 0.35 + i * 0.12);
      ctx.beginPath();
      ctx.arc(0, 0, ringRadius, 0, TAU);
      ctx.stroke();
      ctx.restore();
    }

    if (kind === 'machine') {
      ctx.strokeStyle = rgba([205, 248, 255], 0.42);
      ctx.lineWidth = 0.8;
      for (let i = 0; i < 9; i++) {
        const angle = t * 0.12 + i * TAU / 9;
        const x1 = x + Math.cos(angle) * radius * 0.32;
        const y1 = y + Math.sin(angle * 1.37) * radius * 0.26;
        const x2 = x + Math.cos(angle + 1.9) * radius * 0.68;
        const y2 = y + Math.sin(angle * 1.17 + 1.2) * radius * 0.58;
        ctx.beginPath();
        ctx.moveTo(x1, y1);
        ctx.lineTo(x2, y2);
        ctx.stroke();
      }
    } else if (kind === 'maker') {
      ctx.strokeStyle = rgba([220, 255, 235], 0.52);
      ctx.lineWidth = 1;
      ctx.beginPath();
      for (let i = 0; i <= 40; i++) {
        const angle = i / 40 * TAU * 2 + t * 0.22;
        const rr = radius * (0.13 + 0.011 * i);
        const px = x + Math.cos(angle) * rr;
        const py = y + Math.sin(angle) * rr * 0.55;
        i ? ctx.lineTo(px, py) : ctx.moveTo(px, py);
      }
      ctx.stroke();
    } else {
      ctx.strokeStyle = rgba([235, 215, 255], 0.4);
      ctx.lineWidth = 0.75;
      for (let i = -2; i <= 2; i++) {
        ctx.beginPath();
        ctx.moveTo(x - radius * 0.55, y + i * radius * 0.18);
