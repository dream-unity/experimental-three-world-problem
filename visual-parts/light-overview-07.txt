
  // Dream Unity white kinetic renderer override.
  // Keeps the field genuinely white while restoring depth, motion and visual
  // richness through object-local ribbons, satellites and layered orbit geometry.

  WORLD.machine.css = '#009DFF';
  WORLD.machine.rgb = [0, 157, 255];
  WORLD.maker.css = '#00C97A';
  WORLD.maker.rgb = [0, 201, 122];
  WORLD.reality.css = '#7A36F5';
  WORLD.reality.rgb = [122, 54, 245];

  const WHITE_ALT = {
    machine:[0,92,235],
    maker:[0,146,112],
    reality:[210,35,255],
    unity:[255,151,0]
  };

  function wLerpColor(a,b,t){return [
    Math.round(a[0]+(b[0]-a[0])*t),
    Math.round(a[1]+(b[1]-a[1])*t),
    Math.round(a[2]+(b[2]-a[2])*t)
  ];}

  drawBackground = function whiteKineticBackground(t) {
    ctx.globalCompositeOperation = 'source-over';
    ctx.globalAlpha = 1;
    ctx.fillStyle = '#ffffff';
    ctx.fillRect(0, 0, width, height);

    // Sparse cool micro-particles: enough to suggest depth, never enough to haze.
    const parallaxYaw = activeWorld ? detailYaw * 0.026 : overviewYaw * 0.04;
    const parallaxPitch = activeWorld ? detailPitch * 0.02 : overviewPitch * 0.03;
    const idle = t * 0.0016;
    for (let i = 0; i < stars.length; i += 3) {
      const star = stars[i];
      const rotated = rotate3(star, parallaxYaw + idle, parallaxPitch, 0);
      const projected = project(rotated, 0.74, 15);
      if (projected.x < 0 || projected.x > width || projected.y < 0 || projected.y > height) continue;
      const pulse = 0.45 + 0.55 * Math.sin(t * 0.48 + i * 0.77) ** 2;
      const tint = star.tint === 0 ? WORLD.machine.rgb : star.tint === 1 ? WORLD.reality.rgb : star.tint === 2 ? WORLD.maker.rgb : [95,110,130];
      ctx.fillStyle = rgba(tint, (star.tint < 3 ? 0.13 : 0.038) * pulse);
      const radius = Math.min(1.15, Math.max(0.45, star.size * 0.46 * clamp(projected.f, 0.5, 1.25)));
      ctx.beginPath();
      ctx.arc(projected.x, projected.y, radius, 0, TAU);
      ctx.fill();
    }
  };

  function wCurvePoint(p0,p1,p2,p3,u){
    const o=1-u;
    return {
      x:o*o*o*p0.x+3*o*o*u*p1.x+3*o*u*u*p2.x+u*u*u*p3.x,
      y:o*o*o*p0.y+3*o*o*u*p1.y+3*o*u*u*p2.y+u*u*u*p3.y
    };
  }

  drawCurve = function fluidLongCurve(a, b, rgb, alpha, seed, t, packets = true) {
    const dx=b.x-a.x,dy=b.y-a.y;
    const len=Math.max(1,Math.hypot(dx,dy));
    const tx=dx/len,ty=dy/len;
    const nx=-ty,ny=tx;
    const depthBias=clamp(0.72+((a.z||0)+(b.z||0))*0.035,0.5,1.08);
    const bow=(0.095+0.025*Math.sin(t*0.32+seed))*unit;
    const reach=Math.min(unit*0.075,len*0.13);

    // Extend slightly beyond both anchors so each relation reads as a long field
    // line rather than a short wire between two dots.
    const p0={x:a.x-tx*reach,y:a.y-ty*reach};
    const p3={x:b.x+tx*reach,y:b.y+ty*reach};
    const p1={x:a.x+dx*0.28+nx*bow,y:a.y+dy*0.28+ny*bow};
    const p2={x:a.x+dx*0.72-nx*bow*0.72,y:a.y+dy*0.72-ny*bow*0.72};

    ctx.globalCompositeOperation='source-over';
    ctx.lineCap='round';

    // Soft local under-stroke for visual richness, not atmospheric fog.
    ctx.strokeStyle=rgba(rgb,alpha*0.09*depthBias);
    ctx.lineWidth=5.4;
    ctx.beginPath();ctx.moveTo(p0.x,p0.y);ctx.bezierCurveTo(p1.x,p1.y,p2.x,p2.y,p3.x,p3.y);ctx.stroke();

    ctx.strokeStyle=rgba(rgb,alpha*0.42*depthBias);
    ctx.lineWidth=1.45;
    ctx.beginPath();ctx.moveTo(p0.x,p0.y);ctx.bezierCurveTo(p1.x,p1.y,p2.x,p2.y,p3.x,p3.y);ctx.stroke();

    // Offset echo arc makes each connection feel braided and less rigid.
    const echo=unit*0.018*(seed%2?1:-1);
    ctx.strokeStyle=rgba(rgb,alpha*0.12*depthBias);
    ctx.lineWidth=0.85;
    ctx.beginPath();
    ctx.moveTo(p0.x+nx*echo,p0.y+ny*echo);
    ctx.bezierCurveTo(p1.x+nx*echo,p1.y+ny*echo,p2.x+nx*echo,p2.y+ny*echo,p3.x+nx*echo,p3.y+ny*echo);
    ctx.stroke();

    if (!packets) return;
    for(let j=0;j<4;j++){
      const u=(t*0.045+seed*0.073+j/4)%1;
      const p=wCurvePoint(p0,p1,p2,p3,u);
      const rr=2.0+1.6*Math.sin(u*Math.PI);
      ctx.fillStyle=rgba(rgb,0.82*alpha*depthBias);
      ctx.beginPath();ctx.arc(p.x,p.y,rr,0,TAU);ctx.fill();
      ctx.fillStyle='rgba(255,255,255,.9)';
      ctx.beginPath();ctx.arc(p.x-rr*.22,p.y-rr*.22,Math.max(.65,rr*.22),0,TAU);ctx.fill();
    }
  };

  drawOrb = function sculpturalOrb(x, y, radius, config, kind, t, alpha = 1, focus = 0, depth = 0) {
    if(alpha<=0.002||radius<=1)return;
    ctx.save();
    const depthAlpha=clamp(0.76+depth*0.045,0.58,1.08);
    ctx.globalAlpha=alpha*depthAlpha;
    ctx.globalCompositeOperation='source-over';
    const alt=WHITE_ALT[kind]||config.rgb;

    // Compact local aura only.
    const aura=radius*1.18;
    let g=ctx.createRadialGradient(x,y,0,x,y,aura);
    g.addColorStop(0,rgba(config.rgb,0.16+focus*0.035));
    g.addColorStop(0.5,rgba(config.rgb,0.055));
    g.addColorStop(1,rgba(config.rgb,0));
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(x,y,aura,0,TAU);ctx.fill();

    // Main translucent shell.
    g=ctx.createRadialGradient(x-radius*.22,y-radius*.24,radius*.03,x,y,radius);
    g.addColorStop(0,'rgba(255,255,255,.96)');
    g.addColorStop(.14,rgba(config.rgb,.78));
    g.addColorStop(.58,rgba(config.rgb,.20));
    g.addColorStop(1,rgba(alt,.055));
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(x,y,radius,0,TAU);ctx.fill();
    ctx.strokeStyle=rgba(alt,.72+focus*.1);ctx.lineWidth=Math.max(1.1,radius*.011);ctx.stroke();

    // Offset inner nucleus gives more sculptural depth.
    ctx.fillStyle=rgba(alt,.92);
    ctx.beginPath();ctx.arc(x-radius*.05,y+radius*.035,radius*.22,0,TAU);ctx.fill();
    ctx.fillStyle='rgba(255,255,255,.94)';
    ctx.beginPath();ctx.arc(x-radius*.115,y-radius*.07,Math.max(1.4,radius*.052),0,TAU);ctx.fill();

    // Five distinct orbital ribbons instead of three repeated ellipses.
    for(let i=0;i<5;i++){
      const mix=i/4;
      const ringColor=wLerpColor(config.rgb,alt,mix*.78);
      const ringRadius=radius*(1.18+i*.18);
      ctx.save();ctx.translate(x,y);
      ctx.rotate(t*(0.055+i*.017)+(kind==='maker'?i*.71:kind==='reality'?i*.48:i*.37));
      ctx.scale(1,0.28+i*.095);
      ctx.strokeStyle=rgba(ringColor,.28+i*.065+focus*.025);
      ctx.lineWidth=i===0?1.45:0.9;
      ctx.beginPath();ctx.arc(0,0,ringRadius,0,TAU);ctx.stroke();
      // Tiny satellite riding each ribbon.
      const ang=t*(0.48+i*.09)+i*1.37;
      const sx=Math.cos(ang)*ringRadius,sy=Math.sin(ang)*ringRadius;
      ctx.fillStyle=rgba(ringColor,.78);ctx.beginPath();ctx.arc(sx,sy,Math.max(1.2,radius*.018),0,TAU);ctx.fill();
      ctx.restore();
    }

    // Distinct world-specific interior languages.
    if(kind==='machine'){
      ctx.strokeStyle=rgba(alt,.72);ctx.lineWidth=1.1;
      for(let i=0;i<10;i++){
        const a=t*.11+i*TAU/10;
        const r1=radius*(.24+(i%3)*.055),r2=radius*(.62+(i%2)*.07);
        ctx.beginPath();ctx.moveTo(x+Math.cos(a)*r1,y+Math.sin(a)*r1);ctx.lineTo(x+Math.cos(a+.28)*r2,y+Math.sin(a+.28)*r2);ctx.stroke();
      }
    }else if(kind==='maker'){
      ctx.strokeStyle=rgba(alt,.76);ctx.lineWidth=1.2;ctx.beginPath();
      for(let i=0;i<=58;i++){
        const a=i/58*TAU*2.5+t*.19;
        const rr=radius*(.10+.0085*i);
        const px=x+Math.cos(a)*rr,py=y+Math.sin(a)*rr*.51;
        i?ctx.lineTo(px,py):ctx.moveTo(px,py);
      }ctx.stroke();
    }else{
      ctx.strokeStyle=rgba(alt,.60);ctx.lineWidth=.95;
      for(let i=-3;i<=3;i++){
        ctx.beginPath();ctx.moveTo(x-radius*.58,y+i*radius*.14);ctx.lineTo(x+radius*.58,y+i*radius*.14);ctx.stroke();
        ctx.beginPath();ctx.moveTo(x+i*radius*.14,y-radius*.58);ctx.lineTo(x+i*radius*.14,y+radius*.58);ctx.stroke();
      }
      ctx.save();ctx.translate(x,y);ctx.rotate(t*.08);ctx.strokeStyle=rgba(config.rgb,.35);ctx.strokeRect(-radius*.34,-radius*.34,radius*.68,radius*.68);ctx.restore();
    }
    ctx.restore();
  };

  function wDrawUnity(unity,t,alpha){
    const amber=[245,158,11],orange=[255,105,0];
    const coreR=unit*0.032*overviewZoom;
    const oracleState=app&&app.dataset?app.dataset.voiceState||'idle':'idle';
    const oracleSpeaking=oracleState==='speaking';
    const oracleListening=oracleState==='listening'||oracleState==='connected';
    const oracleHover=app&&app.dataset?app.dataset.voiceHover==='true':false;
    const oraclePress=app&&app.dataset?app.dataset.voicePress==='true':false;
    const reduceMotion=window.matchMedia&&window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    const motionT=reduceMotion?0:t;
    const oraclePulse=oracleSpeaking ? .5+.5*Math.sin(motionT*8.4) : oracleListening ? .5+.5*Math.sin(motionT*2.2) : .5+.5*Math.sin(motionT*1.35);
    const engaged=oracleSpeaking||oracleListening||oracleHover;
    const bodyR=coreR*(1+(oracleSpeaking ? .065*oraclePulse : oracleListening ? .028*oraclePulse : 0)+(oracleHover ? .055 : 0)-(oraclePress ? .035 : 0));
    const rings=[
      [unit*.075,unit*.025,motionT*.11+overviewRoll],
      [unit*.105,unit*.034,-motionT*.075-overviewRoll*.7],
      [unit*.135,unit*.043,motionT*.048+1.05]
    ];

    // Split orbit planes around the sphere: rear arcs first, front arcs last.
    // The resulting occlusion makes Unity read as a dimensional object.
    const drawOrbitStage=front=>rings.forEach((r,i)=>{
      ctx.save();ctx.translate(unity.x,unity.y);ctx.rotate(r[2]);ctx.lineCap='round';
      const oracleLift=oracleSpeaking ? .19*oraclePulse : oracleListening ? .09*oraclePulse : oracleHover ? .07 : 0;
      ctx.strokeStyle=rgba(i===2?orange:amber,(front ? .58-i*.085 : .30-i*.045+oracleLift)*alpha);
      ctx.lineWidth=(front ? (i===0?2:1.3) : .82)+(oracleSpeaking ? .24*oraclePulse : 0);
      ctx.beginPath();ctx.ellipse(0,0,r[0]*overviewZoom,r[1]*overviewZoom,0,front?0:Math.PI,front?Math.PI:TAU);ctx.stroke();
      const nodeA=motionT*(.55+i*.13)+i*2.1;
      const nodeFront=Math.sin(nodeA)>=0;
      if(nodeFront===front){
        const sx=Math.cos(nodeA)*r[0]*overviewZoom,sy=Math.sin(nodeA)*r[1]*overviewZoom;
        const nodeScale=front?1.12:.78;
        ctx.fillStyle=rgba(i===2?orange:amber,(front ? .96 : .55)*alpha);
        ctx.beginPath();ctx.arc(sx,sy,(2.1+i*.25)*nodeScale,0,TAU);ctx.fill();
        if(front){ctx.fillStyle=rgba([255,255,255],.72*alpha);ctx.beginPath();ctx.arc(sx-.55,sy-.55,.65,0,TAU);ctx.fill();}
      }
      ctx.restore();
    });
    drawOrbitStage(false);

    // Projected sound shells make the core visibly transmit and receive voice.
    const shellCount=oracleSpeaking?3:engaged?2:1;
    for(let i=0;i<shellCount;i++){
      const rawPhase=reduceMotion?(i+1)/(shellCount+1):(motionT*(oracleSpeaking?1.35:.34)+i/shellCount)%1;
      const phase=oraclePress?Math.min(rawPhase,.16):rawPhase;
      const shellR=bodyR*(1.34+phase*(oracleSpeaking?1.48:.78));
      const shellAlpha=(oracleSpeaking ? .52 : engaged ? .34 : .23)*(1-phase*.76)*alpha;
      ctx.save();ctx.translate(unity.x,unity.y);ctx.rotate(-.11);ctx.lineCap='round';
      ctx.strokeStyle=rgba(i%2?orange:amber,shellAlpha);ctx.lineWidth=(oracleSpeaking?1.8:1.35)*(1-phase*.22);
      ctx.beginPath();ctx.ellipse(0,0,shellR,shellR*.72,0,-.67,.67);ctx.stroke();
      ctx.beginPath();ctx.ellipse(0,0,shellR,shellR*.72,0,Math.PI-.67,Math.PI+.67);ctx.stroke();
      ctx.restore();
    }

    // Enlarged spherical core with a local aura and grounded shadow.
    const aura=coreR*(2.55+(oracleSpeaking ? .42*oraclePulse : oracleListening ? .2*oraclePulse : oracleHover ? .22 : 0));
    const g=ctx.createRadialGradient(unity.x,unity.y,0,unity.x,unity.y,aura);
    g.addColorStop(0,rgba(amber,.25*alpha));g.addColorStop(.45,rgba(amber,.085*alpha));g.addColorStop(1,rgba(amber,0));
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(unity.x,unity.y,aura,0,TAU);ctx.fill();

    ctx.fillStyle=rgba([116,65,0],.14*alpha);ctx.beginPath();ctx.ellipse(unity.x,unity.y+bodyR*.88,bodyR*.82,bodyR*.22,0,0,TAU);ctx.fill();
    const sphere=ctx.createRadialGradient(unity.x-bodyR*.38,unity.y-bodyR*.43,bodyR*.06,unity.x,unity.y,bodyR*1.12);
    sphere.addColorStop(0,rgba([255,241,184],.99*alpha));
    sphere.addColorStop(.24,rgba([255,200,68],.99*alpha));
    sphere.addColorStop(.7,rgba(amber,.98*alpha));
    sphere.addColorStop(1,rgba([185,91,0],.98*alpha));
    ctx.save();ctx.shadowColor=rgba(orange,.28*alpha);ctx.shadowBlur=bodyR*.58;ctx.fillStyle=sphere;ctx.beginPath();ctx.arc(unity.x,unity.y,bodyR,0,TAU);ctx.fill();ctx.restore();
    ctx.strokeStyle=rgba([134,75,0],.34*alpha);ctx.lineWidth=1.15;ctx.beginPath();ctx.arc(unity.x,unity.y,bodyR-.65,0,TAU);ctx.stroke();

    // Preserve the familiar orange nucleus, now as a glassy oracle iris.
    const irisX=unity.x+bodyR*.16,irisY=unity.y-bodyR*.07;
    const nucleus=ctx.createRadialGradient(irisX-bodyR*.13,irisY-bodyR*.15,0,irisX,irisY,bodyR*.52);
    nucleus.addColorStop(0,rgba([255,171,35],.98*alpha));nucleus.addColorStop(.62,rgba(orange,.96*alpha));nucleus.addColorStop(1,rgba([203,68,0],.93*alpha));
    ctx.fillStyle=nucleus;ctx.beginPath();ctx.arc(irisX,irisY,bodyR*.46,0,TAU);ctx.fill();
    const irisOpen=bodyR*(.075+(oracleSpeaking ? .105*oraclePulse : oracleListening ? .045*oraclePulse : oracleHover ? .035 : 0));
    const iris=ctx.createRadialGradient(irisX-bodyR*.05,irisY,0,irisX,irisY,bodyR*.31);
    iris.addColorStop(0,rgba([35,24,13],.96*alpha));iris.addColorStop(.62,rgba([91,43,0],.92*alpha));iris.addColorStop(1,rgba([178,76,0],.45*alpha));
    ctx.fillStyle=iris;ctx.beginPath();ctx.ellipse(irisX,irisY+bodyR*.045,bodyR*.27,Math.max(1.25,irisOpen),-.08,0,TAU);ctx.fill();
    ctx.strokeStyle=rgba([255,231,169],.72*alpha);ctx.lineWidth=1;ctx.beginPath();ctx.ellipse(irisX,irisY+bodyR*.03,bodyR*.31,Math.max(2,irisOpen*1.45),-.08,Math.PI*.12,Math.PI*.88);ctx.stroke();
    ctx.fillStyle=rgba([255,255,255],.92*alpha);ctx.beginPath();ctx.arc(unity.x-bodyR*.35,unity.y-bodyR*.4,Math.max(1.7,bodyR*.11),0,TAU);ctx.fill();

    drawOrbitStage(true);
  }

  drawOverview = function kineticWhiteOverview(t, alpha) {
    const positions=overviewPositions(t);
    worldKeys.forEach(key=>Object.assign(worldScreen[key],positions[key]));
    const unity={x:width*.5,y:height*.5,r:unit*.05,z:0};

    ctx.globalCompositeOperation='source-over';ctx.globalAlpha=alpha;

    // One broad, barely visible structural spline behind the active connections.
    const ordered=worldKeys.map(key=>positions[key]);
    ctx.strokeStyle=`rgba(65,80,105,${.055*alpha})`;ctx.lineWidth=1;
    ctx.beginPath();ctx.moveTo(ordered[0].x,ordered[0].y);
    ctx.bezierCurveTo(width*.48,height*.16,width*.77,height*.20,ordered[1].x,ordered[1].y);
    ctx.bezierCurveTo(width*.73,height*.55,width*.58,height*.74,ordered[2].x,ordered[2].y);
    ctx.bezierCurveTo(width*.32,height*.67,width*.27,height*.38,ordered[0].x,ordered[0].y);ctx.stroke();

    // Long world-to-world ribbons.
    const pairs=[['machine','maker'],['maker','reality'],['reality','machine']];
    pairs.forEach((pair,index)=>drawCurve(positions[pair[0]],positions[pair[1]],WORLD[pair[0]].rgb,alpha*.88,index+1,t));

    // Stronger, longer radial relations into the larger Unity core.
    worldKeys.forEach((key,index)=>drawCurve(unity,positions[key],WORLD[key].rgb,alpha*.95,8+index,t));

    wDrawUnity(unity,t,alpha);

    const depthOrder=[...worldKeys].sort((a,b)=>positions[a].z-positions[b].z);
    depthOrder.forEach(key=>{
      const p=positions[key];
      drawOrb(p.x,p.y,p.r*1.06,WORLD[key],key,t,alpha,hoverWorld===key?1:0,p.z);
    });
    ctx.globalAlpha=1;
  };

  // Pure white frame. The legacy full-screen black vignette never runs.
  render = function kineticWhiteRender(t) {
    drawBackground(t);
    const mix=ease(viewMix);
    drawOverview(t,1-mix);
    drawDetail(t,mix);
    updateLabels();
    ctx.globalCompositeOperation='source-over';
    ctx.globalAlpha=1;
  };
