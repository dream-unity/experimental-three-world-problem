
  // Dream Unity crisp white renderer override.
  // Runs inside the visual engine IIFE immediately before its closing token.
  // The original renderer was designed for black and painted a dark radial
  // vignette every frame. This override removes all atmospheric darkness and
  // replaces additive/pastel primitives with saturated, high-contrast geometry.

  WORLD.machine.css = '#008CFF';
  WORLD.machine.rgb = [0, 140, 255];
  WORLD.maker.css = '#00C875';
  WORLD.maker.rgb = [0, 200, 117];
  WORLD.reality.css = '#7C3AED';
  WORLD.reality.rgb = [124, 58, 237];

  drawBackground = function crispWhiteBackground(t) {
    ctx.globalCompositeOperation = 'source-over';
    ctx.globalAlpha = 1;
    ctx.fillStyle = '#ffffff';
    ctx.fillRect(0, 0, width, height);

    // Sparse colored micro-particles provide depth without haze or gray fog.
    const parallaxYaw = activeWorld ? detailYaw * 0.03 : overviewYaw * 0.045;
    const parallaxPitch = activeWorld ? detailPitch * 0.022 : overviewPitch * 0.035;
    const idle = t * 0.0018;
    for (let i = 0; i < stars.length; i += 2) {
      const star = stars[i];
      const rotated = rotate3(star, parallaxYaw + idle, parallaxPitch, 0);
      const projected = project(rotated, 0.74, 15);
      if (projected.x < 0 || projected.x > width || projected.y < 0 || projected.y > height) continue;
      const twinkle = 0.35 + 0.65 * Math.sin(t * (0.45 + (i % 5) * 0.05) + i * 0.73) ** 2;
      const tint = star.tint === 0 ? WORLD.machine.rgb : star.tint === 1 ? WORLD.reality.rgb : star.tint === 2 ? WORLD.maker.rgb : [71,85,105];
      const alpha = (star.tint < 3 ? 0.14 : 0.055) * twinkle;
      ctx.fillStyle = rgba(tint, alpha);
      const radius = Math.min(1.15, Math.max(0.45, star.size * 0.55 * clamp(projected.f, 0.5, 1.3)));
      ctx.fillRect(projected.x, projected.y, radius, radius);
    }
  };

  drawCurve = function crispCurve(a, b, rgb, alpha, seed, t, packets = true) {
    const depthBias = clamp(0.72 + ((a.z || 0) + (b.z || 0)) * 0.035, 0.5, 1.08);
    const mx = (a.x + b.x) * 0.5 + Math.sin(t * 0.43 + seed) * unit * 0.021;
    const my = (a.y + b.y) * 0.5 - unit * 0.038 + Math.cos(t * 0.37 + seed) * unit * 0.016;
    ctx.globalCompositeOperation = 'source-over';
    ctx.lineWidth = 1.15;
    ctx.strokeStyle = rgba(rgb, alpha * 0.48 * depthBias);
    ctx.beginPath();
    ctx.moveTo(a.x, a.y);
    ctx.quadraticCurveTo(mx, my, b.x, b.y);
    ctx.stroke();
    if (!packets) return;

    for (let j = 0; j < 3; j++) {
      const u = (t * 0.055 + seed * 0.07 + j / 3) % 1;
      const one = 1 - u;
      const x = one * one * a.x + 2 * one * u * mx + u * u * b.x;
      const y = one * one * a.y + 2 * one * u * my + u * u * b.y;
      const radius = 1.8 + 1.2 * Math.sin(u * Math.PI);
      ctx.fillStyle = rgba(rgb, 0.72 * alpha * depthBias);
      ctx.beginPath();
      ctx.arc(x, y, radius, 0, TAU);
      ctx.fill();
      ctx.strokeStyle = 'rgba(255,255,255,.92)';
      ctx.lineWidth = 0.75;
      ctx.stroke();
    }
  };

  drawOrb = function crispOrb(x, y, radius, config, kind, t, alpha = 1, focus = 0, depth = 0) {
    if (alpha <= 0.002 || radius <= 1) return;
    ctx.save();
    const depthAlpha = clamp(0.76 + depth * 0.045, 0.58, 1.08);
    ctx.globalAlpha = alpha * depthAlpha;
    ctx.globalCompositeOperation = 'source-over';

    // Tight shell: no giant glow field.
    ctx.fillStyle = rgba(config.rgb, 0.075 + focus * 0.025);
    ctx.strokeStyle = rgba(config.rgb, 0.56 + focus * 0.12);
    ctx.lineWidth = Math.max(1.05, radius * 0.012);
    ctx.beginPath();
    ctx.arc(x, y, radius, 0, TAU);
    ctx.fill();
    ctx.stroke();

    // Saturated core with a very small white specular point.
    ctx.fillStyle = rgba(config.rgb, 0.82);
    ctx.beginPath();
    ctx.arc(x, y, radius * 0.24, 0, TAU);
    ctx.fill();
    ctx.fillStyle = 'rgba(255,255,255,.96)';
    ctx.beginPath();
    ctx.arc(x - radius * 0.06, y - radius * 0.07, Math.max(1.5, radius * 0.055), 0, TAU);
    ctx.fill();

    // Crisp orbit geometry.
    ctx.strokeStyle = rgba(config.rgb, 0.48 + focus * 0.1);
    ctx.lineWidth = Math.max(0.9, radius * 0.009);
    for (let i = 0; i < 3; i++) {
      const ringRadius = radius * (1.12 + i * 0.22);
      ctx.save();
      ctx.translate(x, y);
      ctx.rotate(t * (0.08 + i * 0.025) + (kind === 'maker' ? i * 0.65 : i * 0.4));
      ctx.scale(1, 0.34 + i * 0.12);
      ctx.beginPath();
      ctx.arc(0, 0, ringRadius, 0, TAU);
      ctx.stroke();
      ctx.restore();
    }

    if (kind === 'machine') {
      ctx.strokeStyle = rgba(config.rgb, 0.7);
      ctx.lineWidth = 1;
      for (let i = 0; i < 8; i++) {
        const angle = t * 0.1 + i * TAU / 8;
        ctx.beginPath();
        ctx.moveTo(x + Math.cos(angle) * radius * 0.24, y + Math.sin(angle) * radius * 0.24);
        ctx.lineTo(x + Math.cos(angle + 0.55) * radius * 0.66, y + Math.sin(angle + 0.55) * radius * 0.66);
        ctx.stroke();
      }
    } else if (kind === 'maker') {
      ctx.strokeStyle = rgba(config.rgb, 0.72);
      ctx.lineWidth = 1.15;
      ctx.beginPath();
      for (let i = 0; i <= 42; i++) {
        const angle = i / 42 * TAU * 2 + t * 0.2;
        const rr = radius * (0.12 + 0.011 * i);
        const px = x + Math.cos(angle) * rr;
        const py = y + Math.sin(angle) * rr * 0.52;
        i ? ctx.lineTo(px, py) : ctx.moveTo(px, py);
      }
      ctx.stroke();
    } else {
      ctx.strokeStyle = rgba(config.rgb, 0.62);
      ctx.lineWidth = 0.95;
      for (let i = -2; i <= 2; i++) {
        ctx.beginPath();
        ctx.moveTo(x - radius * 0.52, y + i * radius * 0.17);
        ctx.lineTo(x + radius * 0.52, y + i * radius * 0.17);
        ctx.stroke();
        ctx.beginPath();
        ctx.moveTo(x + i * radius * 0.17, y - radius * 0.52);
        ctx.lineTo(x + i * radius * 0.17, y + radius * 0.52);
        ctx.stroke();
      }
    }
    ctx.restore();
  };

  drawOverview = function crispOverview(t, alpha) {
    const positions = overviewPositions(t);
    worldKeys.forEach((key) => Object.assign(worldScreen[key], positions[key]));
    const unity = { x: width * 0.5, y: height * 0.5, r: unit * 0.022, z: 0 };

    // Structural triangle: linework only, never a translucent plane.
    const orderedTriangle = worldKeys.map((key) => positions[key]);
    ctx.globalCompositeOperation = 'source-over';
    ctx.globalAlpha = alpha;
    ctx.strokeStyle = `rgba(15,23,42,${0.075 * alpha})`;
    ctx.lineWidth = 0.85;
    ctx.beginPath();
    ctx.moveTo(orderedTriangle[0].x, orderedTriangle[0].y);
    ctx.lineTo(orderedTriangle[1].x, orderedTriangle[1].y);
    ctx.lineTo(orderedTriangle[2].x, orderedTriangle[2].y);
    ctx.closePath();
    ctx.stroke();

    const pairs = [['machine', 'maker'], ['maker', 'reality'], ['reality', 'machine']];
    pairs.forEach((pair, index) => drawCurve(positions[pair[0]], positions[pair[1]], WORLD[pair[0]].rgb, alpha * 0.9, index + 1, t));
    worldKeys.forEach((key, index) => drawCurve(unity, positions[key], WORLD[key].rgb, alpha, 5 + index, t));

    // Unity becomes a compact amber technical node, not a bloom.
    const unityRgb = [245, 158, 11];
    ctx.fillStyle = rgba(unityRgb, 0.9 * alpha);
    ctx.beginPath();
    ctx.arc(unity.x, unity.y, Math.max(3.2, unit * 0.008), 0, TAU);
    ctx.fill();
    ctx.strokeStyle = rgba(unityRgb, 0.62 * alpha);
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.ellipse(unity.x, unity.y, unit * 0.042 * overviewZoom, unit * 0.014 * overviewZoom, t * 0.08 + overviewRoll, 0, TAU);
    ctx.stroke();
    ctx.beginPath();
    ctx.ellipse(unity.x, unity.y, unit * 0.058 * overviewZoom, unit * 0.02 * overviewZoom, -t * 0.06 - overviewRoll, 0, TAU);
    ctx.stroke();

    const depthOrder = [...worldKeys].sort((a, b) => positions[a].z - positions[b].z);
    depthOrder.forEach((key) => {
      const position = positions[key];
      drawOrb(position.x, position.y, position.r, WORLD[key], key, t, alpha, hoverWorld === key ? 1 : 0, position.z);
    });
    ctx.globalAlpha = 1;
  };

  // Absolute removal of the legacy black radial vignette in part-06.
  render = function crispWhiteRender(t) {
    drawBackground(t);
    const mix = ease(viewMix);
    drawOverview(t, 1 - mix);
    drawDetail(t, mix);
    updateLabels();
    ctx.globalCompositeOperation = 'source-over';
    ctx.globalAlpha = 1;
  };
