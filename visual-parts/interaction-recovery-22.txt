  // Whole Field Orbit 24 — Dream Machine, Dream Maker and Dream World are
  // handles on one rigid 3D instrument. Dragging any body or name rotates the
  // entire simulation exactly as dragging empty space does; no world, thread,
  // membrane or label is translated independently.
  const IR_RELEASE = '20260831-whole-field-orbit-24';
  const irLabelDrag = {
    pointerId: null,
    key: null,
    target: null,
    lastX: 0,
    lastY: 0,
    startX: 0,
    startY: 0,
    lastTime: 0,
    moved: false,
    yawVelocity: 0,
    pitchVelocity: 0,
  };
  let irSuppressClickKey = null;
  let irSuppressClickUntil = 0;

  const irBaseOverviewPositions = overviewPositions;
  const irBaseUpdateLabels = updateLabels;

  // Keep each world centre mathematically on its coloured thread. The complete
  // path remains the original undeformed Catmull–Rom body and is transformed
  // only by the shared overview yaw, pitch, roll and zoom.
  overviewPositions = function wholeFieldPositions(t) {
    const positions = irBaseOverviewPositions(t);
    worldKeys.forEach((key) => {
      const projected = rtRotateProject(rtCatmull(RT_PHASE[key]));
      positions[key].x = projected.x;
      positions[key].y = projected.y;
      positions[key].r = unit * (width < 760 ? 0.064 : 0.075) * projected.f;
      positions[key].f = projected.f;
      positions[key].z = projected.z;
      positions[key].world = projected.world;
    });
    return positions;
  };

  updateLabels = function wholeFieldLabels() {
    irBaseUpdateLabels();
    const clampControl = (element, horizontalPadding, topPadding, bottomPadding) => {
      if (!element || element.style.visibility === 'hidden') return;
      const left = Number.parseFloat(element.style.left);
      const top = Number.parseFloat(element.style.top);
      const halfWidth = Math.max(0, element.offsetWidth * 0.5);
      const halfHeight = Math.max(0, element.offsetHeight * 0.5);
      if (Number.isFinite(left)) element.style.left = `${clamp(left, horizontalPadding + halfWidth, width - horizontalPadding - halfWidth)}px`;
      if (Number.isFinite(top)) element.style.top = `${clamp(top, topPadding + halfHeight, height - bottomPadding - halfHeight)}px`;
    };
    worldKeys.forEach((key) => clampControl(labels[key], 10, 56, 44));
    if (!activeWorld || viewMix <= 0.18 || window.__dreamUnityGameActive) return;
    subLabels.forEach((element) => {
      clampControl(element, 10, 66, 18);
      if (!element) return;
      element.style.visibility = 'visible';
      element.style.zIndex = '14';
    });
  };

  function irScreenSnapshot() {
    return Object.fromEntries(worldKeys.map((key) => {
      const point = worldScreen[key];
      return [key, {
        x: Number(point?.x) || 0,
        y: Number(point?.y) || 0,
        r: Number(point?.r) || 0,
        z: Number(point?.z) || 0,
      }];
    }));
  }

  function irModelSnapshot() {
    return Object.fromEntries(worldKeys.map((key) => {
      const point = rtCatmull(RT_PHASE[key]);
      return [key, { x: point.x, y: point.y, z: point.z }];
    }));
  }

  function irShapeSignature() {
    const model = irModelSnapshot();
    const pair = (a, b) => Math.hypot(
      model[a].x - model[b].x,
      model[a].y - model[b].y,
      model[a].z - model[b].z,
    );
    return {
      machineMaker: pair('machine', 'maker'),
      makerReality: pair('maker', 'reality'),
      realityMachine: pair('reality', 'machine'),
    };
  }

  function irOrientationSnapshot() {
    return {
      yaw: Number(overviewYaw) || 0,
      pitch: Number(overviewPitch) || 0,
      roll: Number(overviewRoll) || 0,
      zoom: Number(overviewZoom) || 1,
    };
  }

  function irAnchorSnapshot() {
    const positions = overviewPositions(time);
    return Object.fromEntries(worldKeys.map((key) => {
      const phase = RT_PHASE[key];
      const thread = rtRotateProject(rtCatmull(phase));
      const before = rtRotateProject(rtCatmull(phase - 0.03));
      const after = rtRotateProject(rtCatmull(phase + 0.03));
      const node = positions[key];
      return [key, {
        node: { x: node.x, y: node.y },
        thread: { x: thread.x, y: thread.y },
        before: { x: before.x, y: before.y },
        after: { x: after.x, y: after.y },
        distance: Math.hypot(node.x - thread.x, node.y - thread.y),
      }];
    }));
  }

  function irThreadSamples(count = 320) {
    const total = clamp(Math.floor(Number(count) || 320), 48, 720);
    return Array.from({ length: total + 1 }, (_, index) => {
      const phase = index / total;
      const point = rtRotateProject(rtCatmull(phase));
      return { phase, x: point.x, y: point.y, z: point.z };
    });
  }

  function irIntegritySnapshot() {
    const anchors = irAnchorSnapshot();
    const samples = irThreadSamples(360);
    let maxSegmentGap = 0;
    let allFinite = true;
    for (let index = 1; index < samples.length; index += 1) {
      const a = samples[index - 1];
      const b = samples[index];
      if (![a.x, a.y, a.z, b.x, b.y, b.z].every(Number.isFinite)) allFinite = false;
      maxSegmentGap = Math.max(maxSegmentGap, Math.hypot(b.x - a.x, b.y - a.y));
    }
    return {
      connected: worldKeys.every((key) => anchors[key].distance < 1),
      rigid: true,
      allFinite,
      maxSegmentGap,
      anchors,
      shape: irShapeSignature(),
    };
  }

  function irRedrawNow() {
    if (window.__dreamUnityGameActive) return;
    render(time);
  }

  function irApplyWholeFieldOrbit(dx, dy, dt = 1 / 60) {
    const safeDt = Math.max(0.008, Number(dt) || 1 / 60);
    overviewYaw -= dx * 0.0065;
    overviewPitch = clamp(overviewPitch - dy * 0.0055, -1.18, 1.18);
    overviewLastInput = time;
    return {
      yawVelocity: clamp(-dx * 0.0065 / safeDt, -9, 9),
      pitchVelocity: clamp(-dy * 0.0055 / safeDt, -8, 8),
    };
  }

  function irBeginLabelDrag(event, key, element) {
    if (!WORLD[key] || activeWorld || viewMix > 0.28 || window.__dreamUnityGameActive) return;
    if (event.pointerType === 'mouse' && event.button !== 0) return;
    irLabelDrag.pointerId = event.pointerId;
    irLabelDrag.key = key;
    irLabelDrag.target = element;
    irLabelDrag.lastX = irLabelDrag.startX = event.clientX;
    irLabelDrag.lastY = irLabelDrag.startY = event.clientY;
    irLabelDrag.lastTime = performance.now();
    irLabelDrag.moved = false;
    irLabelDrag.yawVelocity = 0;
    irLabelDrag.pitchVelocity = 0;
    overviewYawVelocity = 0;
    overviewPitchVelocity = 0;
    overviewRollVelocity = 0;
    overviewLastInput = time;
    app.dataset.draggingWholeField = 'true';
    app.dataset.wholeFieldHandle = key;
    element.classList.add('dragging');
    try { element.setPointerCapture?.(event.pointerId); } catch {}
    if (hint) hint.textContent = `${WORLD[key].name} · ORBITING THE WHOLE FORM`;
    event.stopPropagation();
  }

  function irMoveLabelDrag(event) {
    if (irLabelDrag.pointerId === null || event.pointerId !== irLabelDrag.pointerId || !irLabelDrag.key) return;
    const now = performance.now();
    const dx = event.clientX - irLabelDrag.lastX;
    const dy = event.clientY - irLabelDrag.lastY;
    const dt = Math.max(0.008, (now - irLabelDrag.lastTime) / 1000);
    irLabelDrag.lastX = event.clientX;
    irLabelDrag.lastY = event.clientY;
    irLabelDrag.lastTime = now;
    if (Math.hypot(event.clientX - irLabelDrag.startX, event.clientY - irLabelDrag.startY) > 6) irLabelDrag.moved = true;
    const velocity = irApplyWholeFieldOrbit(dx, dy, dt);
    irLabelDrag.yawVelocity = velocity.yawVelocity;
    irLabelDrag.pitchVelocity = velocity.pitchVelocity;
    // The normal inertia integrator sees no canvas pointer during a label drag,
    // so hold global velocity at zero until release to avoid double motion.
    overviewYawVelocity = 0;
    overviewPitchVelocity = 0;
    overviewRollVelocity = 0;
    irRedrawNow();
    event.preventDefault();
    event.stopImmediatePropagation();
  }

  function irFinishLabelDrag(event, cancelled = false) {
    if (irLabelDrag.pointerId === null || event.pointerId !== irLabelDrag.pointerId || !irLabelDrag.key) return;
    const key = irLabelDrag.key;
    const element = irLabelDrag.target;
    const moved = irLabelDrag.moved || cancelled;
    const yawVelocity = irLabelDrag.yawVelocity;
    const pitchVelocity = irLabelDrag.pitchVelocity;
    try { element?.releasePointerCapture?.(event.pointerId); } catch {}
    element?.classList.remove('dragging');
    delete app.dataset.draggingWholeField;
    delete app.dataset.wholeFieldHandle;
    irLabelDrag.pointerId = null;
    irLabelDrag.key = null;
    irLabelDrag.target = null;
    irLabelDrag.moved = false;
    irLabelDrag.yawVelocity = 0;
    irLabelDrag.pitchVelocity = 0;

    if (moved) {
      overviewYawVelocity = cancelled || reduced ? 0 : yawVelocity;
      overviewPitchVelocity = cancelled || reduced ? 0 : pitchVelocity;
      overviewRollVelocity = 0;
      irSuppressClickKey = key;
      irSuppressClickUntil = performance.now() + 520;
      window.dispatchEvent(new CustomEvent('dreamunity:whole-field-orbited', {
        detail: { handle: key, orientation: irOrientationSnapshot(), connected: true, rigid: true },
      }));
      event.preventDefault();
      event.stopImmediatePropagation();
    }
    irRedrawNow();
    if (hint && !activeWorld) hint.textContent = coarse
      ? 'DRAG ANY WORLD OR EMPTY SPACE TO ORBIT · PINCH TO ZOOM · TAP TO ENTER'
      : 'DRAG ANY WORLD, NAME OR EMPTY SPACE TO ORBIT THE WHOLE FORM · TAP TO ENTER';
  }

  Object.entries(labels).forEach(([key, element]) => {
    if (!element) return;
    element.style.touchAction = 'none';
    element.style.userSelect = 'none';
    element.addEventListener('pointerdown', (event) => irBeginLabelDrag(event, key, element), { passive: false });
    element.addEventListener('click', (event) => {
      if (irSuppressClickKey !== key || performance.now() > irSuppressClickUntil) return;
      irSuppressClickKey = null;
      event.preventDefault();
      event.stopImmediatePropagation();
    }, { capture: true });
  });

  window.addEventListener('pointermove', irMoveLabelDrag, { capture: true, passive: false });
  window.addEventListener('pointerup', (event) => irFinishLabelDrag(event, false), { capture: true, passive: false });
  window.addEventListener('pointercancel', (event) => irFinishLabelDrag(event, true), { capture: true, passive: false });

  const irBaseSetHint = setHint;
  setHint = function wholeFieldHint() {
    irBaseSetHint();
    if (!activeWorld && hint && irLabelDrag.pointerId === null) hint.textContent = coarse
      ? 'DRAG ANY WORLD OR EMPTY SPACE TO ORBIT · PINCH TO ZOOM · TAP TO ENTER'
      : 'DRAG ANY WORLD, NAME OR EMPTY SPACE TO ORBIT THE WHOLE FORM · TAP TO ENTER';
  };

  window.__dreamUnityInteractions = Object.freeze({
    id: 'whole-field-orbit',
    version: IR_RELEASE,
    ready: true,
    connected: true,
    wholeField: true,
    independent: false,
    screen: irScreenSnapshot,
    model: irModelSnapshot,
    shape: irShapeSignature,
    orientation: irOrientationSnapshot,
    anchors: irAnchorSnapshot,
    threadSamples: irThreadSamples,
    integrity: irIntegritySnapshot,
    orbitBy(dx, dy) {
      overviewYawVelocity = 0;
      overviewPitchVelocity = 0;
      overviewRollVelocity = 0;
      irApplyWholeFieldOrbit(Number(dx) || 0, Number(dy) || 0, 1 / 60);
      irRedrawNow();
      return irOrientationSnapshot();
    },
    reset() {
      overviewYaw = 0.18;
      overviewPitch = -0.06;
      overviewRoll = 0;
      overviewZoom = 1;
      overviewYawVelocity = 0;
      overviewPitchVelocity = 0;
      overviewRollVelocity = 0;
      overviewLastInput = time;
      irRedrawNow();
      return irOrientationSnapshot();
    },
  });
  app.dataset.independentWorlds = 'false';
  app.dataset.connectedWorldRig = 'true';
  app.dataset.wholeFieldOrbit = 'true';
  setHint();
