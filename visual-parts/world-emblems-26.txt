  function weRadius(point, key) {
    const responsive = width < 520 ? 0.106 : width < 860 ? 0.104 : width < 1280 ? 0.109 : 0.114;
    const minimum = width < 760 ? 36 : 64;
    const maximum = width < 760 ? 61 : 112;
    return clamp(unit * responsive * point.f * WE_SCALE[key], minimum, maximum);
  }

  overviewPositions = function worldEmblemPositions(t) {
    const positions = weBaseOverviewPositions(t);
    worldKeys.forEach((key) => {
      const point = positions[key];
      if (!point) return;
      point.r = weRadius(point, key);
    });
    return positions;
  };

  function weDrawChassis(position, kind, t, alpha, focus) {
    const radius = position.r * (1 + focus * 0.045);
    const color = WORLD[kind].rgb;
    const motion = reduced ? 0 : t;
    ctx.save();
    ctx.translate(position.x, position.y);
    ctx.lineCap = 'round';
    ctx.lineJoin = 'round';

    const aura = ctx.createRadialGradient(0, 0, radius * 0.18, 0, 0, radius * 1.08);
    aura.addColorStop(0, rgba(RT_WHITE, alpha * 0.50));
    aura.addColorStop(0.54, rgba(color, alpha * (0.035 + focus * 0.025)));
    aura.addColorStop(0.79, rgba(color, alpha * (0.018 + focus * 0.018)));
    aura.addColorStop(1, rgba(color, 0));
    ctx.fillStyle = aura;
    ctx.beginPath();
    ctx.arc(0, 0, radius * 1.08, 0, TAU);
    ctx.fill();

    for (let ring = 0; ring < 3; ring += 1) {
      const rr = radius * (0.81 + ring * 0.085);
      const start = motion * (0.012 + ring * 0.006) + ring * 1.74;
      const span = Math.PI * (0.56 + ring * 0.10);
      ctx.strokeStyle = rgba(ring === 1 ? RT_GOLD : color, alpha * (0.17 + focus * 0.10 - ring * 0.025));
      ctx.lineWidth = ring === 1 ? 0.86 + focus * 0.25 : 0.58 + focus * 0.18;
      ctx.beginPath();
      ctx.arc(0, 0, rr, start, start + span);
      ctx.stroke();
      ctx.beginPath();
      ctx.arc(0, 0, rr, start + Math.PI, start + Math.PI + span * 0.72);
      ctx.stroke();
    }

    for (let index = 0; index < 8; index += 1) {
      const angle = index / 8 * TAU + (kind === 'maker' ? motion * 0.009 : 0);
      const inner = radius * (index % 2 ? 0.90 : 0.94);
      const outer = radius * (index % 2 ? 0.97 : 1.025);
      ctx.strokeStyle = rgba(index % 3 === 1 ? RT_GOLD : color, alpha * (0.22 + focus * 0.10));
      ctx.lineWidth = index % 2 ? 0.52 : 0.78;
      ctx.beginPath();
      ctx.moveTo(Math.cos(angle) * inner, Math.sin(angle) * inner);
      ctx.lineTo(Math.cos(angle) * outer, Math.sin(angle) * outer);
      ctx.stroke();
    }

    if (kind === 'machine') {
      ctx.strokeStyle = rgba(color, alpha * (0.20 + focus * 0.08));
      ctx.lineWidth = 0.60;
      ctx.beginPath();
      ctx.ellipse(0, 0, radius * 0.91, radius * 0.25, 0.38 + motion * 0.012, 0, TAU);
      ctx.stroke();
      ctx.beginPath();
      ctx.ellipse(0, 0, radius * 0.91, radius * 0.25, -0.38 - motion * 0.009, 0, TAU);
      ctx.stroke();
    } else if (kind === 'maker') {
      for (let index = 0; index < 3; index += 1) {
        const start = motion * 0.024 + index * TAU / 3;
        ctx.strokeStyle = rgba(index === 2 ? RT_GOLD : color, alpha * (0.19 + focus * 0.08));
        ctx.lineWidth = index === 1 ? 0.86 : 0.56;
        ctx.beginPath();
        ctx.arc(0, 0, radius * (0.78 + index * 0.055), start, start + Math.PI * 0.72);
        ctx.stroke();
      }
    } else {
      ctx.strokeStyle = rgba(color, alpha * (0.22 + focus * 0.08));
      ctx.lineWidth = 0.68;
      ctx.beginPath();
      for (let index = 0; index <= 6; index += 1) {
        const angle = -Math.PI / 2 + index / 6 * TAU;
        const x = Math.cos(angle) * radius * 0.90;
        const y = Math.sin(angle) * radius * 0.90;
        index ? ctx.lineTo(x, y) : ctx.moveTo(x, y);
      }
      ctx.stroke();
    }
    ctx.restore();
  }

  rtDrawNode = function worldEmblemNode(position, kind, t, alpha = 1, focus = 0) {
    if (!position || position.r <= 1 || alpha <= 0.002) return;
    weDrawChassis(position, kind, t, alpha, focus);
    weBaseDrawNode(position, kind, t, alpha, focus);

    const radius = position.r * (1 + focus * 0.045);
    const color = WORLD[kind].rgb;
    ctx.save();
    ctx.translate(position.x, position.y);
    ctx.strokeStyle = rgba(color, alpha * (0.36 + focus * 0.18));
    ctx.lineWidth = 0.74 + focus * 0.28;
    ctx.setLineDash([radius * 0.055, radius * 0.032]);
    ctx.beginPath();
    ctx.arc(0, 0, radius * 0.985, -Math.PI * 0.44, Math.PI * 0.42);
    ctx.stroke();
    ctx.setLineDash([]);
    ctx.fillStyle = rgba(RT_GOLD, alpha * (0.70 + focus * 0.18));
    ctx.save();
    ctx.rotate(Math.PI / 4);
    ctx.fillRect(-1.65, -1.65 - radius * 0.985, 3.3, 3.3);
    ctx.restore();
    ctx.restore();
  };

  updateLabels = function worldEmblemLabels() {
    weBaseUpdateLabels();
    const overviewAlpha = clamp(1 - ease(viewMix) * 1.42, 0, 1);
    worldKeys.forEach((key) => {
      const element = labels[key];
      const point = worldScreen[key];
      if (!element || !point || !Number.isFinite(point.x) || !Number.isFinite(point.y)) return;
      const diameter = clamp(point.r * 1.52, width < 760 ? 57 : 98, width < 760 ? 92 : 170);
      element.classList.add('world-emblem-label');
      element.style.left = `${point.x}px`;
      element.style.top = `${point.y}px`;
      element.style.setProperty('--we-diameter', `${diameter}px`);
      element.style.setProperty('--we-rgb', WE_RGB[key]);
      element.style.setProperty('--we-title-size', `${clamp(diameter * 0.064, width < 760 ? 5.8 : 7.2, 10.2)}px`);
      element.style.setProperty('--we-subtitle-size', `${clamp(diameter * 0.0305, 3.8, 5.35)}px`);
      element.style.setProperty('--we-index-size', `${clamp(diameter * 0.033, 4.0, 5.8)}px`);
      element.style.setProperty('--we-gap', `${clamp(diameter * 0.047, 3, 7)}px`);
      element.style.setProperty('--we-title-pad', `${clamp(diameter * 0.020, 1.5, 3.2)}px`);
      element.style.opacity = `${overviewAlpha}`;
      element.style.visibility = overviewAlpha > 0.015 ? 'visible' : 'hidden';
      element.style.pointerEvents = !activeWorld && viewMix < 0.28 ? 'auto' : 'none';
      element.style.zIndex = `${Math.round(31 + clamp(point.z, -4, 4) * 2)}`;
    });
  };

  function weMetrics() {
    return Object.fromEntries(worldKeys.map((key) => {
      const point = worldScreen[key];
      const element = labels[key];
      const rect = element?.getBoundingClientRect?.();
      const title = element?.querySelector?.('strong')?.getBoundingClientRect?.();
      return [key, {
        radius: Number(point?.r) || 0,
        world: { x: Number(point?.x) || 0, y: Number(point?.y) || 0 },
        label: rect ? { x: rect.x + rect.width * 0.5, y: rect.y + rect.height * 0.5, width: rect.width, height: rect.height } : null,
        title: title ? { x: title.x + title.width * 0.5, y: title.y + title.height * 0.5, width: title.width, height: title.height } : null,
      }];
    }));
  }
