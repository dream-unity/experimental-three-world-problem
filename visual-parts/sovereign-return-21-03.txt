      ctx.strokeStyle = rgba(index % 2 ? SR_GOLD : SR_CORAL, alpha * (0.10 + pressure * 0.065 + beat * 0.055));
      ctx.lineWidth = 0.72 + pressure * 0.5;
      ctx.beginPath();

      ctx.lineTo(tip.x, tip.y);

    }

      coreGradient.addColorStop(0, rgba(RT_WHITE, alpha * 0.98));
      coreGradient.addColorStop(0.28, rgba(SR_GOLD, alpha * 0.86));
      coreGradient.addColorStop(0.68, rgba(SR_CORAL, alpha * 0.62));
      coreGradient.addColorStop(1, rgba(SR_CORAL, 0));
      ctx.fillStyle = coreGradient;
      ctx.beginPath();
      ctx.arc(origin.x, origin.y, coreRadius * 1.9, 0, TAU);
      ctx.fill();
    }

    // Return leaves matter with memory: the final filament becomes quieter,
    // not blank, as it moves back toward possibility.
    if (release > 0.002) {

          z: Math.sin(angle) * radius * 0.55,
        }, rtLivingYaw, rtLivingPitch, rtLivingRoll);
        srLine(previous, point, u < 0.44 ? SR_GOLD : RT_INK, alpha * release * (1 - u * 0.74) * 0.16, 1.0 - u * 0.45);
        previous = point;
      }
    }

    ctx.restore();
  }
