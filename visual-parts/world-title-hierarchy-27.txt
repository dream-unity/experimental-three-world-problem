  function wthTitleSize(key, diameter) {
    if (width < 760) {
      const factor = key === 'machine' ? 0.090 : key === 'maker' ? 0.096 : 0.097;
      const minimum = key === 'machine' ? 6.7 : 6.9;
      return clamp(diameter * factor, minimum, 9.2);
    }
    const factor = key === 'machine' ? 0.092 : key === 'maker' ? 0.098 : 0.099;
    const minimum = key === 'machine' ? 10.2 : 10.5;
    const maximum = key === 'machine' ? 14.8 : 15.5;
    return clamp(diameter * factor, minimum, maximum);
  }

  updateLabels = function worldTitleHierarchyLabels() {
    wthBaseUpdateLabels();
    worldKeys.forEach((key) => {
      const element = labels[key];
      if (!element) return;
      const diameter = Number.parseFloat(element.style.getPropertyValue('--we-diameter')) || element.offsetWidth || 0;
      if (!Number.isFinite(diameter) || diameter <= 0) return;
      element.style.setProperty('--wth-title-size', `${wthTitleSize(key, diameter)}px`);
      element.style.setProperty('--wth-subtitle-size', `${clamp(diameter * (width < 760 ? 0.046 : 0.0345), width < 760 ? 3.5 : 4.25, width < 760 ? 4.35 : 5.9)}px`);

    });
